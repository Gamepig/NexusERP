**編輯 User 模型**: `/Users/gamepig/projects/NexusERP/frontend/app/Models/User.php`

```php
<?php

namespace App\Models;

// use Illuminate\Contracts\Auth\MustVerifyEmail;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use Laravel\Sanctum\HasApiTokens;
use Illuminate\Support\Str;
use Carbon\Carbon;

class User extends Authenticatable
{
    use HasApiTokens, HasFactory, Notifiable, SoftDeletes;

    /**
     * The attributes that are mass assignable.
     *
     * @var array<int, string>
     */
    protected $fillable = [
        'name',
        'email',
        'password',
        'first_name',
        'last_name', 
        'phone',
        'timezone',
        'locale',
        'is_active',
        'avatar_url',
        'google_id',
        'line_id',
        'avatar',
        'business_type',
        'role',
        'api_token',           // 新增
        'api_token_expires_at', // 新增
    ];

    /**
     * The attributes that should be hidden for serialization.
     *
     * @var array<int, string>
     */
    protected $hidden = [
        'password',
        'remember_token',
        'api_token',           // 新增 - 隱藏敏感資訊
        'first_name_encrypted',
        'last_name_encrypted',
        'email_encrypted',
    ];

    /**
     * The attributes that should be cast.
     *
     * @var array<string, string>
     */
    protected $casts = [
        'email_verified_at' => 'datetime',
        'last_login_at' => 'datetime', 
        'is_active' => 'boolean',
        'api_token_expires_at' => 'datetime', // 新增
        'password' => 'hashed',
    ];

    /**
     * 生成新的 API Token
     *
     * @param int $expirationHours Token 有效期限（小時）
     * @return string
     */
    public function generateApiToken(int $expirationHours = 24 * 30): string
    {
        // 生成安全的隨機 token
        $this->api_token = hash('sha256', Str::random(60) . microtime() . $this->id);
        
        // 設定過期時間 (預設 30 天)
        $this->api_token_expires_at = Carbon::now()->addHours($expirationHours);
        
        $this->save();
        
        return $this->api_token;
    }

    /**
     * 確保用戶有有效的 API Token
     *
     * @return string
     */
    public function ensureApiToken(): string
    {
        // 檢查是否已有有效 token
        if ($this->hasValidApiToken()) {
            return $this->api_token;
        }
        
        // 生成新 token
        return $this->generateApiToken();
    }

    /**
     * 檢查用戶是否有有效的 API Token
     *
     * @return bool
     */
    public function hasValidApiToken(): bool
    {
        return !empty($this->api_token) && 
               ($this->api_token_expires_at === null || 
                $this->api_token_expires_at->isFuture());
    }

    /**
     * 撤銷 API Token
     *
     * @return void
     */
    public function revokeApiToken(): void
    {
        $this->update([
            'api_token' => null,
            'api_token_expires_at' => null,
        ]);
    }

    /**
     * 刷新 API Token (撤銷舊的，生成新的)
     *
     * @param int $expirationHours
     * @return string
     */
    public function refreshApiToken(int $expirationHours = 24 * 30): string
    {
        return $this->generateApiToken($expirationHours);
    }

    /**
     * 檢查 API Token 是否即將過期
     *
     * @param int $hoursBeforeExpiry 過期前幾小時算作即將過期
     * @return bool
     */
    public function isApiTokenExpiringSoon(int $hoursBeforeExpiry = 24): bool
    {
        if (!$this->api_token_expires_at) {
            return false;
        }
        
        return $this->api_token_expires_at->diffInHours(Carbon::now()) <= $hoursBeforeExpiry;
    }

    /**
     * Scope: 查找有有效 API Token 的用戶
     */
    public function scopeWithValidApiToken($query)
    {
        return $query->whereNotNull('api_token')
                    ->where(function($q) {
                        $q->whereNull('api_token_expires_at')
                          ->orWhere('api_token_expires_at', '>', Carbon::now());
                    });
    }

    /**
     * 取得用戶的 API Token 統計資訊
     *
     * @return array
     */
    public function getApiTokenStats(): array
    {
        return [
            'has_token' => !empty($this->api_token),
            'expires_at' => $this->api_token_expires_at,
            'is_expired' => $this->api_token_expires_at && $this->api_token_expires_at->isPast(),
            'expires_in_hours' => $this->api_token_expires_at 
                ? max(0, $this->api_token_expires_at->diffInHours(Carbon::now()))
                : null,
            'is_expiring_soon' => $this->isApiTokenExpiringSoon(),
        ];
    }
}
```

**編輯指令檔案**: `/Users/gamepig/projects/NexusERP/frontend/app/Console/Commands/GenerateApiTokensForUsers.php`

```php
<?php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Models\User;
use Illuminate\Support\Facades\DB;

class GenerateApiTokensForUsers extends Command
{
    /**
     * The name and signature of the console command.
     *
     * @var string
     */
    protected $signature = 'users:generate-api-tokens {--force : 強制為所有用戶重新生成 token}';

    /**
     * The console command description.
     *
     * @var string
     */
    protected $description = '為沒有 API Token 的用戶生成 Token';

    /**
     * Execute the console command.
     */
    public function handle()
    {
        $force = $this->option('force');
        
        if ($force) {
            $this->warn('⚠️  強制模式：將為所有用戶重新生成 API Token');
            if (!$this->confirm('確定要繼續嗎？')) {
                $this->info('操作已取消');
                return 0;
            }
        }

        $this->info('開始為用戶生成 API Token...');
        
        // 取得需要生成 token 的用戶
        $query = User::query();
        
        if (!$force) {
            $query->where(function($q) {
                $q->whereNull('api_token')
                  ->orWhere('api_token', '')
                  ->orWhere('api_token_expires_at', '<', now());
            });
        }
        
        $users = $query->get();
        $totalUsers = $users->count();
        
        if ($totalUsers === 0) {
            $this->info('✅ 所有用戶都已有有效的 API Token');
            return 0;
        }
        
        $this->info("找到 {$totalUsers} 個用戶需要生成 API Token");
        
        $progressBar = $this->output->createProgressBar($totalUsers);
        $progressBar->start();
        
        $successCount = 0;
        $errorCount = 0;
        
        foreach ($users as $user) {
            try {
                DB::beginTransaction();
                
                $user->generateApiToken();
                
                DB::commit();
                $successCount++;
                
            } catch (\Exception $e) {
                DB::rollBack();
                $errorCount++;
                
                $this->newLine();
                $this->error("為用戶 {$user->email} 生成 Token 失敗: " . $e->getMessage());
            }
            
            $progressBar->advance();
        }
        
        $progressBar->finish();
        $this->newLine(2);
        
        // 顯示結果
        $this->table(
            ['狀態', '數量'],
            [
                ['✅ 成功生成', $successCount],
                ['❌ 生成失敗', $errorCount],
                ['📊 總計', $totalUsers],
            ]
        );
        
        if ($successCount > 0) {
            $this->info("✅ 成功為 {$successCount} 個用戶生成 API Token");
        }
        
        if ($errorCount > 0) {
            $this->warn("⚠️  {$errorCount} 個用戶生成失敗，請檢查錯誤日誌");
            return 1;
        }
        
        return 0;
    }
}
```

```php
<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;
use App\Models\User;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Log;
use Carbon\Carbon;

class ApiTokenAuth
{
    /**
     * Handle an incoming request.
     *
     * @param  \Closure(\Illuminate\Http\Request): (\Symfony\Component\HttpFoundation\Response)  $next
     */
    public function handle(Request $request, Closure $next): Response
    {
        // 檢查 Authorization header
        $authHeader = $request->header('Authorization');
        
        if (!$authHeader || !str_starts_with($authHeader, 'Bearer ')) {
            Log::warning('API request missing Authorization header', [
                'url' => $request->url(),
                'ip' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ]);
            
            return response()->json([
                'error' => 'Authorization header required',
                'message' => '需要提供有效的 Authorization header'
            ], 401);
        }

        // 提取 token
        $token = str_replace('Bearer ', '', $authHeader);
        
        if (empty($token)) {
            return response()->json([
                'error' => 'Invalid token format',
                'message' => 'Token 格式不正確'
            ], 401);
        }

        // 查找用戶
        $user = User::where('api_token', $token)->first();

        if (!$user) {
            Log::warning('Invalid API token used', [
                'token_prefix' => substr($token, 0, 8) . '...',
                'url' => $request->url(),
                'ip' => $request->ip(),
            ]);
            
            return response()->json([
                'error' => 'Invalid token',
                'message' => '無效的認證 token'
            ], 401);
        }

        // 檢查 token 是否過期
        if ($user->api_token_expires_at && $user->api_token_expires_at->isPast()) {
            Log::info('Expired API token used', [
                'user_id' => $user->id,
                'expired_at' => $user->api_token_expires_at->toISOString(),
            ]);
            
            return response()->json([
                'error' => 'Token expired',
                'message' => 'Token 已過期，請重新登入',
                'expired_at' => $user->api_token_expires_at->toISOString(),
            ], 401);
        }

        // 檢查用戶是否啟用
        if (!$user->is_active) {
            Log::warning('Inactive user attempted API access', [
                'user_id' => $user->id,
                'email' => $user->email,
            ]);
            
            return response()->json([
                'error' => 'Account disabled',
                'message' => '帳號已被停用'
            ], 403);
        }

        // 登入用戶 (設定 Auth 上下文)
        Auth::login($user);
        
        // 更新最後登入時間
        $user->update(['last_login_at' => Carbon::now()]);
        
        // 檢查 token 是否即將過期，如果是則自動刷新
        if ($user->isApiTokenExpiringSoon(24)) { // 24小時內過期
            $newToken = $user->refreshApiToken();
            
            // 在回應中包含新 token
            $response = $next($request);
            $response->headers->set('X-New-Token', $newToken);
            $response->headers->set('X-Token-Expires-At', $user->api_token_expires_at->toISOString());
            
            Log::info('API token auto-refreshed', [
                'user_id' => $user->id,
                'new_expires_at' => $user->api_token_expires_at->toISOString(),
            ]);
            
            return $response;
        }
        
        // 記錄成功的 API 訪問
        Log::debug('Successful API authentication', [
            'user_id' => $user->id,
            'email' => $user->email,
            'endpoint' => $request->path(),
            'method' => $request->method(),
        ]);

        return $next($request);
    }
}
```

**檔案**: `/Users/gamepig/projects/NexusERP/frontend/app/Http/Controllers/Auth/AuthenticatedSessionController.php`

```php
<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Http\Requests\Auth\LoginRequest;
use App\Providers\RouteServiceProvider;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Log;
use Illuminate\View\View;

class AuthenticatedSessionController extends Controller
{
    /**
     * Display the login view.
     */
    public function create(): View
    {
        return view('auth.login');
    }

    /**
     * Handle an incoming authentication request.
     */
    public function store(LoginRequest $request): RedirectResponse
    {
        $request->authenticate();

        $request->session()->regenerate();
        
        // 確保用戶有有效的 API Token
        $user = Auth::user();
        if ($user) {
            try {
                // 為用戶確保有效的 API Token
                $apiToken = $user->ensureApiToken();
                
                // 記錄成功登入和 Token 生成
                Log::info('User logged in with API token', [
                    'user_id' => $user->id,
                    'email' => $user->email,
                    'ip' => $request->ip(),
                    'user_agent' => $request->userAgent(),
                    'has_new_token' => !empty($apiToken),
                    'token_expires_at' => $user->api_token_expires_at?->toISOString(),
                ]);
                
                // 可選：顯示成功訊息
                session()->flash('success', '登入成功！系統已為您準備好 API 存取權限。');
                
            } catch (\Exception $e) {
                // Token 生成失敗不應阻止登入，但要記錄錯誤
                Log::error('Failed to generate API token on login', [
                    'user_id' => $user->id,
                    'email' => $user->email,
                    'error' => $e->getMessage(),
                    'trace' => $e->getTraceAsString(),
                ]);
                
                session()->flash('warning', '登入成功，但 API 存取權限設定遇到問題。請稍後再試或聯繫管理員。');
            }
        }

        return redirect()->intended(RouteServiceProvider::HOME);
    }

    /**
     * Destroy an authenticated session.
     */
    public function destroy(Request $request): RedirectResponse
    {
        $user = Auth::user();
        
        // 可選：登出時撤銷 API Token (增強安全性)
        if ($user && config('auth.revoke_token_on_logout', false)) {
            try {
                $user->revokeApiToken();
                
                Log::info('API token revoked on logout', [
                    'user_id' => $user->id,
                    'email' => $user->email,
                ]);
                
            } catch (\Exception $e) {
                Log::error('Failed to revoke API token on logout', [
                    'user_id' => $user->id,
                    'error' => $e->getMessage(),
                ]);
            }
        }

        Auth::guard('web')->logout();

        $request->session()->invalidate();

        $request->session()->regenerateToken();

        return redirect('/');
    }
}
```

```php
<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Models\User;
use App\Providers\RouteServiceProvider;
use Illuminate\Auth\Events\Registered;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Log;
use Illuminate\Validation\Rules;
use Illuminate\View\View;

class RegisteredUserController extends Controller
{
    /**
     * Display the registration view.
     */
    public function create(): View
    {
        return view('auth.register');
    }

    /**
     * Handle an incoming registration request.
     *
     * @throws \Illuminate\Validation\ValidationException
     */
    public function store(Request $request): RedirectResponse
    {
        $request->validate([
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'string', 'lowercase', 'email', 'max:255', 'unique:'.User::class],
```

**更新 `callGoAPI` 方法**:
```php
/**
 * 呼叫 Go Backend API
 */
private function callGoAPI($endpoint, $method = 'GET', $data = null)
{
    try {
        $headers = [
            'Content-Type' => 'application/json',
            'Accept' => 'application/json',
        ];

        // 如果用戶已登入，添加認證標頭
        if (Auth::check()) {
            $user = Auth::user();
            
            // 確保用戶有有效的 API Token
            if (!$user->hasValidApiToken()) {
                $user->ensureApiToken();
                Log::info('Auto-generated API token in callGoAPI', [
                    'user_id' => $user->id,
                    'endpoint' => $endpoint,
                ]);
            }

            $headers['Authorization'] = 'Bearer ' . $user->api_token;
            $headers['X-User-ID'] = $user->id;
            $headers['X-User-Email'] = $user->email;
            
            // 添加額外的上下文標頭
            $headers['X-Request-Source'] = 'laravel-frontend';
            $headers['X-Request-ID'] = uniqid();
        }

        $httpClient = Http::withHeaders($headers)->timeout(30);

        // 記錄 API 調用 (除敏感資訊)
        Log::debug('Making Go API call', [
            'endpoint' => $endpoint,
            'method' => $method,
            'has_auth' => isset($headers['Authorization']),
            'user_id' => Auth::id(),
        ]);

        switch (strtoupper($method)) {
            case 'GET':
                $response = $httpClient->get($this->goApiUrl . $endpoint);
                break;
            case 'POST':
                $response = $httpClient->post($this->goApiUrl . $endpoint, $data);
                break;
            case 'PUT':
                $response = $httpClient->put($this->goApiUrl . $endpoint, $data);
                break;
            case 'DELETE':
                $response = $httpClient->delete($this->goApiUrl . $endpoint);
                break;
            default:
                throw new \InvalidArgumentException("Unsupported HTTP method: {$method}");
        }

        // 檢查是否有新的 token (自動刷新)
        $newToken = $response->header('X-New-Token');
        if ($newToken && Auth::check()) {
            Log::info('Received new API token from Go backend', [
                'user_id' => Auth::id(),
            ]);
        }

        return $response;
        
    } catch (\Exception $e) {
        Log::error('Go API call failed', [
            'endpoint' => $endpoint,
            'method' => $method,
            'error' => $e->getMessage(),
            'error_type' => get_class($e),
            'data' => $data,
            'user_id' => Auth::id(),
            'go_api_url' => $this->goApiUrl,
        ]);
        throw $e;
    }
}
```

```php
<?php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Models\User;

class CheckApiTokenStatus extends Command
{
    protected $signature = 'users:check-api-tokens {--user-id= : 檢查特定用戶 ID}';
    protected $description = '檢查用戶 API Token 狀態';

    public function handle()
    {
        $userId = $this->option('user-id');
        
        if ($userId) {
            $user = User::find($userId);
            if (!$user) {
                $this->error("找不到用戶 ID: {$userId}");
                return 1;
            }
            $this->displayUserTokenStatus($user);
        } else {
            $this->displayAllUsersTokenStatus();
        }

        return 0;
    }

    private function displayUserTokenStatus($user)
    {
        $stats = $user->getApiTokenStats();
        
        $this->info("用戶: {$user->name} ({$user->email})");
        $this->table(
            ['屬性', '值'],
            [
                ['用戶 ID', $user->id],
                ['有 Token', $stats['has_token'] ? '✅ 是' : '❌ 否'],
                ['Token 過期時間', $stats['expires_at'] ? $stats['expires_at']->format('Y-m-d H:i:s') : '永不過期'],
                ['是否已過期', $stats['is_expired'] ? '❌ 是' : '✅ 否'],
                ['剩餘時間 (小時)', $stats['expires_in_hours'] ?? '無限制'],
                ['即將過期', $stats['is_expiring_soon'] ? '⚠️ 是' : '✅ 否'],
            ]
        );
    }

    private function displayAllUsersTokenStatus()
    {
        $users = User::all();
        $totalUsers = $users->count();
        $withTokens = $users->filter(fn($u) => $u->hasValidApiToken())->count();
        $expiringSoon = $users->filter(fn($u) => $u->isApiTokenExpiringSoon())->count();

        $this->info("API Token 狀態概覽:");
        $this->table(
            ['統計項目', '數量', '百分比'],
            [
                ['總用戶數', $totalUsers, '100%'],
                ['有有效 Token', $withTokens, round(($withTokens / $totalUsers) * 100, 1) . '%'],
                ['無 Token', $totalUsers - $withTokens, round((($totalUsers - $withTokens) / $totalUsers) * 100, 1) . '%'],
                ['即將過期', $expiringSoon, round(($expiringSoon / $totalUsers) * 100, 1) . '%'],
            ]
        );

        if ($withTokens < $totalUsers) {
            $this->warn("⚠️  有 " . ($totalUsers - $withTokens) . " 個用戶沒有有效的 API Token");
            $this->info("執行以下指令為他們生成 Token:");
            $this->line("php artisan users:generate-api-tokens");
        }

        if ($expiringSoon > 0) {
            $this->warn("⚠️  有 {$expiringSoon} 個用戶的 Token 即將過期");
        }
    }
}
```
