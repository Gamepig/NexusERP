```javascript
/**
 * ProductAutocomplete - 產品搜尋自動完成組件
 * 提供即時產品搜尋、選擇和資料填入功能
 */
class ProductAutocomplete {
    constructor(inputElement, options = {}) {
        this.input = inputElement;
        this.options = {
            minLength: 2,                    // 最小搜尋字元數
            debounceTime: 300,              // 防抖動延遲時間 (毫秒)
            apiUrl: '/api/products/search', // 搜尋 API 端點
            maxResults: 10,                 // 最大顯示結果數
            onSelect: null,                 // 選擇回調函數
            onError: null,                  // 錯誤回調函數
            placeholder: '搜尋產品名稱或編號...', // 佔位符文字
            noResultsText: '找不到相關產品',      // 無結果文字
            loadingText: '搜尋中...',            // 載入中文字
            ...options
        };
        
        // 組件狀態
        this.isOpen = false;
        this.selectedIndex = -1;
        this.results = [];
        this.currentRequest = null;
        
        this.init();
    }

    /**
     * 初始化組件
     */
    init() {
        this.setupInput();
        this.createDropdown();
        this.bindEvents();
        this.setupKeyboardNavigation();
    }

    /**
     * 設置輸入框
     */
    setupInput() {
        this.input.setAttribute('autocomplete', 'off');
        this.input.setAttribute('role', 'combobox');
        this.input.setAttribute('aria-expanded', 'false');
        this.input.setAttribute('aria-haspopup', 'listbox');
        
        if (this.options.placeholder) {
            this.input.setAttribute('placeholder', this.options.placeholder);
        }
    }

    /**
     * 建立下拉選單
     */
    createDropdown() {
        this.dropdown = document.createElement('div');
        this.dropdown.className = 'product-autocomplete-dropdown absolute z-50 w-full bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-600 rounded-md shadow-lg max-h-60 overflow-y-auto';
        this.dropdown.style.display = 'none';
        this.dropdown.setAttribute('role', 'listbox');
        
        // 確保父容器有相對定位
        const parent = this.input.parentNode;
        if (getComputedStyle(parent).position === 'static') {
            parent.style.position = 'relative';
        }
        
        parent.appendChild(this.dropdown);
    }

    /**
     * 綁定事件監聽器
     */
    bindEvents() {
        let searchTimeout;
        
        // 輸入事件 (防抖動搜尋)
        this.input.addEventListener('input', (e) => {
            clearTimeout(searchTimeout);
            const query = e.target.value.trim();
            
            if (query.length === 0) {
                this.hideDropdown();
                return;
            }
            
            if (query.length < this.options.minLength) {
                this.showMessage(this.options.loadingText);
                return;
            }
            
            searchTimeout = setTimeout(() => {
                this.search(query);
            }, this.options.debounceTime);
        });

        // 焦點事件
        this.input.addEventListener('focus', () => {
            if (this.results.length > 0) {
                this.showDropdown();
            }
        });

        // 失焦事件 (延遲隱藏以允許點擊選項)
        this.input.addEventListener('blur', () => {
            setTimeout(() => {
                this.hideDropdown();
            }, 200);
        });

        // 點擊外部關閉
        document.addEventListener('click', (e) => {
            if (!this.input.contains(e.target) && !this.dropdown.contains(e.target)) {
                this.hideDropdown();
            }
        });
    }

    /**
     * 設置鍵盤導航
     */
    setupKeyboardNavigation() {
        this.input.addEventListener('keydown', (e) => {
            if (!this.isOpen) return;

            switch (e.key) {
                case 'ArrowDown':
                    e.preventDefault();
                    this.navigateDown();
                    break;
                case 'ArrowUp':
                    e.preventDefault();
                    this.navigateUp();
                    break;
                case 'Enter':
                    e.preventDefault();
                    if (this.selectedIndex >= 0 && this.results[this.selectedIndex]) {
                        this.selectProduct(this.results[this.selectedIndex]);
                    }
                    break;
                case 'Escape':
                    e.preventDefault();
                    this.hideDropdown();
                    break;
            }
        });
    }

    /**
     * 向下導航
     */
    navigateDown() {
        this.selectedIndex = Math.min(this.selectedIndex + 1, this.results.length - 1);
        this.updateSelection();
    }

    /**
     * 向上導航
     */
    navigateUp() {
        this.selectedIndex = Math.max(this.selectedIndex - 1, -1);
        this.updateSelection();
    }

    /**
     * 更新選中狀態顯示
     */
    updateSelection() {
        const items = this.dropdown.querySelectorAll('.autocomplete-item');
        items.forEach((item, index) => {
            if (index === this.selectedIndex) {
                item.classList.add('bg-indigo-50', 'dark:bg-indigo-900');
                item.setAttribute('aria-selected', 'true');
            } else {
                item.classList.remove('bg-indigo-50', 'dark:bg-indigo-900');
                item.setAttribute('aria-selected', 'false');
            }
        });
    }

    /**
     * 執行產品搜尋
     */
    async search(query) {
        try {
            // 取消前一個請求
            if (this.currentRequest) {
                this.currentRequest.abort();
            }

            // 顯示載入狀態
            this.showMessage(this.options.loadingText);

            // 建立新請求
            const controller = new AbortController();
            this.currentRequest = controller;

            const url = new URL(this.options.apiUrl, window.location.origin);
            url.searchParams.set('q', query);
            url.searchParams.set('limit', this.options.maxResults);

            const response = await fetch(url, {
                signal: controller.signal,
                headers: {
                    'Accept': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest',
                    'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.getAttribute('content') || ''
                }
            });

            if (!response.ok) {
                throw new Error(`HTTP ${response.status}: ${response.statusText}`);
            }

            const data = await response.json();
            this.currentRequest = null;

            if (data.products && data.products.length > 0) {
                this.results = data.products;
                this.showResults(data.products);
            } else {
                this.results = [];
                this.showMessage(this.options.noResultsText);
            }

        } catch (error) {
            this.currentRequest = null;
            
            if (error.name === 'AbortError') {
                return; // 請求被取消，忽略
            }

            console.error('Product search failed:', error);
            this.showMessage('搜尋失敗，請稍後再試');
            
            if (this.options.onError) {
                this.options.onError(error);
            }
        }
    }

    /**
     * 顯示搜尋結果
     */
    showResults(products) {
        this.dropdown.innerHTML = '';
        this.selectedIndex = -1;

        products.forEach((product, index) => {
            const item = this.createProductItem(product, index);
            this.dropdown.appendChild(item);
        });

        this.showDropdown();
    }

    /**
     * 建立產品項目元素
     */
    createProductItem(product, index) {
        const item = document.createElement('div');
        item.className = 'autocomplete-item px-4 py-3 hover:bg-gray-100 dark:hover:bg-gray-700 cursor-pointer border-b border-gray-200 dark:border-gray-600 last:border-b-0';
        item.setAttribute('role', 'option');
        item.setAttribute('aria-selected', 'false');
        
        // 產品資訊 HTML
        const imageUrl = product.image_url || '/images/product-placeholder.png';
        const stockStatus = this.getStockStatusDisplay(product);
        
        item.innerHTML = `
            <div class="flex items-center space-x-3">
                <div class="flex-shrink-0">
                    <img class="h-10 w-10 rounded-md object-cover" 
                         src="${imageUrl}" 
                         alt="${product.name}"
                         onerror="this.src='/images/product-placeholder.png'">
                </div>
                <div class="flex-1 min-w-0">
                    <div class="font-medium text-gray-900 dark:text-white truncate">
                        ${this.highlightMatch(product.name, this.input.value)}
                    </div>
                    <div class="text-sm text-gray-500 dark:text-gray-400 flex items-center space-x-4">
                        <span>編號: ${product.sku || 'N/A'}</span>
                        <span>價格: $${this.formatPrice(product.unit_price)}</span>
                        ${stockStatus}
                    </div>
                    ${product.description ? `<div class="text-xs text-gray-400 dark:text-gray-500 mt-1 truncate">${product.description}</div>` : ''}
                </div>
            </div>
        `;
        
        // 點擊事件
        item.addEventListener('click', (e) => {
            e.preventDefault();
            this.selectProduct(product);
        });

        // 滑鼠懸停事件
        item.addEventListener('mouseenter', () => {
            this.selectedIndex = index;
            this.updateSelection();
        });

        return item;
    }

    /**
     * 取得庫存狀態顯示
     */
    getStockStatusDisplay(product) {
        if (product.stock_quantity === undefined) {
            return '<span class="text-gray-400">庫存: 未知</span>';
        }

        const stock = parseInt(product.stock_quantity);
        if (stock > 10) {
            return '<span class="text-green-600">現貨充足</span>';
        } else if (stock > 0) {
            return `<span class="text-yellow-600">庫存: ${stock}</span>`;
        } else {
            return '<span class="text-red-600">缺貨</span>';
        }
    }

    /**
     * 格式化價格顯示
     */
    formatPrice(price) {
        if (price === undefined || price === null) {
            return '0.00';
        }
        return parseFloat(price).toLocaleString('en-US', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
    }

    /**
     * 高亮搜尋匹配文字
     */
    highlightMatch(text, query) {
        if (!query) return text;
        
        const regex = new RegExp(`(${query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
        return text.replace(regex, '<mark class="bg-yellow-200 dark:bg-yellow-800">$1</mark>');
    }

    /**
     * 顯示訊息 (載入中、無結果等)
     */
    showMessage(message) {
        this.dropdown.innerHTML = `
            <div class="px-4 py-3 text-sm text-gray-500 dark:text-gray-400 text-center">
                ${message}
            </div>
        `;
        this.showDropdown();
    }

    /**
     * 選擇產品
     */
    selectProduct(product) {
        // 設置輸入框值
        this.input.value = product.name;
        this.input.setAttribute('data-product-id', product.id);
        
        // 隱藏下拉選單
        this.hideDropdown();
        
        // 觸發自定義事件
        const event = new CustomEvent('product-selected', {
            detail: { product, input: this.input }
        });
        this.input.dispatchEvent(event);
        
        // 調用回調函數
        if (this.options.onSelect) {
            this.options.onSelect(product, this.input);
        }
    }

    /**
     * 顯示下拉選單
     */
    showDropdown() {
        this.dropdown.style.display = 'block';
        this.isOpen = true;
        this.input.setAttribute('aria-expanded', 'true');
    }

    /**
     * 隱藏下拉選單
     */
    hideDropdown() {
        this.dropdown.style.display = 'none';
        this.isOpen = false;
        this.selectedIndex = -1;
        this.input.setAttribute('aria-expanded', 'false');
    }

    /**
     * 清除搜尋結果
     */
    clear() {
        this.input.value = '';
        this.input.removeAttribute('data-product-id');
        this.hideDropdown();
        this.results = [];
    }

    /**
     * 銷毀組件
     */
    destroy() {
        if (this.currentRequest) {
            this.currentRequest.abort();
        }
        
        if (this.dropdown && this.dropdown.parentNode) {
            this.dropdown.parentNode.removeChild(this.dropdown);
        }
        
        // 移除所有事件監聽器
        this.input.replaceWith(this.input.cloneNode(true));
    }
}

// 全域註冊
window.ProductAutocomplete = ProductAutocomplete;
```

```php
<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Auth;

class ProductController extends Controller
{
    private $goApiUrl;

    public function __construct()
    {
        $this->goApiUrl = config('services.go_backend.url', 'http://localhost:8082');
        $this->middleware('auth');
    }

    /**
     * 搜尋產品 API
     */
    public function search(Request $request)
    {
        // 驗證請求參數
        $validated = $request->validate([
            'q' => 'required|string|min:2|max:100',
            'limit' => 'integer|min:1|max:50',
            'category_id' => 'integer|min:1',
            'in_stock_only' => 'boolean'
        ]);

        try {
            // 準備查詢參數
            $queryParams = array_filter([
                'search' => $validated['q'],
                'limit' => $validated['limit'] ?? 10,
                'category_id' => $validated['category_id'] ?? null,
                'in_stock_only' => $validated['in_stock_only'] ?? false
            ], function($value) {
                return $value !== null && $value !== '';
            });

            // 調用 Go API
            $response = $this->callGoAPI('/api/products/search?' . http_build_query($queryParams));

            if ($response->successful()) {
                $data = $response->json();
                
                return response()->json([
                    'success' => true,
                    'products' => $data['products'] ?? [],
                    'total' => $data['total'] ?? 0,
                    'query' => $validated['q']
                ]);
            }

            // API 調用失敗
            Log::warning('Product search API failed', [
                'status' => $response->status(),
                'body' => $response->body(),
                'query' => $validated['q']
            ]);

            return response()->json([
                'success' => false,
                'message' => '搜尋服務暫時無法使用',
                'products' => []
            ], 503);

        } catch (\Exception $e) {
            Log::error('Product search exception: ' . $e->getMessage(), [
                'query' => $validated['q'],
                'trace' => $e->getTraceAsString()
            ]);

            return response()->json([
                'success' => false,
                'message' => '搜尋時發生錯誤',
                'products' => []
            ], 500);
        }
    }

    /**
     * 調用 Go Backend API
     */
    private function callGoAPI($endpoint, $method = 'GET', $data = null)
    {
        try {
            $headers = [
                'Content-Type' => 'application/json',
                'Accept' => 'application/json',
            ];

            // 添加認證標頭
            if (Auth::check()) {
                $headers['Authorization'] = 'Bearer ' . Auth::user()->api_token;
                $headers['X-User-ID'] = Auth::id();
                $headers['X-User-Email'] = Auth::user()->email;
            }

            return Http::withHeaders($headers)
                      ->timeout(10)
                      ->$method($this->goApiUrl . $endpoint, $data);

        } catch (\Exception $e) {
            Log::error('Go API call failed', [
                'endpoint' => $endpoint,
                'method' => $method,
                'error' => $e->getMessage()
            ]);
            throw $e;
        }
    }
}
```

```javascript
/**
 * 驗證報價項目
 */
function validateQuoteItems() {
    const itemRows = document.querySelectorAll('.item-row');
    let isValid = true;
    const errors = [];

    itemRows.forEach((row, index) => {
        const nameInput = row.querySelector('.product-search-input');
        const quantityInput = row.querySelector('.quantity-input');
        const priceInput = row.querySelector('.price-input');

        // 檢查必填欄位
        if (!nameInput?.value.trim()) {
            errors.push(`項目 ${index + 1}: 產品名稱為必填`);
            nameInput?.classList.add('border-red-500');
            isValid = false;
        } else {
            nameInput?.classList.remove('border-red-500');
        }

        if (!quantityInput?.value || parseFloat(quantityInput.value) <= 0) {
            errors.push(`項目 ${index + 1}: 數量必須大於 0`);
            quantityInput?.classList.add('border-red-500');
            isValid = false;
        } else {
            quantityInput?.classList.remove('border-red-500');
        }

        if (!priceInput?.value || parseFloat(priceInput.value) < 0) {
            errors.push(`項目 ${index + 1}: 單價不能為負數`);
            priceInput?.classList.add('border-red-500');
            isValid = false;
        } else {
            priceInput?.classList.remove('border-red-500');
        }
    });

    if (!isValid) {
        showNotification(errors.join('<br>'), 'error');
    }

    return isValid;
}

/**
 * 批量操作項目
 */
function bulkRemoveItems() {
    const checkboxes = document.querySelectorAll('.item-checkbox:checked');
    
    if (checkboxes.length === 0) {
        showNotification('請選擇要刪除的項目', 'warning');
        return;
    }
    
    const itemsList = document.getElementById('itemsList');
    if (itemsList.children.length - checkboxes.length < 1) {
        showNotification('至少需要保留一個報價項目', 'warning');
        return;
    }
    
    if (confirm(`確定要刪除選中的 ${checkboxes.length} 個項目嗎？`)) {
        checkboxes.forEach(checkbox => {
            const itemRow = checkbox.closest('.item-row');
            itemRow?.remove();
        });
        
        updateTotals();
        showNotification(`已刪除 ${checkboxes.length} 個項目`, 'success');
    }
}

/**
 * 複製項目
 */
function duplicateItem(button) {
    const itemRow = button.closest('.item-row');
    const newRow = itemRow.cloneNode(true);
    
    // 更新名稱屬性中的索引
    newRow.querySelectorAll('input, textarea, select').forEach(input => {
        if (input.name && input.name.includes('[')) {
            input.name = input.name.replace(/\[\d+\]/, `[${itemIndex}]`);
        }
        
        // 清除隱藏的產品 ID (需要重新選擇)
        if (input.classList.contains('product-id-input')) {
            input.value = '';
        }
    });
    
    // 重新初始化產品自動完成
    const productInput = newRow.querySelector('.product-search-input');
    if (productInput) {
        initializeSingleProductAutocomplete(productInput);
    }
    
    // 插入到當前項目後面
    itemRow.parentNode.insertBefore(newRow, itemRow.nextSibling);
    
    itemIndex++;
    updateTotals();
    showNotification('項目已複製', 'success');
}
```

```javascript
function updateItemSubtotal(input) {
    const itemRow = input.closest('.item-row');
    if (!itemRow) return;
    
    const quantityInput = itemRow.querySelector('.quantity-input');
    const priceInput = itemRow.querySelector('.price-input');
    const discountInput = itemRow.querySelector('.discount-input');
    const subtotalElement = itemRow.querySelector('.item-subtotal');
    
    // 取得數值
    const quantity = parseFloat(quantityInput?.value) || 0;
    const price = parseFloat(priceInput?.value) || 0;
    const discountPercent = parseFloat(discountInput?.value) || 0;
    
    // 計算小計
    const baseAmount = quantity * price;
    const discountAmount = baseAmount * (discountPercent / 100);
    const subtotal = baseAmount - discountAmount;
    
    // 更新顯示
    if (subtotalElement) {
        subtotalElement.textContent = '$' + subtotal.toLocaleString('en-US', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
    }
    
    // 更新總計
    updateTotals();
    
    // 視覺反饋
    if (subtotal > 0) {
        subtotalElement?.parentElement.classList.add('bg-green-50', 'dark:bg-green-900');
        setTimeout(() => {
            subtotalElement?.parentElement.classList.remove('bg-green-50', 'dark:bg-green-900');
        }, 1000);
    }
}

 * 更新總計金額

function updateTotals() {
    const itemRows = document.querySelectorAll('.item-row');
    let itemsSubtotal = 0;
    let totalDiscount = 0;
    
    itemRows.forEach(row => {
        const quantityInput = row.querySelector('.quantity-input');
        const priceInput = row.querySelector('.price-input');
        const discountInput = row.querySelector('.discount-input');
        
        const quantity = parseFloat(quantityInput?.value) || 0;
        const price = parseFloat(priceInput?.value) || 0;
        const discountPercent = parseFloat(discountInput?.value) || 0;
        
        const baseAmount = quantity * price;
        const discountAmount = baseAmount * (discountPercent / 100);
        
        itemsSubtotal += baseAmount;
        totalDiscount += discountAmount;
    });
    
    const finalTotal = itemsSubtotal - totalDiscount;
    
    // 更新顯示元素
    const itemsSubtotalElement = document.getElementById('itemsSubtotal');
    const totalDiscountElement = document.getElementById('totalDiscount');
    const finalTotalElement = document.getElementById('finalTotal');
    const totalAmountElement = document.getElementById('totalAmount');
    
    if (itemsSubtotalElement) {
        itemsSubtotalElement.textContent = '$' + itemsSubtotal.toLocaleString('en-US', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
    }
    
    if (totalDiscountElement) {
        totalDiscountElement.textContent = '-$' + totalDiscount.toLocaleString('en-US', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
    }
    
    if (finalTotalElement) {
        finalTotalElement.textContent = '$' + finalTotal.toLocaleString('en-US', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
    }
    
    if (totalAmountElement) {
        totalAmountElement.textContent = '$' + finalTotal.toLocaleString('en-US', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
    }
    
    // 更新總計顏色
    if (finalTotal > 1000) {
        totalAmountElement?.classList.add('text-green-600', 'dark:text-green-400');
        totalAmountElement?.classList.remove('text-indigo-600', 'dark:text-indigo-400');
    } else {
        totalAmountElement?.classList.remove('text-green-600', 'dark:text-green-400');
        totalAmountElement?.classList.add('text-indigo-600', 'dark:text-indigo-400');
    }
}

/**
 * 格式化數字輸入 (即時格式化)
 */
function formatNumberInput(input) {
    let value = input.value.replace(/[^\d.]/g, '');
    
    // 確保只有一個小數點
    const parts = value.split('.');
    if (parts.length > 2) {
        value = parts[0] + '.' + parts.slice(1).join('');
    }
    
    // 限制小數位數
    if (parts[1] && parts[1].length > 2) {
        value = parts[0] + '.' + parts[1].substring(0, 2);
    }
    
    input.value = value;
}

function exportItemsToCSV() {
    const itemRows = document.querySelectorAll('.item-row');
    const csvData = [
        ['產品名稱', '描述', '數量', '單價', '折扣%', '小計']
    ];
    
    itemRows.forEach(row => {
        const name = row.querySelector('.product-search-input')?.value || '';
        const description = row.querySelector('.description-input')?.value || '';
        const quantity = row.querySelector('.quantity-input')?.value || 0;
        const price = row.querySelector('.price-input')?.value || 0;
        const discount = row.querySelector('.discount-input')?.value || 0;
        
        const baseAmount = parseFloat(quantity) * parseFloat(price);
        const discountAmount = baseAmount * (parseFloat(discount) / 100);
        const subtotal = baseAmount - discountAmount;
        
        csvData.push([name, description, quantity, price, discount, subtotal.toFixed(2)]);
    });
    
    // 建立 CSV 內容
    const csvContent = csvData.map(row => 
        row.map(field => `"${field}"`).join(',')
    ).join('\n');
    
    // 下載檔案
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    const url = URL.createObjectURL(blob);
    link.setAttribute('href', url);
    link.setAttribute('download', `quote_items_${new Date().toISOString().split('T')[0]}.csv`);
    link.style.visibility = 'hidden';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    
    showNotification('CSV 檔案已下載', 'success');
}
```
