<?php

namespace Tests\Feature;

use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;
use App\Models\User;
use Carbon\Carbon;

class ApiTokenAuthTest extends TestCase
{
    use RefreshDatabase;

    /**
     * 測試有效 Token 認證成功
     */
    public function test_valid_token_authentication_success()
    {
        $user = User::factory()->create();
        $token = $user->generateApiToken();

        $response = $this->withHeaders([
            'Authorization' => 'Bearer ' . $token,
        ])->getJson('/api/health');

        $response->assertStatus(200);
    }

    /**
     * 測試無效 Token 被拒絕
     */
    public function test_invalid_token_rejected()
    {
        $response = $this->withHeaders([
            'Authorization' => 'Bearer invalid-token',
        ])->getJson('/api/health');

        $response->assertStatus(401);
        $response->assertJson(['error' => 'Invalid token']);
    }

    /**
     * 測試過期 Token 被拒絕
     */
    public function test_expired_token_rejected()
    {
        $user = User::factory()->create();
        $user->generateApiToken();
        
        // 手動設定 Token 為過期
        $user->update(['api_token_expires_at' => Carbon::now()->subHour()]);

        $response = $this->withHeaders([
            'Authorization' => 'Bearer ' . $user->api_token,
        ])->getJson('/api/health');

        $response->assertStatus(401);
        $response->assertJson(['error' => 'Token expired']);
    }

    /**
     * 測試缺少認證標頭
     */
    public function test_missing_authorization_header()
    {
        $response = $this->getJson('/api/health');

        $response->assertStatus(401);
        $response->assertJson(['error' => 'Authorization header required']);
    }

    /**
     * 測試已停用用戶被拒絕
     */
    public function test_inactive_user_rejected()
    {
        $user = User::factory()->create(['is_active' => false]);
        $token = $user->generateApiToken();

        $response = $this->withHeaders([
            'Authorization' => 'Bearer ' . $token,
        ])->getJson('/api/health');

        $response->assertStatus(403);
        $response->assertJson(['error' => 'Account disabled']);
    }
}
