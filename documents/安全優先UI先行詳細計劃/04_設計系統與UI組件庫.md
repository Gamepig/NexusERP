#### **按鈕組件系統**
```php
{{-- 檔案: resources/views/components/ui/button.blade.php --}}

@props([
    'variant' => 'primary',    // primary, secondary, outline, ghost, danger
    'size' => 'md',           // xs, sm, md, lg, xl
    'loading' => false,       // 載入狀態
    'disabled' => false,      // 禁用狀態
    'icon' => null,          // 圖示名稱
    'iconPosition' => 'left', // left, right
    'fullWidth' => false,     // 全寬度
    'type' => 'button',       // button, submit, reset
    'href' => null,           // 連結模式
])

@php
// 基礎樣式
$baseClasses = 'inline-flex items-center justify-center font-medium transition-all duration-200 focus-ring disabled:opacity-50 disabled:cursor-not-allowed';

// 尺寸樣式
$sizeClasses = [
    'xs' => 'px-2.5 py-1.5 text-xs rounded-sm gap-1',
    'sm' => 'px-3 py-2 text-sm rounded-md gap-1.5',
    'md' => 'px-4 py-2.5 text-sm rounded-md gap-2',
    'lg' => 'px-5 py-3 text-base rounded-lg gap-2',
    'xl' => 'px-6 py-3.5 text-base rounded-lg gap-2.5',
];

// 變體樣式
$variantClasses = [
    'primary' => 'bg-primary-600 hover:bg-primary-700 active:bg-primary-800 text-white shadow-sm hover:shadow-md',
    'secondary' => 'bg-secondary-100 hover:bg-secondary-200 active:bg-secondary-300 text-secondary-900 dark:bg-secondary-800 dark:hover:bg-secondary-700 dark:text-secondary-100',
    'outline' => 'border-2 border-primary-600 text-primary-600 hover:bg-primary-50 active:bg-primary-100 dark:border-primary-400 dark:text-primary-400 dark:hover:bg-primary-950',
    'ghost' => 'text-primary-600 hover:bg-primary-50 active:bg-primary-100 dark:text-primary-400 dark:hover:bg-primary-950',
    'danger' => 'bg-danger-600 hover:bg-danger-700 active:bg-danger-800 text-white shadow-sm hover:shadow-md',
];

// 組合最終樣式
$classes = implode(' ', [
    $baseClasses,
    $sizeClasses[$size] ?? $sizeClasses['md'],
    $variantClasses[$variant] ?? $variantClasses['primary'],
    $fullWidth ? 'w-full' : '',
    $loading ? 'loading' : '',
]);

// 圖示大小映射
$iconSizes = [
    'xs' => 'w-3 h-3',
    'sm' => 'w-4 h-4', 
    'md' => 'w-4 h-4',
    'lg' => 'w-5 h-5',
    'xl' => 'w-5 h-5',
];

$iconSize = $iconSizes[$size] ?? $iconSizes['md'];
@endphp

@if($href)
    {{-- 連結模式 --}}
    <a href="{{ $href }}" {{ $attributes->merge(['class' => $classes]) }}>
        @if($loading)
            <x-ui.spinner :size="$size" />
        @else
            @if($icon && $iconPosition === 'left')
                <x-ui.icon :name="$icon" :class="$iconSize" />
            @endif
            
            {{ $slot }}
            
            @if($icon && $iconPosition === 'right') 
                <x-ui.icon :name="$icon" :class="$iconSize" />
            @endif
        @endif
    </a>
@else
    {{-- 按鈕模式 --}}
    <button 
        type="{{ $type }}"
        @disabled($disabled || $loading)
        {{ $attributes->merge(['class' => $classes]) }}
    >
        @if($loading)
            <x-ui.spinner :size="$size" />
            <span>載入中...</span>
        @else
            @if($icon && $iconPosition === 'left')
                <x-ui.icon :name="$icon" :class="$iconSize" />
            @endif
            
            {{ $slot }}
            
            @if($icon && $iconPosition === 'right')
                <x-ui.icon :name="$icon" :class="$iconSize" />
            @endif
        @endif
    </button>
@endif

{{-- 使用範例:
<x-ui.button variant="primary" size="lg" icon="plus">
    新增產品
</x-ui.button>

<x-ui.button variant="outline" loading="true">
    儲存中
</x-ui.button>

<x-ui.button variant="danger" icon="trash" icon-position="left">
    刪除
</x-ui.button>
--}}
```

```javascript
// 檔案: resources/js/theme.js

class ThemeManager {
    constructor() {
        this.currentTheme = this.getStoredTheme() || this.getSystemPreference();
        this.initializeTheme();
        this.bindEvents();
    }
    
    // 獲取儲存的主題
    getStoredTheme() {
        return localStorage.getItem('nx-theme');
    }
    
    // 獲取系統偏好
    getSystemPreference() {
        return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
    }
    
    // 初始化主題
    initializeTheme() {
        this.applyTheme(this.currentTheme);
    }
    
    // 應用主題
    applyTheme(theme) {
        const html = document.documentElement;
        
        // 移除現有主題類
        html.classList.remove('light', 'dark');
        
        // 添加新主題類
        html.classList.add(theme);
        
        // 更新狀態
        this.currentTheme = theme;
        
        // 儲存到本地儲存
        localStorage.setItem('nx-theme', theme);
        
        // 觸發主題變更事件
        this.dispatchThemeChange(theme);
        
        // 更新meta標籤 (行動裝置狀態列)
        this.updateMetaThemeColor(theme);
    }
    
    // 切換主題
    toggleTheme() {
        const newTheme = this.currentTheme === 'dark' ? 'light' : 'dark';
        this.applyTheme(newTheme);
        return newTheme;
    }
    
    // 設定特定主題
    setTheme(theme) {
        if (['light', 'dark', 'system'].includes(theme)) {
            if (theme === 'system') {
                theme = this.getSystemPreference();
            }
            this.applyTheme(theme);
        }
    }
    
    // 綁定事件
    bindEvents() {
        // 監聽系統主題變更
        window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', (e) => {
            if (!this.getStoredTheme()) {
                this.applyTheme(e.matches ? 'dark' : 'light');
            }
        });
        
        // 監聽儲存變更 (多標籤頁同步)
        window.addEventListener('storage', (e) => {
            if (e.key === 'nx-theme') {
                this.applyTheme(e.newValue || this.getSystemPreference());
            }
        });
    }
    
    // 觸發主題變更事件
    dispatchThemeChange(theme) {
        window.dispatchEvent(new CustomEvent('theme-changed', {
            detail: { theme }
        }));
    }
    
    // 更新meta主題顏色
    updateMetaThemeColor(theme) {
        const metaThemeColor = document.querySelector('meta[name="theme-color"]');
        if (metaThemeColor) {
            const colors = {
                light: '#ffffff',
                dark: '#0f172a'
            };
            metaThemeColor.setAttribute('content', colors[theme]);
        }
    }
    
    // 獲取當前主題
    getCurrentTheme() {
        return this.currentTheme;
    }
    
    // 檢查是否為深色主題
    isDark() {
        return this.currentTheme === 'dark';
    }
    
    // 檢查是否為淺色主題
    isLight() {
        return this.currentTheme === 'light';
    }
}

// 初始化主題管理器
const themeManager = new ThemeManager();

// 全域訪問
window.themeManager = themeManager;

// 匯出供其他模組使用
export default themeManager;
```

```javascript
// 檔案: resources/js/tests/theme.test.js

import themeManager from '../theme.js';

describe('ThemeManager', () => {
    beforeEach(() => {
        localStorage.clear();
        document.documentElement.className = '';
    });
    
    test('should initialize with system preference', () => {
        Object.defineProperty(window, 'matchMedia', {
            writable: true,
            value: jest.fn().mockImplementation(query => ({
                matches: query === '(prefers-color-scheme: dark)',
                addEventListener: jest.fn(),
                removeEventListener: jest.fn(),
            })),
        });
        
        const manager = new ThemeManager();
        expect(manager.getCurrentTheme()).toBe('dark');
    });
    
    test('should toggle theme correctly', () => {
        const manager = new ThemeManager();
        manager.setTheme('light');
        
        const newTheme = manager.toggleTheme();
        
        expect(newTheme).toBe('dark');
        expect(document.documentElement.classList.contains('dark')).toBe(true);
    });
    
    test('should persist theme to localStorage', () => {
        const manager = new ThemeManager();
        manager.setTheme('dark');
        
        expect(localStorage.getItem('nx-theme')).toBe('dark');
    });
});
```
