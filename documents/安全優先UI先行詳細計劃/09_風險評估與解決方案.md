## 🔧 **風險監控系統實作**

### **自動化風險檢測服務**
```php
<?php
// 檔案: app/Services/RiskMonitoringService.php

namespace App\Services;

use App\Models\{RiskEvent, SystemMetric, SecurityAlert};
use Illuminate\Support\Facades\{Log, Cache, DB};

class RiskMonitoringService
{
    private array $riskThresholds = [
        'security' => [
            'failed_login_attempts' => 10,
            'unusual_api_calls' => 100,
            'suspicious_ip_requests' => 50,
        ],
        'performance' => [
            'response_time_ms' => 2000,
            'error_rate_percent' => 5,
            'cpu_usage_percent' => 85,
            'memory_usage_percent' => 90,
        ],
        'business' => [
            'daily_revenue_drop_percent' => 20,
            'user_activity_drop_percent' => 30,
            'conversion_rate_drop_percent' => 25,
        ],
    ];
    
    /**
     * 安全風險監控
     */
    public function monitorSecurityRisks(): array
    {
        $risks = [];
        
        // 1. 異常登入檢測
        $failedLogins = $this->getFailedLoginAttempts();
        if ($failedLogins > $this->riskThresholds['security']['failed_login_attempts']) {
            $risks[] = $this->createRiskEvent([
                'type' => 'security',
                'category' => 'authentication',
                'severity' => 'high',
                'title' => '異常登入嘗試檢測',
                'description' => "檢測到 {$failedLogins} 次失敗登入嘗試",
                'metrics' => ['failed_logins' => $failedLogins],
                'recommended_actions' => [
                    '檢查IP來源是否為攻擊',
                    '考慮暫時封鎖可疑IP',
                    '檢視帳號安全狀態',
                ],
            ]);
        }
        
        // 2. API異常呼叫檢測
        $unusualApiCalls = $this->detectUnusualApiCalls();
        if ($unusualApiCalls['count'] > $this->riskThresholds['security']['unusual_api_calls']) {
            $risks[] = $this->createRiskEvent([
                'type' => 'security',
                'category' => 'api_abuse',
                'severity' => 'medium',
                'title' => 'API異常呼叫檢測',
                'description' => "檢測到異常API呼叫模式",
                'metrics' => $unusualApiCalls,
                'recommended_actions' => [
                    '檢查API Rate Limiting設定',
                    '分析呼叫來源和模式',
                    '考慮調整API存取權限',
                ],
            ]);
        }
        
        // 3. 資料存取異常檢測
        $dataAccessRisks = $this->monitorDataAccessPatterns();
        if (!empty($dataAccessRisks)) {
            $risks = array_merge($risks, $dataAccessRisks);
        }
        
        // 4. JWT Token異常檢測
        $tokenRisks = $this->monitorJWTTokenUsage();
        if (!empty($tokenRisks)) {
            $risks = array_merge($risks, $tokenRisks);
        }
        
        return $risks;
    }
    
    /**
     * 效能風險監控
     */
    public function monitorPerformanceRisks(): array
    {
        $risks = [];
        
        // 1. 回應時間監控
        $avgResponseTime = $this->getAverageResponseTime();
        if ($avgResponseTime > $this->riskThresholds['performance']['response_time_ms']) {
            $risks[] = $this->createRiskEvent([
                'type' => 'performance',
                'category' => 'response_time',
                'severity' => 'medium',
                'title' => '系統回應時間過長',
                'description' => "平均回應時間: {$avgResponseTime}ms",
                'metrics' => ['avg_response_time' => $avgResponseTime],
                'recommended_actions' => [
                    '檢查資料庫查詢效能',
                    '分析慢請求路徑',
                    '考慮增加快取層',
                ],
            ]);
        }
        
        // 2. 錯誤率監控
        $errorRate = $this->getErrorRate();
        if ($errorRate > $this->riskThresholds['performance']['error_rate_percent']) {
            $risks[] = $this->createRiskEvent([
                'type' => 'performance',
                'category' => 'error_rate',
                'severity' => 'high',
                'title' => '系統錯誤率過高',
                'description' => "錯誤率: {$errorRate}%",
                'metrics' => ['error_rate' => $errorRate],
                'recommended_actions' => [
                    '檢查錯誤日誌',
                    '識別主要錯誤來源',
                    '修復關鍵錯誤',
                ],
            ]);
        }
        
        // 3. 資源使用監控
        $resourceUsage = $this->getResourceUsage();
        foreach (['cpu_usage_percent', 'memory_usage_percent'] as $metric) {
            if ($resourceUsage[$metric] > $this->riskThresholds['performance'][$metric]) {
                $risks[] = $this->createRiskEvent([
                    'type' => 'performance',
                    'category' => 'resource_usage',
                    'severity' => 'medium',
                    'title' => '系統資源使用率過高',
                    'description' => "{$metric}: {$resourceUsage[$metric]}%",
                    'metrics' => $resourceUsage,
                    'recommended_actions' => [
                        '檢查資源消耗來源',
                        '優化程式碼效能',
                        '考慮擴展資源',
                    ],
                ]);
            }
        }
        
        return $risks;
    }
    
    /**
     * 業務風險監控
     */
    public function monitorBusinessRisks(): array
    {
        $risks = [];
        
        // 1. 收入異常監控
        $revenueChange = $this->getRevenueChangePercent();
        if ($revenueChange < -$this->riskThresholds['business']['daily_revenue_drop_percent']) {
            $risks[] = $this->createRiskEvent([
                'type' => 'business',
                'category' => 'revenue',
                'severity' => 'high',
                'title' => '收入異常下降',
                'description' => "收入較昨日下降 {$revenueChange}%",
                'metrics' => ['revenue_change' => $revenueChange],
                'recommended_actions' => [
                    '分析收入下降原因',
                    '檢查系統功能是否正常',
                    '了解市場變化情況',
                ],
            ]);
        }
        
        // 2. 用戶活躍度監控
        $userActivityChange = $this->getUserActivityChangePercent();
        if ($userActivityChange < -$this->riskThresholds['business']['user_activity_drop_percent']) {
            $risks[] = $this->createRiskEvent([
                'type' => 'business',
                'category' => 'user_activity',
                'severity' => 'medium',
                'title' => '用戶活躍度下降',
                'description' => "用戶活躍度較昨日下降 {$userActivityChange}%",
                'metrics' => ['activity_change' => $userActivityChange],
                'recommended_actions' => [
                    '分析用戶行為變化',
                    '檢查UI/UX是否有問題',
                    '了解用戶反饋',
                ],
            ]);
        }
        
        return $risks;
    }
    
    /**
     * 綜合風險評估
     */
    public function performComprehensiveRiskAssessment(): array
    {
        $allRisks = array_merge(
            $this->monitorSecurityRisks(),
            $this->monitorPerformanceRisks(),
            $this->monitorBusinessRisks()
        );
        
        // 風險關聯分析
        $correlatedRisks = $this->analyzeRiskCorrelations($allRisks);
        
        // 風險優先級排序
        usort($allRisks, function($a, $b) {
            $severityOrder = ['low' => 1, 'medium' => 2, 'high' => 3, 'critical' => 4];
            return $severityOrder[$b['severity']] <=> $severityOrder[$a['severity']];
        });
        
        return [
            'total_risks' => count($allRisks),
            'high_severity_count' => count(array_filter($allRisks, fn($r) => in_array($r['severity'], ['high', 'critical']))),
            'risks' => $allRisks,
            'correlations' => $correlatedRisks,
            'recommendations' => $this->generateRiskRecommendations($allRisks),
        ];
    }
    
    // 私有輔助方法
    
    private function createRiskEvent(array $riskData): array
    {
        // 建立風險事件記錄
        $riskEvent = RiskEvent::create([
            'type' => $riskData['type'],
            'category' => $riskData['category'],
            'severity' => $riskData['severity'],
            'title' => $riskData['title'],
            'description' => $riskData['description'],
            'metrics' => $riskData['metrics'],
            'recommended_actions' => $riskData['recommended_actions'],
            'status' => 'detected',
            'detected_at' => now(),
            'company_id' => auth()->user()->company_id ?? null,
        ]);
        
        // 發送警報
        $this->sendRiskAlert($riskEvent);
        
        return $riskEvent->toArray();
    }
    
    private function sendRiskAlert(RiskEvent $riskEvent): void
    {
        // 根據嚴重程度決定通知方式
        switch ($riskEvent->severity) {
            case 'critical':
                // 立即通知所有管理員
                $this->sendImmediateAlert($riskEvent);
                break;
                
            case 'high':
                // 通知相關負責人
                $this->sendHighPriorityAlert($riskEvent);
                break;
                
            case 'medium':
                // 記錄到日誌並發送郵件
                $this->sendStandardAlert($riskEvent);
                break;
                
            case 'low':
                // 僅記錄到系統中
                Log::info('風險事件檢測', $riskEvent->toArray());
                break;
        }
    }
    
    private function getFailedLoginAttempts(): int
    {
        return DB::table('login_attempts')
                ->where('success', false)
                ->where('created_at', '>=', now()->subHour())
                ->count();
    }
    
    private function detectUnusualApiCalls(): array
    {
        $currentHour = now()->format('Y-m-d H:00:00');
        $currentCalls = DB::table('api_logs')
                         ->where('created_at', '>=', $currentHour)
                         ->count();
        
        // 取得過去24小時同時段的平均值
        $historicalAvg = DB::table('api_logs')
                          ->selectRaw('AVG(hourly_count) as avg_count')
                          ->from(DB::raw('(
                              SELECT COUNT(*) as hourly_count 
                              FROM api_logs 
                              WHERE created_at >= ? AND created_at < ?
                              GROUP BY DATE_FORMAT(created_at, "%Y-%m-%d %H:00:00")
                          ) as hourly_stats'))
                          ->addBinding([now()->subDays(7), now()])
                          ->value('avg_count') ?? 0;
        
        $deviationPercent = $historicalAvg > 0 ? (($currentCalls - $historicalAvg) / $historicalAvg) * 100 : 0;
        
        return [
            'count' => $currentCalls,
            'historical_avg' => $historicalAvg,
            'deviation_percent' => $deviationPercent,
        ];
    }
    
    private function monitorDataAccessPatterns(): array
    {
        $risks = [];
        
        // 檢測異常大量資料存取
        $largeDataAccess = DB::table('audit_logs')
                           ->where('action', 'data_export')
                           ->where('created_at', '>=', now()->subHour())
                           ->where('affected_records', '>', 10000)
                           ->get();
        
        foreach ($largeDataAccess as $access) {
            $risks[] = $this->createRiskEvent([
                'type' => 'security',
                'category' => 'data_access',
                'severity' => 'medium',
                'title' => '大量資料存取檢測',
                'description' => "用戶 {$access->user_id} 存取了 {$access->affected_records} 筆記錄",
                'metrics' => ['records_accessed' => $access->affected_records],
                'recommended_actions' => [
                    '驗證資料存取的合理性',
                    '檢查用戶權限設定',
                    '確認是否為正常業務需求',
                ],
            ]);
        }
        
        return $risks;
    }
    
    private function monitorJWTTokenUsage(): array
    {
        $risks = [];
        
        // 檢測異常Token使用
        $suspiciousTokens = DB::table('user_sessions')
                            ->selectRaw('user_id, COUNT(*) as session_count')
                            ->where('created_at', '>=', now()->subHour())
                            ->groupBy('user_id')
                            ->having('session_count', '>', 10)
                            ->get();
        
        foreach ($suspiciousTokens as $token) {
            $risks[] = $this->createRiskEvent([
                'type' => 'security',
                'category' => 'token_abuse',
                'severity' => 'medium',
                'title' => 'JWT Token異常使用',
                'description' => "用戶 {$token->user_id} 在1小時內建立了 {$token->session_count} 個會話",
                'metrics' => ['session_count' => $token->session_count],
                'recommended_actions' => [
                    '檢查是否為自動化攻擊',
                    '驗證用戶行為合理性',
                    '考慮限制會話數量',
                ],
            ]);
        }
        
        return $risks;
    }
}
```

```php
<?php
// 檔案: app/Services/IncidentResponseService.php

namespace App\Services;

use App\Models\{Incident, IncidentResponse, RiskEvent};
use App\Notifications\IncidentAlert;

class IncidentResponseService
{
    private array $responseTeams = [
        'security' => ['security-team@company.com', 'cto@company.com'],
        'performance' => ['devops-team@company.com', 'tech-lead@company.com'],
        'business' => ['product-manager@company.com', 'ceo@company.com'],
    ];
    
    /**
     * 事件響應處理
     */
    public function handleIncident(RiskEvent $riskEvent): Incident
    {
        // 1. 建立事件記錄
        $incident = Incident::create([
            'risk_event_id' => $riskEvent->id,
            'title' => $riskEvent->title,
            'description' => $riskEvent->description,
            'severity' => $riskEvent->severity,
            'category' => $riskEvent->category,
            'status' => 'open',
            'priority' => $this->calculatePriority($riskEvent),
            'assigned_team' => $this->getResponseTeam($riskEvent->type),
            'detected_at' => $riskEvent->detected_at,
            'company_id' => $riskEvent->company_id,
        ]);
        
        // 2. 執行自動響應
        $this->executeAutomaticResponse($incident);
        
        // 3. 通知響應團隊
        $this->notifyResponseTeam($incident);
        
        // 4. 啟動監控強化
        $this->enhanceMonitoring($incident);
        
        return $incident;
    }
    
    /**
     * 自動響應處理
     */
    private function executeAutomaticResponse(Incident $incident): void
    {
        switch ($incident->category) {
            case 'authentication':
                $this->handleAuthenticationIncident($incident);
                break;
                
            case 'api_abuse':
                $this->handleApiAbuseIncident($incident);
                break;
                
            case 'performance':
                $this->handlePerformanceIncident($incident);
                break;
                
            case 'data_access':
                $this->handleDataAccessIncident($incident);
                break;
        }
    }
    
    private function handleAuthenticationIncident(Incident $incident): void
    {
        // 1. 暫時鎖定可疑IP
        $suspiciousIPs = $this->getSuspiciousIPs($incident);
        foreach ($suspiciousIPs as $ip) {
            $this->blockIP($ip, 3600); // 鎖定1小時
        }
        
        // 2. 強化認證要求
        $this->enableStrictAuthentication();
        
        // 3. 記錄響應動作
        IncidentResponse::create([
            'incident_id' => $incident->id,
            'action_type' => 'automatic',
            'action_description' => '自動封鎖可疑IP並啟用嚴格認證',
            'executed_at' => now(),
        ]);
    }
    
    private function handleApiAbuseIncident(Incident $incident): void
    {
        // 1. 降低API Rate Limit
        $this->adjustRateLimit(0.5); // 降低50%
        
        // 2. 啟用額外驗證
        $this->enableApiKeyValidation();
        
        // 3. 記錄響應動作
        IncidentResponse::create([
            'incident_id' => $incident->id,
            'action_type' => 'automatic',
            'action_description' => '調整API限流並啟用額外驗證',
            'executed_at' => now(),
        ]);
    }
}
```

### **災難恢復計劃**
```php
<?php
// 檔案: app/Services/DisasterRecoveryService.php

namespace App\Services;

class DisasterRecoveryService
{
    /**
     * 系統備份策略
     */
    public function executeBackupPlan(): array
    {
        $backupResults = [];
        
        // 1. 資料庫備份
        $backupResults['database'] = $this->backupDatabase();
        
        // 2. 檔案系統備份
        $backupResults['files'] = $this->backupFiles();
        
        // 3. 設定檔備份
        $backupResults['configs'] = $this->backupConfigurations();
        
        // 4. 快照建立
        $backupResults['snapshots'] = $this->createSystemSnapshot();
        
        return $backupResults;
    }
    
    /**
     * 快速恢復機制
     */
    public function quickRecovery(string $backupId): bool
    {
        try {
            // 1. 驗證備份完整性
            if (!$this->validateBackup($backupId)) {
                throw new \Exception('備份檔案驗證失敗');
            }
            
            // 2. 停止相關服務
            $this->stopServices();
            
            // 3. 恢復資料庫
            $this->restoreDatabase($backupId);
            
            // 4. 恢復檔案
            $this->restoreFiles($backupId);
            
            // 5. 重啟服務
            $this->startServices();
            
            // 6. 驗證系統狀態
            return $this->validateSystemHealth();
            
        } catch (\Exception $e) {
            Log::error('災難恢復失敗', ['error' => $e->getMessage()]);
            return false;
        }
    }
}
```

```php
                    this.resolvedCount = data.resolved_count;
                    this.activeRisks = data.active_risks;

            const classes = {
                'low': 'border-l-4 border-blue-400 bg-blue-50',
                'medium': 'border-l-4 border-yellow-400 bg-yellow-50',
                'high': 'border-l-4 border-red-400 bg-red-50',
                'critical': 'border-l-4 border-red-600 bg-red-100'
            };
            return classes[severity] || classes['low'];
```
