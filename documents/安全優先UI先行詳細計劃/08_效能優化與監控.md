### **1. 智慧快取系統**

#### **多層快取架構設計**
```php
<?php
// 檔案: app/Services/CacheService.php

namespace App\Services;

use Illuminate\Support\Facades\{Cache, Redis};
use Illuminate\Database\Eloquent\Model;
use Carbon\Carbon;

class CacheService
{
    private const CACHE_PREFIXES = [
        'user' => 'usr',
        'product' => 'prd',
        'quote' => 'qte',
        'customer' => 'cst',
        'inventory' => 'inv',
    ];
    
    private const CACHE_TTLS = [
        'short' => 300,      // 5分鐘
        'medium' => 3600,    // 1小時
        'long' => 86400,     // 24小時
        'permanent' => 604800, // 7天
    ];
    
    /**
     * 智慧快取策略
     */
    public function smartCache(string $key, callable $callback, array $options = []): mixed
    {
        $cacheKey = $this->buildCacheKey($key, $options);
        $ttl = $options['ttl'] ?? self::CACHE_TTLS['medium'];
        $tags = $options['tags'] ?? [];
        
        // 檢查是否有熱點數據快取
        if ($this->isHotData($cacheKey)) {
            $ttl *= 2; // 熱點數據延長快取時間
        }
        
        // 使用標籤快取
        $cache = empty($tags) ? Cache::store('redis') : Cache::tags($tags);
        
        return $cache->remember($cacheKey, $ttl, function() use ($callback, $cacheKey) {
            $startTime = microtime(true);
            $result = $callback();
            $executionTime = microtime(true) - $startTime;
            
            // 記錄快取統計
            $this->recordCacheStats($cacheKey, $executionTime);
            
            return $result;
        });
    }
    
    /**
     * 預熱快取
     */
    public function warmupCache(): void
    {
        $warmupTasks = [
            'popular_products' => fn() => $this->warmupPopularProducts(),
            'active_customers' => fn() => $this->warmupActiveCustomers(),
            'user_permissions' => fn() => $this->warmupUserPermissions(),
            'system_settings' => fn() => $this->warmupSystemSettings(),
            'inventory_summary' => fn() => $this->warmupInventorySummary(),
        ];
        
        foreach ($warmupTasks as $task => $callback) {
            try {
                $this->smartCache("warmup.{$task}", $callback, [
                    'ttl' => self::CACHE_TTLS['long'],
                    'tags' => ['warmup', $task],
                ]);
            } catch (\Exception $e) {
                \Log::warning("快取預熱失敗: {$task}", ['error' => $e->getMessage()]);
            }
        }
    }
    
    /**
     * 快取失效策略
     */
    public function invalidateRelatedCache(Model $model, string $action): void
    {
        $modelClass = class_basename($model);
        $modelId = $model->id;
        
        // 基礎快取標籤
        $baseTags = [
            strtolower($modelClass),
            strtolower($modelClass) . ".{$modelId}",
        ];
        
        // 根據模型類型添加額外標籤
        $additionalTags = $this->getAdditionalCacheTags($model, $action);
        $allTags = array_merge($baseTags, $additionalTags);
        
        // 清除相關快取
        foreach ($allTags as $tag) {
            Cache::tags($tag)->flush();
        }
        
        // 記錄快取清除
        \Log::debug('快取已清除', [
            'model' => $modelClass,
            'model_id' => $modelId,
            'action' => $action,
            'tags' => $allTags,
        ]);
    }
    
    /**
     * 分散式快取鎖
     */
    public function lock(string $key, int $seconds = 10): \Illuminate\Contracts\Cache\Lock
    {
        return Cache::store('redis')->lock("lock.{$key}", $seconds);
    }
    
    /**
     * 快取穿透防護
     */
    public function safeCache(string $key, callable $callback, array $options = []): mixed
    {
        $lockKey = "lock.{$key}";
        $lock = $this->lock($lockKey, 30);
        
        try {
            if ($lock->get()) {
                return $this->smartCache($key, $callback, $options);
            } else {
                // 鎖定失敗，返回快取值或預設值
                return Cache::get($this->buildCacheKey($key, $options)) 
                    ?? $options['default'] 
                    ?? null;
            }
        } finally {
            $lock->release();
        }
    }
    
    // 私有輔助方法
    
    private function buildCacheKey(string $key, array $options): string
    {
        $prefix = $options['prefix'] ?? 'app';
        $companyId = auth()->user()->company_id ?? 'global';
        
        return "{$prefix}.{$companyId}.{$key}";
    }
    
    private function isHotData(string $cacheKey): bool
    {
        $accessCount = Redis::get("hot_data.{$cacheKey}") ?? 0;
        return $accessCount > 100; // 超過100次存取視為熱點數據
    }
    
    private function recordCacheStats(string $cacheKey, float $executionTime): void
    {
        Redis::pipeline(function ($pipe) use ($cacheKey, $executionTime) {
            $pipe->incr("cache_stats.{$cacheKey}.hits");
            $pipe->incr("hot_data.{$cacheKey}");
            $pipe->lpush("cache_stats.{$cacheKey}.times", $executionTime);
            $pipe->ltrim("cache_stats.{$cacheKey}.times", 0, 99); // 保留最近100次
            $pipe->expire("hot_data.{$cacheKey}", 3600); // 1小時過期
        });
    }
    
    private function getAdditionalCacheTags(Model $model, string $action): array
    {
        $tags = [];
        
        switch (class_basename($model)) {
            case 'Product':
                $tags = ['inventory', 'products', 'catalog'];
                if ($model->category_id) {
                    $tags[] = "category.{$model->category_id}";
                }
                break;
                
            case 'Quote':
                $tags = ['quotes', 'sales'];
                if ($model->customer_id) {
                    $tags[] = "customer.{$model->customer_id}";
                }
                break;
                
            case 'Customer':
                $tags = ['customers', 'sales'];
                break;
                
            case 'User':
                $tags = ['users', 'auth'];
                if ($model->company_id) {
                    $tags[] = "company.{$model->company_id}";
                }
                break;
        }
        
        return $tags;
    }
    
    private function warmupPopularProducts(): array
    {
        return \App\Models\Product::select('id', 'name', 'sku', 'unit_price', 'stock_quantity')
                                 ->where('is_featured', true)
                                 ->orWhere('stock_quantity', '>', 0)
                                 ->orderBy('sales_count', 'desc')
                                 ->limit(100)
                                 ->get()
                                 ->toArray();
    }
    
    private function warmupActiveCustomers(): array
    {
        return \App\Models\Customer::select('id', 'company_name', 'contact_name', 'email')
                                  ->where('status', 'active')
                                  ->where('last_order_date', '>=', now()->subMonths(6))
                                  ->orderBy('total_orders', 'desc')
                                  ->limit(50)
                                  ->get()
                                  ->toArray();
    }
}
```

```php
<?php
// 檔案: app/Services/QueryOptimizationService.php

namespace App\Services;

use Illuminate\Database\Query\Builder;
use Illuminate\Support\Facades\{DB, Log};

class QueryOptimizationService
{
    private array $slowQueryThreshold = [
        'critical' => 1.0,  // 1秒
        'warning' => 0.5,   // 500ms
        'notice' => 0.1,    // 100ms
    ];
    
    /**
     * 智慧查詢優化
     */
    public function optimizeQuery(Builder $query, array $options = []): Builder
    {
        // 1. 添加適當的索引提示
        $this->addIndexHints($query, $options);
        
        // 2. 優化 JOIN 順序
        $this->optimizeJoinOrder($query);
        
        // 3. 添加查詢快取
        if ($options['cache'] ?? true) {
            $this->addQueryCache($query, $options);
        }
        
        // 4. 監控查詢效能
        $this->monitorQueryPerformance($query);
        
        return $query;
    }
    
    /**
     * 批量查詢優化
     */
    public function batchOptimizedQuery(array $ids, string $model, array $with = []): \Illuminate\Database\Eloquent\Collection
    {
        $modelClass = "App\\Models\\{$model}";
        
        // 分批處理避免 IN 查詢過長
        $batchSize = 100;
        $batches = array_chunk($ids, $batchSize);
        $results = collect();
        
        foreach ($batches as $batch) {
            $batchResults = $modelClass::whereIn('id', $batch)
                                     ->with($with)
                                     ->get();
            $results = $results->merge($batchResults);
        }
        
        return $results;
    }
    
    /**
     * 動態索引建議
     */
    public function suggestIndexes(): array
    {
        $suggestions = [];
        
        // 分析慢查詢日誌
        $slowQueries = $this->getSlowQueries();
        
        foreach ($slowQueries as $query) {
            $analysis = $this->analyzeQuery($query);
            
            if ($analysis['needs_index']) {
                $suggestions[] = [
                    'table' => $analysis['table'],
                    'columns' => $analysis['suggested_columns'],
                    'type' => $analysis['index_type'],
                    'query_pattern' => $query['pattern'],
                    'frequency' => $query['count'],
                    'avg_execution_time' => $query['avg_time'],
                ];
            }
        }
        
        return $suggestions;
    }
    
    /**
     * 資料庫連線池優化
     */
    public function optimizeConnectionPool(): void
    {
        $currentConnections = DB::select('SHOW PROCESSLIST');
        $activeConnections = count($currentConnections);
        
        $config = config('database.connections.mysql');
        $maxConnections = $config['max_connections'] ?? 100;
        
        // 動態調整連線池大小
        if ($activeConnections > $maxConnections * 0.8) {
            // 連線使用率超過80%，記錄警告
            Log::warning('資料庫連線使用率過高', [
                'active_connections' => $activeConnections,
                'max_connections' => $maxConnections,
                'usage_percentage' => ($activeConnections / $maxConnections) * 100,
            ]);
            
            // 清理空閒連線
            $this->cleanupIdleConnections();
        }
    }
    
    /**
     * 分頁查詢優化
     */
    public function optimizedPagination(Builder $query, int $page, int $perPage): array
    {
        // 使用遊標分頁避免 OFFSET 效能問題
        if ($page > 100) { // 超過100頁使用遊標分頁
            return $this->cursorPagination($query, $page, $perPage);
        }
        
        // 標準分頁優化
        $total = $this->optimizedCount($query);
        
        $items = $query->offset(($page - 1) * $perPage)
                      ->limit($perPage)
                      ->get();
        
        return [
            'data' => $items,
            'total' => $total,
            'per_page' => $perPage,
            'current_page' => $page,
            'last_page' => ceil($total / $perPage),
        ];
    }
    
    /**
     * 讀寫分離優化
     */
    public function routeQuery(string $operation, callable $callback)
    {
        $connection = $this->determineConnection($operation);
        
        return DB::connection($connection)->transaction($callback);
    }
    
    // 私有輔助方法
    
    private function addIndexHints(Builder $query, array $options): void
    {
        if (isset($options['use_index'])) {
            $query->from(DB::raw($query->from . ' USE INDEX (' . $options['use_index'] . ')'));
        }
        
        if (isset($options['force_index'])) {
            $query->from(DB::raw($query->from . ' FORCE INDEX (' . $options['force_index'] . ')'));
        }
    }
    
    private function monitorQueryPerformance(Builder $query): void
    {
        DB::listen(function ($sql) {
            if ($sql->time > $this->slowQueryThreshold['warning'] * 1000) {
                Log::warning('慢查詢檢測', [
                    'sql' => $sql->sql,
                    'bindings' => $sql->bindings,
                    'time' => $sql->time,
                ]);
                
                // 自動分析查詢計劃
                $this->analyzeQueryPlan($sql);
            }
        });
    }
    
    private function analyzeQueryPlan($sql): void
    {
        try {
            $explain = DB::select("EXPLAIN {$sql->sql}", $sql->bindings);
            
            foreach ($explain as $row) {
                if ($row->type === 'ALL' || $row->rows > 10000) {
                    Log::warning('查詢計劃警告', [
                        'sql' => $sql->sql,
                        'explain' => $row,
                        'suggestion' => '建議添加索引或優化查詢條件',
                    ]);
                }
            }
        } catch (\Exception $e) {
            Log::debug('查詢計劃分析失敗', ['error' => $e->getMessage()]);
        }
    }
    
    private function cursorPagination(Builder $query, int $page, int $perPage): array
    {
        // 實作遊標分頁邏輯
        $lastId = $this->getLastIdFromPage($page - 1, $perPage);
        
        $items = $query->where('id', '>', $lastId)
                      ->orderBy('id')
                      ->limit($perPage)
                      ->get();
        
        return [
            'data' => $items,
            'has_more' => $items->count() === $perPage,
            'next_cursor' => $items->last()?->id,
        ];
    }
    
    private function determineConnection(string $operation): string
    {
        $readOperations = ['select', 'show', 'describe', 'explain'];
        
        return in_array(strtolower($operation), $readOperations) ? 'mysql_read' : 'mysql_write';
    }
}
```

#### **效能監控服務**
```php
<?php
// 檔案: app/Services/PerformanceMonitoringService.php

namespace App\Services;

use Illuminate\Support\Facades\{Log, Cache, Redis};
use Illuminate\Http\Request;

class PerformanceMonitoringService
{
    private array $metrics = [];
    private float $requestStartTime;
    
    public function __construct()
    {
        $this->requestStartTime = microtime(true);
    }
    
    /**
     * 請求效能監控
     */
    public function monitorRequest(Request $request): void
    {
        $startTime = microtime(true);
        $memoryStart = memory_get_usage(true);
        
        // 註冊請求結束時的監控
        register_shutdown_function(function() use ($request, $startTime, $memoryStart) {
            $this->recordRequestMetrics($request, $startTime, $memoryStart);
        });
    }
    
    /**
     * API回應時間監控
     */
    public function trackApiResponse(string $endpoint, float $responseTime, int $statusCode): void
    {
        $date = date('Y-m-d');
        $hour = date('H');
        
        // 記錄到 Redis 時序數據
        Redis::pipeline(function ($pipe) use ($endpoint, $responseTime, $statusCode, $date, $hour) {
            // 每小時統計
            $pipe->lpush("api_response_times:{$endpoint}:{$date}:{$hour}", $responseTime);
            $pipe->ltrim("api_response_times:{$endpoint}:{$date}:{$hour}", 0, 999); // 保留1000筆
            $pipe->expire("api_response_times:{$endpoint}:{$date}:{$hour}", 86400 * 7); // 7天
            
            // 狀態碼統計
            $pipe->incr("api_status_codes:{$endpoint}:{$statusCode}:{$date}");
            $pipe->expire("api_status_codes:{$endpoint}:{$statusCode}:{$date}", 86400 * 30); // 30天
            
            // 整體統計
            $pipe->incr("api_requests_total:{$date}");
            $pipe->incr("api_requests_hourly:{$date}:{$hour}");
        });
        
        // 檢查是否需要警報
        $this->checkPerformanceAlerts($endpoint, $responseTime);
    }
    
    /**
     * 記憶體使用監控
     */
    public function monitorMemoryUsage(): array
    {
        $metrics = [
            'current_usage' => memory_get_usage(true),
            'peak_usage' => memory_get_peak_usage(true),
            'limit' => $this->getMemoryLimit(),
        ];
        
        $metrics['usage_percentage'] = ($metrics['current_usage'] / $metrics['limit']) * 100;
        
        // 記憶體使用率過高警告
        if ($metrics['usage_percentage'] > 80) {
            Log::warning('記憶體使用率過高', $metrics);
            
            // 觸發垃圾回收
            gc_collect_cycles();
        }
        
        return $metrics;
    }
    
    /**
     * 資料庫效能監控
     */
    public function monitorDatabasePerformance(): void
    {
        DB::listen(function ($query) {
            $metrics = [
                'sql' => $query->sql,
                'bindings' => $query->bindings,
                'time' => $query->time,
                'connection' => $query->connection->getName(),
            ];
            
            // 記錄到時序數據庫
            $this->recordDatabaseMetrics($metrics);
            
            // 慢查詢警報
            if ($query->time > 1000) { // 超過1秒
                Log::warning('慢查詢檢測', $metrics);
                $this->sendSlowQueryAlert($metrics);
            }
        });
    }
    
    /**
     * 快取命中率監控
     */
    public function monitorCacheHitRate(): array
    {
        $cacheInfo = Cache::store('redis')->getRedis()->info('stats');
        
        $hits = $cacheInfo['keyspace_hits'] ?? 0;
        $misses = $cacheInfo['keyspace_misses'] ?? 0;
        $total = $hits + $misses;
        
        $hitRate = $total > 0 ? ($hits / $total) * 100 : 0;
        
        $metrics = [
            'hits' => $hits,
            'misses' => $misses,
            'hit_rate' => $hitRate,
            'total_requests' => $total,
        ];
        
        // 快取命中率過低警告
        if ($hitRate < 70) {
            Log::warning('快取命中率過低', $metrics);
        }
        
        return $metrics;
    }
    
    /**
     * 系統資源監控
     */
    public function monitorSystemResources(): array
    {
        $metrics = [
            'cpu_usage' => $this->getCpuUsage(),
            'memory' => $this->monitorMemoryUsage(),
            'disk_usage' => $this->getDiskUsage(),
            'load_average' => sys_getloadavg(),
        ];
        
        // 檢查資源使用警報
        $this->checkResourceAlerts($metrics);
        
        return $metrics;
    }
    
    /**
     * 業務指標監控
     */
    public function trackBusinessMetrics(): void
    {
        $date = date('Y-m-d');
        
        // 快取昨日數據比較
        $yesterday = date('Y-m-d', strtotime('-1 day'));
        
        $metrics = [
            'daily_quotes' => $this->getDailyQuoteCount($date),
            'daily_revenue' => $this->getDailyRevenue($date),
            'active_users' => $this->getActiveUserCount($date),
            'api_requests' => $this->getApiRequestCount($date),
        ];
        
        // 與昨日比較
        $yesterdayMetrics = Cache::get("business_metrics.{$yesterday}", []);
        
        foreach ($metrics as $key => $value) {
            $yesterdayValue = $yesterdayMetrics[$key] ?? 0;
            $change = $yesterdayValue > 0 ? (($value - $yesterdayValue) / $yesterdayValue) * 100 : 0;
            
            $metrics["{$key}_change"] = $change;
        }
        
        // 快取今日數據
        Cache::put("business_metrics.{$date}", $metrics, 86400 * 7);
        
        // 異常變化警報
        $this->checkBusinessMetricAlerts($metrics);
    }
    
    /**
     * 效能報表生成
     */
    public function generatePerformanceReport(array $filters = []): array
    {
        $dateRange = $filters['date_range'] ?? [
            'start' => now()->subDays(7)->toDateString(),
            'end' => now()->toDateString(),
        ];
        
        return [
            'summary' => $this->getPerformanceSummary($dateRange),
            'api_performance' => $this->getApiPerformanceMetrics($dateRange),
            'database_performance' => $this->getDatabasePerformanceMetrics($dateRange),
            'cache_performance' => $this->getCachePerformanceMetrics($dateRange),
            'system_resources' => $this->getSystemResourceMetrics($dateRange),
            'business_metrics' => $this->getBusinessMetricsReport($dateRange),
            'alerts' => $this->getPerformanceAlerts($dateRange),
            'recommendations' => $this->generatePerformanceRecommendations(),
        ];
    }
    
    // 私有輔助方法
    
    private function recordRequestMetrics(Request $request, float $startTime, int $memoryStart): void
    {
        $endTime = microtime(true);
        $memoryEnd = memory_get_usage(true);
        
        $metrics = [
            'url' => $request->fullUrl(),
            'method' => $request->method(),
            'response_time' => ($endTime - $startTime) * 1000, // 毫秒
            'memory_usage' => $memoryEnd - $memoryStart,
            'peak_memory' => memory_get_peak_usage(true),
            'status_code' => http_response_code(),
            'user_id' => auth()->id(),
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
        ];
        
        // 記錄到日誌
        Log::info('請求效能指標', $metrics);
        
        // 記錄到時序數據庫
        $this->recordToTimeSeries('request_metrics', $metrics);
        
        // 追踪 API 回應時間
        if ($request->is('api/*')) {
            $this->trackApiResponse(
                $request->path(),
                $metrics['response_time'],
                $metrics['status_code']
            );
        }
    }
    
    private function checkPerformanceAlerts(string $endpoint, float $responseTime): void
    {
        $threshold = config('monitoring.response_time_threshold', 1000); // 1秒
        
        if ($responseTime > $threshold) {
            $this->sendPerformanceAlert([
                'type' => 'slow_response',
                'endpoint' => $endpoint,
                'response_time' => $responseTime,
                'threshold' => $threshold,
            ]);
        }
    }
    
    private function sendPerformanceAlert(array $alertData): void
    {
        // 實作警報發送邏輯 (Email, Slack, 釘釘等)
        Log::critical('效能警報', $alertData);
        
        // 防止警報疲勞 - 同類型警報5分鐘內只發送一次
        $alertKey = "alert.{$alertData['type']}.{$alertData['endpoint'] ?? 'global'}";
        
        if (!Cache::has($alertKey)) {
            // 發送警報到外部系統
            $this->notifyExternalSystems($alertData);
            
            // 設定冷卻時間
            Cache::put($alertKey, true, 300); // 5分鐘
        }
    }
    
    private function getCpuUsage(): float
    {
        if (function_exists('sys_getloadavg')) {
            $load = sys_getloadavg();
            return $load[0]; // 1分鐘平均負載
        }
        
        return 0.0;
    }
    
    private function getDiskUsage(): array
    {
        $bytes = disk_total_space('.');
        $free = disk_free_space('.');
        $used = $bytes - $free;
        
        return [
            'total' => $bytes,
            'used' => $used,
            'free' => $free,
            'usage_percentage' => ($used / $bytes) * 100,
        ];
    }
    
    private function getMemoryLimit(): int
    {
        $limit = ini_get('memory_limit');
        
        if ($limit == -1) {
            return PHP_INT_MAX;
        }
        
        return $this->parseSize($limit);
    }
    
    private function parseSize(string $size): int
    {
        $unit = strtolower(substr($size, -1));
        $value = (int) substr($size, 0, -1);
        
        switch ($unit) {
            case 'g': return $value * 1024 * 1024 * 1024;
            case 'm': return $value * 1024 * 1024;
            case 'k': return $value * 1024;
            default: return (int) $size;
        }
    }
}
```

#### **自動調優服務**
```php
<?php
// 檔案: app/Services/AutoTuningService.php

namespace App\Services;

class AutoTuningService
{
    /**
     * 自動效能調優
     */
    public function performAutoTuning(): array
    {
        $results = [];
        
        // 1. 自動快取調優
        $results['cache_tuning'] = $this->tuneCacheSettings();
        
        // 2. 資料庫連線池調優
        $results['database_tuning'] = $this->tuneDatabaseSettings();
        
        // 3. PHP 設定調優
        $results['php_tuning'] = $this->tunePhpSettings();
        
        // 4. 自動索引建議
        $results['index_suggestions'] = $this->generateIndexSuggestions();
        
        return $results;
    }
    
    /**
     * 智慧快取策略調整
     */
    private function tuneCacheSettings(): array
    {
        $hitRate = $this->getCacheHitRate();
        $memoryUsage = $this->getCacheMemoryUsage();
        
        $recommendations = [];
        
        if ($hitRate < 70) {
            $recommendations[] = '增加快取 TTL 時間';
            $recommendations[] = '擴大快取容量';
        }
        
        if ($memoryUsage > 80) {
            $recommendations[] = '啟用 LRU 淘汰策略';
            $recommendations[] = '壓縮快取數據';
        }
        
        return $recommendations;
    }
    
    /**
     * 動態索引優化
     */
    private function generateIndexSuggestions(): array
    {
        // 分析查詢模式並建議索引
        $queryPatterns = $this->analyzeQueryPatterns();
        $suggestions = [];
        
        foreach ($queryPatterns as $pattern) {
            if ($pattern['frequency'] > 100 && $pattern['avg_time'] > 0.1) {
                $suggestions[] = [
                    'table' => $pattern['table'],
                    'columns' => $pattern['where_columns'],
                    'type' => $this->recommendIndexType($pattern),
                ];
            }
        }
        
        return $suggestions;
    }
}
```
