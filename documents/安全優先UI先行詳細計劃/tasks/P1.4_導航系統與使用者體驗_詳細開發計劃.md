#### **下午 (4小時)**：導航組件架構設計
```php
<?php
// 檔案: frontend/app/Services/NavigationService.php

namespace App\Services;

use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Cache;
use App\Models\User;

class NavigationService
{
    protected $navigationConfig;
    
    public function __construct()
    {
        $this->navigationConfig = config('navigation');
    }
    
    public function getNavigationStructure(User $user = null): array
    {
        $user = $user ?? Auth::user();
        
        if (!$user) {
            return $this->getGuestNavigation();
        }
        
        // 快取使用者的導航結構
        return Cache::remember(
            "navigation_structure_{$user->id}",
            now()->addMinutes(30),
            function () use ($user) {
                return $this->buildNavigationForUser($user);
            }
        );
    }
    
    protected function buildNavigationForUser(User $user): array
    {
        $navigation = [
            'primary_menu' => [],
            'quick_access' => [],
            'user_menu' => [],
            'breadcrumbs' => [],
            'metadata' => [
                'user_role' => $user->role,
                'permissions' => $user->getAllPermissions()->pluck('name'),
                'last_updated' => now(),
            ]
        ];
        
        // 建立主選單
        foreach ($this->navigationConfig['primary_menu'] as $item) {
            if ($this->userCanAccess($user, $item)) {
                $processedItem = $this->processNavigationItem($item, $user);
                $navigation['primary_menu'][] = $processedItem;
            }
        }
        
        // 建立快速存取
        foreach ($this->navigationConfig['quick_access'] as $item) {
            if ($this->userCanAccess($user, $item)) {
                $processedItem = $this->processQuickAccessItem($item, $user);
                $navigation['quick_access'][] = $processedItem;
            }
        }
        
        // 建立使用者選單
        $navigation['user_menu'] = $this->buildUserMenu($user);
        
        return $navigation;
    }
    
    protected function processNavigationItem(array $item, User $user): array
    {
        $processedItem = [
            'id' => $item['id'],
            'name' => $item['name'],
            'icon' => $item['icon'],
            'url' => $item['url'] ?? null,
            'description' => $item['description'] ?? '',
            'has_submenu' => $item['has_submenu'] ?? false,
            'submenu' => [],
            'badge_count' => 0,
            'is_active' => $this->isCurrentPage($item['url'] ?? ''),
        ];
        
        // 處理子選單
        if ($item['has_submenu'] && isset($item['submenu'])) {
            foreach ($item['submenu'] as $subItem) {
                if ($this->userCanAccess($user, $subItem)) {
                    $processedSubItem = [
                        'id' => $subItem['id'],
                        'name' => $subItem['name'],
                        'url' => $subItem['url'],
                        'description' => $subItem['description'] ?? '',
                        'features' => $subItem['features'] ?? [],
                        'badge_count' => $this->getBadgeCount($subItem['id'], $user),
                        'is_active' => $this->isCurrentPage($subItem['url']),
                    ];
                    
                    $processedItem['submenu'][] = $processedSubItem;
                    
                    // 如果子項目是活躍的，父項目也應該標記為活躍
                    if ($processedSubItem['is_active']) {
                        $processedItem['is_active'] = true;
                    }
                }
            }
        }
        
        return $processedItem;
    }
    
    protected function processQuickAccessItem(array $item, User $user): array
    {
        return [
            'id' => $item['id'],
            'name' => $item['name'],
            'icon' => $item['icon'],
            'url' => $item['url'],
            'color' => $item['color'] ?? 'gray',
            'badge_count' => $this->getBadgeCount($item['badge_count'] ?? '', $user),
        ];
    }
    
    protected function userCanAccess(User $user, array $item): bool
    {
        // 檢查權限要求
        if (isset($item['permission_required'])) {
            return $user->hasPermissionTo($item['permission_required']);
        }
        
        // 檢查角色要求
        if (isset($item['role_required'])) {
            return $user->hasRole($item['role_required']);
        }
        
        // 預設允許存取
        return true;
    }
    
    protected function getBadgeCount(string $badgeType, User $user): int
    {
        switch ($badgeType) {
            case 'inventory_alerts':
                return $this->getInventoryAlertsCount($user);
            case 'pending_orders':
                return $this->getPendingOrdersCount($user);
            case 'pending_quotes':
                return $this->getPendingQuotesCount($user);
            default:
                return 0;
        }
    }
    
    protected function getInventoryAlertsCount(User $user): int
    {
        // 實作庫存警報計數邏輯
        return Cache::remember(
            "inventory_alerts_count_{$user->id}",
            now()->addMinutes(5),
            function () {
                // 這裡應該查詢實際的庫存警報數量
                return 0;
            }
        );
    }
    
    protected function getPendingOrdersCount(User $user): int
    {
        // 實作待處理訂單計數邏輯
        return Cache::remember(
            "pending_orders_count_{$user->id}",
            now()->addMinutes(5),
            function () {
                // 這裡應該查詢實際的待處理訂單數量
                return 0;
            }
        );
    }
    
    protected function isCurrentPage(string $url): bool
    {
        if (empty($url)) {
            return false;
        }
        
        $currentPath = request()->path();
        $urlPath = ltrim(parse_url($url, PHP_URL_PATH), '/');
        
        return $currentPath === $urlPath || str_starts_with($currentPath, $urlPath . '/');
    }
    
    public function getBreadcrumbs(): array
    {
        $path = request()->path();
        $segments = explode('/', $path);
        $breadcrumbs = [];
        $currentPath = '';
        
        foreach ($segments as $segment) {
            if (empty($segment)) continue;
            
            $currentPath .= '/' . $segment;
            $breadcrumbs[] = [
                'name' => $this->getPageTitle($currentPath),
                'url' => $currentPath,
                'is_current' => $currentPath === '/' . $path,
            ];
        }
        
        return $breadcrumbs;
    }
    
    protected function getPageTitle(string $path): string
    {
        $titles = [
            '/dashboard' => '儀表板',
            '/quotes' => '報價管理',
            '/quotes/create' => '建立報價',
            '/orders' => '訂單管理',
            '/customers' => '客戶管理',
            '/products' => '產品管理',
            '/inventory' => '庫存管理',
            '/suppliers' => '供應商管理',
            '/admin' => '系統管理',
            '/admin/users' => '使用者管理',
            '/admin/settings' => '系統設定',
        ];
        
        return $titles[$path] ?? ucfirst(basename($path));
    }
    
    public function clearNavigationCache(User $user = null): void
    {
        if ($user) {
            Cache::forget("navigation_structure_{$user->id}");
        } else {
            // 清除所有導航快取
            Cache::tags(['navigation'])->flush();
        }
    }
}
```

```php
<script>
function globalSearch() {
    return {
        searchQuery: '',
        loading: false,
        selectedIndex: 0,
        results: {
            pages: [],
            data: []
        },
        searchHistory: [],
        
        init() {
            this.loadSearchHistory();
        },
        
        async search() {
            if (!this.searchQuery.trim()) {
                this.results = { pages: [], data: [] };
                return;
            }
            
            this.loading = true;
            this.selectedIndex = 0;
            
            try {
                const response = await fetch('/api/search', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content
                    },
                    body: JSON.stringify({
                        query: this.searchQuery,
                        types: ['pages', 'customers', 'products', 'quotes', 'orders']
                    })
                });
                
                const data = await response.json();
                
                if (data.success) {
                    this.results = data.results;
                } else {
                    console.error('搜尋失敗:', data.message);
                }
            } catch (error) {
                console.error('搜尋請求失敗:', error);
            } finally {
                this.loading = false;
            }
        },
        
        navigateDown() {
            const totalResults = this.results.pages.length + this.results.data.length;
            if (this.selectedIndex < totalResults - 1) {
                this.selectedIndex++;
            }
        },
        
        navigateUp() {
            if (this.selectedIndex > 0) {
                this.selectedIndex--;
            }
        },
        
        selectFirst() {
            const allResults = [...this.results.pages, ...this.results.data];
            if (allResults.length > 0) {
                const selected = allResults[this.selectedIndex];
                this.recordSearch(selected);
                window.location.href = selected.url;
            }
        },
        
        clearSearch() {
            this.searchQuery = '';
            this.results = { pages: [], data: [] };
            this.selectedIndex = 0;
        },
        
        recordSearch(item) {
            // 記錄搜尋歷史
            this.searchHistory.unshift({
                query: this.searchQuery,
                item: item,
                timestamp: Date.now()
            });
            
            // 保持最近20筆記錄
            this.searchHistory = this.searchHistory.slice(0, 20);
            
            // 儲存到localStorage
            localStorage.setItem('nexus_search_history', JSON.stringify(this.searchHistory));
            
            // 發送分析數據
            this.sendAnalytics(item);
        },
        
        loadSearchHistory() {
            const stored = localStorage.getItem('nexus_search_history');
            if (stored) {
                try {
                    this.searchHistory = JSON.parse(stored);
                } catch (error) {
                    console.error('載入搜尋歷史失敗:', error);
                }
            }
        },
        
        sendAnalytics(item) {
            // 發送搜尋分析數據到後端
            fetch('/api/analytics/search', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content
                },
                body: JSON.stringify({
                    query: this.searchQuery,
                    result_type: item.type,
                    result_id: item.id,
                    timestamp: Date.now()
                })
            }).catch(error => {
                console.error('搜尋分析發送失敗:', error);
            });
        },
        
        getDataTypeColor(type) {
            const colors = {
                'customer': 'bg-blue-500',
                'product': 'bg-green-500',
                'quote': 'bg-purple-500',
                'order': 'bg-orange-500',
                'supplier': 'bg-indigo-500'
            };
            return colors[type] || 'bg-gray-500';
        }
    }
}
```

```php
<?php
// 檔案: frontend/app/Services/NavigationAnalyticsService.php

namespace App\Services;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;
use App\Models\NavigationAnalytics;

class NavigationAnalyticsService
{
    public function trackNavigation(array $data): void
    {
        NavigationAnalytics::create([
            'user_id' => auth()->id(),
            'path' => $data['path'],
            'source' => $data['source'], // menu, search, breadcrumb, mobile
            'device_type' => $data['device_type'],
            'session_id' => session()->getId(),
            'timestamp' => now(),
        ]);
    }
    
    public function getNavigationHeatmap(string $period = '7d'): array
    {
        return Cache::remember(
            "navigation_heatmap_{$period}",
            now()->addHours(1),
            function () use ($period) {
                $days = match($period) {
                    '1d' => 1,
                    '7d' => 7,
                    '30d' => 30,
                    default => 7
                };
                
                return DB::table('navigation_analytics')
                    ->where('created_at', '>=', now()->subDays($days))
                    ->groupBy('path')
                    ->selectRaw('path, COUNT(*) as visits, COUNT(DISTINCT user_id) as unique_users')
                    ->orderByDesc('visits')
                    ->get()
                    ->toArray();
            }
        );
    }
    
    public function getUnusedPages(): array
    {
        $allPages = $this->getAllRegisteredPages();
        $visitedPages = DB::table('navigation_analytics')
            ->where('created_at', '>=', now()->subDays(30))
            ->distinct()
            ->pluck('path')
            ->toArray();
            
        return array_diff($allPages, $visitedPages);
    }
    
    public function getSearchFailures(): array
    {
        return DB::table('search_analytics')
            ->where('created_at', '>=', now()->subDays(7))
            ->where('results_count', 0)
            ->groupBy('query')
            ->selectRaw('query, COUNT(*) as failure_count')
            ->orderByDesc('failure_count')
            ->limit(20)
            ->get()
            ->toArray();
    }
    
    private function getAllRegisteredPages(): array
    {
        // 從路由中取得所有已註冊的頁面
        $routes = app('router')->getRoutes();
        $pages = [];
        
        foreach ($routes as $route) {
            if ($route->getName() && !str_starts_with($route->getName(), 'api.')) {
                $pages[] = '/' . $route->uri();
            }
        }
        
        return $pages;
    }
}
```

```php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;

use App\Services\NavigationService;

class NavigationOptimization
{
    public function __construct(
        private NavigationService $navigationService
    ) {}
    
    public function handle(Request $request, Closure $next)
    {
        // 預載入導航資料
        $this->preloadNavigationData($request);
        
        // 設定導航快取標頭
        $this->setNavigationCacheHeaders($request);
        
        return $next($request);
    }
    
    private function preloadNavigationData(Request $request): void
    {
        if ($request->user()) {
            // 預載入使用者導航
            $navigation = $this->navigationService->getNavigationStructure($request->user());
            
            // 共享給所有 View
            View::share('globalNavigation', $navigation);
            
            // 預載入麵包屑
            $breadcrumbs = $this->navigationService->getBreadcrumbs();
            View::share('globalBreadcrumbs', $breadcrumbs);
            
            // 預載入徽章計數
            $this->preloadBadgeCounts($request->user());
        }
    }
    
    private function preloadBadgeCounts($user): void
    {
        $badges = Cache::remember(
            "badge_counts_{$user->id}",
            now()->addMinutes(5),
```
