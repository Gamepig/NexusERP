```javascript
// 檔案: frontend/resources/js/components/dashboard/DashboardManager.js

class DashboardManager {
    constructor(options = {}) {
        this.apiEndpoint = options.apiEndpoint || '/api/dashboard';
        this.refreshInterval = options.refreshInterval || 30000; // 30秒
        this.charts = new Map();
        this.stats = new Map();
        this.isAutoRefresh = true;
        this.lastUpdate = null;
        
        this.init();
    }
    
    async init() {
        await this.loadDashboardData();
        this.setupEventListeners();
        this.startAutoRefresh();
        
        console.log('DashboardManager 初始化完成');
    }
    
    async loadDashboardData() {
        try {
            this.showLoadingState();
            
            const response = await fetch(this.apiEndpoint, {
                method: 'GET',
                headers: {
                    'Accept': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest'
                }
            });
            
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            
            const data = await response.json();
            
            if (data.success) {
                await this.updateStatistics(data.statistics);
                await this.updateCharts(data.charts);
                await this.updateQuickActions(data.quickActions);
                this.lastUpdate = new Date();
                this.updateLastRefreshTime();
            } else {
                throw new Error(data.message || '載入儀表板數據失敗');
            }
            
        } catch (error) {
            console.error('載入儀表板數據錯誤:', error);
            this.showErrorState(error.message);
        } finally {
            this.hideLoadingState();
        }
    }
    
    async updateStatistics(statistics) {
        const statsContainer = document.getElementById('dashboardStats');
        if (!statsContainer) return;
        
        // 統計數據映射
        const statConfigs = {
            totalRevenue: {
                title: '總營收',
                icon: 'currency-dollar',
                prefix: '$',
                changeType: statistics.totalRevenue?.change > 0 ? 'positive' : 'negative'
            },
            totalOrders: {
                title: '總訂單數',
                icon: 'shopping-cart',
                changeType: statistics.totalOrders?.change > 0 ? 'positive' : 'negative'
            },
            totalCustomers: {
                title: '總客戶數',
                icon: 'users',
                changeType: statistics.totalCustomers?.change > 0 ? 'positive' : 'neutral'
            },
            pendingQuotes: {
                title: '待處理報價',
                icon: 'document-text',
                changeType: statistics.pendingQuotes?.change > 0 ? 'neutral' : 'positive'
            },
            lowStockAlerts: {
                title: '低庫存警報',
                icon: 'exclamation-triangle',
                changeType: statistics.lowStockAlerts?.change > 0 ? 'negative' : 'positive'
            },
            conversionRate: {
                title: '轉換率',
                icon: 'chart-line',
                suffix: '%',
                changeType: statistics.conversionRate?.change > 0 ? 'positive' : 'negative'
            }
        };
        
        // 更新每個統計卡片
        Object.entries(statistics).forEach(([key, data]) => {
            const config = statConfigs[key];
            if (!config) return;
            
            const cardElement = document.querySelector(`[data-stat="${key}"]`);
            if (cardElement) {
                this.updateStatCard(cardElement, {
                    ...config,
                    value: data.value,
                    change: data.change,
                    trend: data.trend,
                    description: data.description
                });
            }
        });
        
        this.stats.clear();
        Object.entries(statistics).forEach(([key, data]) => {
            this.stats.set(key, data);
        });
    }
    
    updateStatCard(element, config) {
        // 更新數值
        const valueElement = element.querySelector('.stat-value');
        if (valueElement) {
            valueElement.textContent = `${config.prefix || ''}${this.formatNumber(config.value)}${config.suffix || ''}`;
        }
        
        // 更新變化百分比
        const changeElement = element.querySelector('.stat-change');
        if (changeElement && config.change !== undefined) {
            changeElement.textContent = `${config.change > 0 ? '+' : ''}${config.change.toFixed(1)}%`;
            changeElement.className = `stat-change ${config.changeType}`;
        }
        
        // 更新描述
        const descElement = element.querySelector('.stat-description');
        if (descElement && config.description) {
            descElement.textContent = config.description;
        }
        
        // 添加更新動畫
        element.classList.add('updated');
        setTimeout(() => element.classList.remove('updated'), 500);
    }
    
    formatNumber(num) {
        if (num >= 1000000) {
            return (num / 1000000).toFixed(1) + 'M';
        } else if (num >= 1000) {
            return (num / 1000).toFixed(1) + 'K';
        }
        return num.toLocaleString();
    }
    
    setupEventListeners() {
        // 手動刷新
        document.addEventListener('click', (e) => {
            if (e.target.matches('[data-action="refresh-dashboard"]')) {
                this.loadDashboardData();
            }
        });
        
        // 自動刷新切換
        document.addEventListener('change', (e) => {
            if (e.target.matches('[data-toggle="auto-refresh"]')) {
                this.isAutoRefresh = e.target.checked;
                if (this.isAutoRefresh) {
                    this.startAutoRefresh();
                } else {
                    this.stopAutoRefresh();
                }
            }
        });
        
        // 自定義事件監聽
        document.addEventListener('refresh-stat', (e) => {
            this.refreshSpecificStat(e.detail);
        });
        
        // 圖表互動事件
        document.addEventListener('chart-interaction', (e) => {
            this.handleChartInteraction(e.detail);
        });
    }
    
    showLoadingState() {
        document.querySelectorAll('[data-stat]').forEach(card => {
            card.classList.add('loading');
        });
        
        const refreshBtn = document.querySelector('[data-action="refresh-dashboard"]');
        if (refreshBtn) {
            refreshBtn.disabled = true;
            refreshBtn.innerHTML = '<svg class="animate-spin h-4 w-4" fill="none" viewBox="0 0 24 24"><circle cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4" class="opacity-25"></circle><path fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" class="opacity-75"></path></svg>';
        }
    }
    
    hideLoadingState() {
        document.querySelectorAll('[data-stat]').forEach(card => {
            card.classList.remove('loading');
        });
        
        const refreshBtn = document.querySelector('[data-action="refresh-dashboard"]');
        if (refreshBtn) {
            refreshBtn.disabled = false;
            refreshBtn.innerHTML = '<x-heroicon-o-arrow-path class="w-4 h-4" />';
        }
    }
    
    startAutoRefresh() {
        if (this.refreshTimer) {
            clearInterval(this.refreshTimer);
        }
        
        this.refreshTimer = setInterval(() => {
            if (this.isAutoRefresh) {
                this.loadDashboardData();
            }
        }, this.refreshInterval);
    }
    
    stopAutoRefresh() {
        if (this.refreshTimer) {
            clearInterval(this.refreshTimer);
            this.refreshTimer = null;
        }
    }
    
    updateLastRefreshTime() {
        const timeElement = document.getElementById('lastUpdateTime');
        if (timeElement && this.lastUpdate) {
            timeElement.textContent = `最後更新: ${this.lastUpdate.toLocaleTimeString()}`;
        }
    }
}

// 全域匯出
window.DashboardManager = DashboardManager;
```

```javascript
                        ...this.defaultOptions.scales.y,

                                return '$' + value.toLocaleString();

                    }
                }

                        'var(--nexus-accent-orange)',

                ...this.defaultOptions,

        this.charts.set(canvasId, chart);
        return chart;

        const canvas = document.getElementById(canvasId);
        if (!canvas) return null;
        
        const ctx = canvas.getContext('2d');
        
        const chartConfig = {
            type: 'bar',
            data: {
                labels: data.labels,
                datasets: [{
                    label: '庫存數量',
                    data: data.values,
                    backgroundColor: 'var(--nexus-accent-blue)',
                    borderColor: 'var(--nexus-primary-500)',
                    borderWidth: 1
                }]
            },
            options: {
                ...this.defaultOptions,
                plugins: {
                    ...this.defaultOptions.plugins,
                    title: {
                        display: true,
                        text: '庫存水準'
                    }
                }
            }
        };
        
        const chart = new Chart(ctx, chartConfig);
        this.charts.set(canvasId, chart);
        return chart;
    }
    
    updateChart(chartId, newData) {
        const chart = this.charts.get(chartId);
        if (!chart) return;
        
        chart.data.labels = newData.labels;
        chart.data.datasets[0].data = newData.values;
        chart.update('active');
    }
    
    destroyChart(chartId) {
        const chart = this.charts.get(chartId);
        if (chart) {
            chart.destroy();
            this.charts.delete(chartId);
        }
    }
    
    destroyAllCharts() {
        this.charts.forEach((chart, id) => {
            chart.destroy();
        });
        this.charts.clear();
    }
}
```

```javascript
// 檔案: frontend/resources/js/components/dashboard/RealtimeUpdater.js

class RealtimeUpdater {
    constructor(options = {}) {
        this.wsUrl = options.wsUrl || `ws://localhost:8080/ws/dashboard`;
        this.apiEndpoint = options.apiEndpoint || '/api/dashboard/realtime';
        this.fallbackInterval = options.fallbackInterval || 10000; // 10秒
        this.maxReconnectAttempts = options.maxReconnectAttempts || 5;
        
        this.ws = null;
        this.isConnected = false;
        this.reconnectAttempts = 0;
        this.fallbackTimer = null;
        this.subscribers = new Map();
        
        this.init();
    }
    
    init() {
        this.connectWebSocket();
        this.setupFallbackPolling();
        
        // 頁面可見性變化處理
        document.addEventListener('visibilitychange', () => {
            if (document.hidden) {
                this.pauseUpdates();
            } else {
                this.resumeUpdates();
            }
        });
    }
    
    connectWebSocket() {
        try {
            this.ws = new WebSocket(this.wsUrl);
            
            this.ws.onopen = (event) => {
                console.log('WebSocket 連線建立成功');
                this.isConnected = true;
                this.reconnectAttempts = 0;
                this.stopFallbackPolling();
                
                // 發送認證訊息
                this.ws.send(JSON.stringify({
                    type: 'auth',
                    token: this.getAuthToken()
                }));
            };
            
            this.ws.onmessage = (event) => {
                try {
                    const data = JSON.parse(event.data);
                    this.handleRealtimeData(data);
                } catch (error) {
                    console.error('WebSocket 訊息解析錯誤:', error);
                }
            };
            
            this.ws.onclose = (event) => {
                console.log('WebSocket 連線關閉');
                this.isConnected = false;
                this.handleReconnection();
            };
            
            this.ws.onerror = (error) => {
                console.error('WebSocket 錯誤:', error);
                this.isConnected = false;
            };
            
        } catch (error) {
            console.error('WebSocket 建立失敗:', error);
            this.startFallbackPolling();
        }
    }
    
    handleRealtimeData(data) {
        switch (data.type) {
            case 'dashboard_update':
                this.updateDashboardData(data.payload);
                break;
            case 'stat_update':
                this.updateSpecificStat(data.payload);
                break;
            case 'alert':
                this.handleAlert(data.payload);
                break;
            default:
                console.warn('未知的實時數據類型:', data.type);
        }
    }
    
    updateDashboardData(payload) {
        // 更新統計數據
        if (payload.statistics) {
            Object.entries(payload.statistics).forEach(([key, value]) => {
                this.notifySubscribers('stat_update', { key, value });
            });
        }
        
        // 更新圖表數據
        if (payload.charts) {
            Object.entries(payload.charts).forEach(([chartId, chartData]) => {
                this.notifySubscribers('chart_update', { chartId, chartData });
            });
        }
        
        // 更新警報
        if (payload.alerts) {
            payload.alerts.forEach(alert => {
                this.notifySubscribers('alert', alert);
            });
        }
    }
    
    updateSpecificStat(payload) {
        const { key, value, change, trend } = payload;
        
        // 尋找並更新對應的統計卡片
        const statElement = document.querySelector(`[data-stat="${key}"]`);
        if (statElement && window.dashboardManager) {
            window.dashboardManager.updateStatCard(statElement, {
                value: value,
                change: change,
                trend: trend
            });
        }
        
        this.notifySubscribers('stat_update', payload);
    }
    
    handleAlert(alert) {
        // 顯示實時警報
        this.showRealtimeAlert(alert);
        this.notifySubscribers('alert', alert);
    }
    
    showRealtimeAlert(alert) {
        const alertContainer = document.getElementById('realtimeAlerts');
        if (!alertContainer) return;
        
        const alertElement = document.createElement('div');
        alertElement.className = `alert alert-${alert.type} animate-fade-in`;
        alertElement.innerHTML = `
            <div class="flex items-center space-x-3">
                <div class="flex-shrink-0">
                    ${this.getAlertIcon(alert.type)}
                </div>
                <div class="flex-1">
                    <p class="font-medium">${alert.title}</p>
                    <p class="text-sm opacity-90">${alert.message}</p>
                </div>
                <button onclick="this.parentElement.parentElement.remove()" 
                        class="flex-shrink-0 text-current opacity-70 hover:opacity-100">
                    <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                        <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12" />
                    </svg>
                </button>
            </div>
        `;
        
        alertContainer.appendChild(alertElement);
        
        // 自動移除警報
        setTimeout(() => {
            if (alertElement.parentElement) {
                alertElement.remove();
            }
        }, alert.duration || 5000);
    }
    
    getAlertIcon(type) {

            info: '<svg class="w-5 h-5" fill="currentColor" viewBox="0 0 20 20"><path fill-rule="evenodd" d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zm-7-4a1 1 0 11-2 0 1 1 0 012 0zM9 9a1 1 0 000 2v3a1 1 0 001 1h1a1 1 0 100-2v-3a1 1 0 00-1-1H9z" clip-rule="evenodd" /></svg>',

        };
        return icons[type] || icons.info;

    subscribe(eventType, callback) {
        if (!this.subscribers.has(eventType)) {
            this.subscribers.set(eventType, new Set());
        }
        this.subscribers.get(eventType).add(callback);
    }
    
    unsubscribe(eventType, callback) {
        if (this.subscribers.has(eventType)) {
            this.subscribers.get(eventType).delete(callback);
        }
    }
    
    notifySubscribers(eventType, data) {
        if (this.subscribers.has(eventType)) {
            this.subscribers.get(eventType).forEach(callback => {
                try {
                    callback(data);
                } catch (error) {
                    console.error('訂閱者回調錯誤:', error);
                }
            });
        }
    }
    
    handleReconnection() {
        if (this.reconnectAttempts < this.maxReconnectAttempts) {
            this.reconnectAttempts++;
            const delay = Math.pow(2, this.reconnectAttempts) * 1000; // 指數退避
            
            console.log(`嘗試重新連線 (${this.reconnectAttempts}/${this.maxReconnectAttempts})，${delay}ms 後重試`);
            
            setTimeout(() => {
                this.connectWebSocket();
            }, delay);
        } else {
            console.log('WebSocket 重連次數達到上限，切換到輪詢模式');
            this.startFallbackPolling();
        }
    }
    
    startFallbackPolling() {
        if (this.fallbackTimer) return;
        
        console.log('啟動輪詢更新機制');
        this.fallbackTimer = setInterval(() => {
            this.fetchDataViaPoll();
        }, this.fallbackInterval);
    }
    
    stopFallbackPolling() {
        if (this.fallbackTimer) {
            clearInterval(this.fallbackTimer);
            this.fallbackTimer = null;
            console.log('停止輪詢更新機制');
        }
    }
    
    async fetchDataViaPoll() {
        try {
            const response = await fetch(this.apiEndpoint, {
                method: 'GET',
                headers: {
                    'Accept': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest',
                    'Authorization': `Bearer ${this.getAuthToken()}`
                }
            });
            
            if (response.ok) {
                const data = await response.json();
                this.handleRealtimeData({
                    type: 'dashboard_update',
                    payload: data
                });
            }
        } catch (error) {
            console.error('輪詢更新錯誤:', error);
        }
    }
    
    getAuthToken() {
        return document.querySelector('meta[name="csrf-token"]')?.getAttribute('content') || '';
    }
    
    pauseUpdates() {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(JSON.stringify({ type: 'pause' }));
        }
        this.stopFallbackPolling();
    }
    
    resumeUpdates() {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(JSON.stringify({ type: 'resume' }));
        } else if (!this.isConnected) {
            this.startFallbackPolling();
        }
    }
    
    destroy() {
        if (this.ws) {
            this.ws.close();
        }
        this.stopFallbackPolling();
        this.subscribers.clear();
    }
}
```

```javascript
// 檔案: frontend/resources/js/components/dashboard/PerformanceMonitor.js

class PerformanceMonitor {
    constructor() {
        this.metrics = new Map();
        this.observers = new Map();
        this.init();
    }
    
    init() {
        // 監控頁面載入效能
        this.monitorPageLoad();
        
        // 監控圖表渲染效能
        this.monitorChartPerformance();
        
        // 監控記憶體使用
        this.monitorMemoryUsage();
        
        // 監控網路請求
        this.monitorNetworkRequests();
    }
    
    monitorPageLoad() {
        window.addEventListener('load', () => {
            const perfData = performance.getEntriesByType('navigation')[0];
            
            this.recordMetric('page_load', {
                domContentLoaded: perfData.domContentLoadedEventEnd - perfData.domContentLoadedEventStart,
                loadComplete: perfData.loadEventEnd - perfData.loadEventStart,
                firstPaint: this.getFirstPaint(),
                firstContentfulPaint: this.getFirstContentfulPaint()
            });
        });
    }
    
    monitorChartPerformance() {
        const originalCreateChart = Chart.prototype.render;
        Chart.prototype.render = function() {
            const startTime = performance.now();
            const result = originalCreateChart.apply(this, arguments);
            const endTime = performance.now();
            
            window.performanceMonitor?.recordMetric('chart_render', {
                chartId: this.canvas.id,
                renderTime: endTime - startTime
            });
            
            return result;
        };
    }
    
    monitorMemoryUsage() {
        if ('memory' in performance) {
            setInterval(() => {
                this.recordMetric('memory_usage', {
                    used: performance.memory.usedJSHeapSize,
                    total: performance.memory.totalJSHeapSize,
                    limit: performance.memory.jsHeapSizeLimit
                });
            }, 10000); // 每10秒記錄一次
        }
    }
    
    monitorNetworkRequests() {
        const originalFetch = window.fetch;
        window.fetch = async function(...args) {
            const startTime = performance.now();
            const url = args[0];
            
            try {
                const response = await originalFetch.apply(this, args);
                const endTime = performance.now();
                
                window.performanceMonitor?.recordMetric('network_request', {
                    url: url,
                    method: args[1]?.method || 'GET',
                    status: response.status,
                    duration: endTime - startTime,
                    success: response.ok
                });
                
                return response;
            } catch (error) {
                const endTime = performance.now();
                
                window.performanceMonitor?.recordMetric('network_request', {
                    url: url,
                    method: args[1]?.method || 'GET',
                    duration: endTime - startTime,
                    success: false,
                    error: error.message
                });
                
                throw error;
            }
        };
    }
    
    recordMetric(name, data) {
        if (!this.metrics.has(name)) {
            this.metrics.set(name, []);
        }
        
        this.metrics.get(name).push({
            timestamp: Date.now(),
            ...data
        });
        
        // 保持最近 100 筆記錄
        const records = this.metrics.get(name);
        if (records.length > 100) {
            records.splice(0, records.length - 100);
        }
        
        // 通知觀察者
        this.notifyObservers(name, data);
    }
    
    getMetric(name) {
        return this.metrics.get(name) || [];
    }
    
    getAverageMetric(name, field) {
        const records = this.getMetric(name);
        if (records.length === 0) return 0;
        
        const values = records.map(r => r[field]).filter(v => typeof v === 'number');
        return values.reduce((sum, val) => sum + val, 0) / values.length;
    }
    
    getPerformanceReport() {
        return {
            pageLoad: {
                averageDomContentLoaded: this.getAverageMetric('page_load', 'domContentLoaded'),
                averageLoadComplete: this.getAverageMetric('page_load', 'loadComplete')
            },
            chartRender: {
                averageRenderTime: this.getAverageMetric('chart_render', 'renderTime'),
                totalCharts: new Set(this.getMetric('chart_render').map(r => r.chartId)).size
            },
            networkRequests: {
                averageResponseTime: this.getAverageMetric('network_request', 'duration'),
                successRate: this.calculateSuccessRate(),
                totalRequests: this.getMetric('network_request').length
            },
            memoryUsage: {
                currentUsed: this.getCurrentMemoryUsage(),
                peakUsed: this.getPeakMemoryUsage()
            }
        };
    }
    
    calculateSuccessRate() {
        const requests = this.getMetric('network_request');
        if (requests.length === 0) return 0;
        
        const successful = requests.filter(r => r.success).length;
        return (successful / requests.length) * 100;
    }
    
    getCurrentMemoryUsage() {
        const memoryRecords = this.getMetric('memory_usage');
        if (memoryRecords.length === 0) return 0;
        
        return memoryRecords[memoryRecords.length - 1].used;
    }
    
    getPeakMemoryUsage() {
        const memoryRecords = this.getMetric('memory_usage');
        if (memoryRecords.length === 0) return 0;
        
        return Math.max(...memoryRecords.map(r => r.used));
    }
    
    getFirstPaint() {
        const paintEntries = performance.getEntriesByType('paint');
        const firstPaint = paintEntries.find(entry => entry.name === 'first-paint');
        return firstPaint ? firstPaint.startTime : 0;
    }
    
    getFirstContentfulPaint() {
        const paintEntries = performance.getEntriesByType('paint');
        const fcp = paintEntries.find(entry => entry.name === 'first-contentful-paint');
        return fcp ? fcp.startTime : 0;
    }
    
    subscribe(metricName, callback) {
        if (!this.observers.has(metricName)) {
            this.observers.set(metricName, new Set());
        }
        this.observers.get(metricName).add(callback);
    }
    
    notifyObservers(metricName, data) {
        if (this.observers.has(metricName)) {
            this.observers.get(metricName).forEach(callback => {
                try {
                    callback(data);
                } catch (error) {
                    console.error('效能監控觀察者錯誤:', error);
                }
            });
        }
    }
    
    exportReport() {
        const report = this.getPerformanceReport();
        const blob = new Blob([JSON.stringify(report, null, 2)], {
            type: 'application/json'
        });
        
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `dashboard-performance-${new Date().toISOString()}.json`;
        a.click();
        
        URL.revokeObjectURL(url);
    }
}

// 全域匯出
window.PerformanceMonitor = PerformanceMonitor;

// 自動初始化
document.addEventListener('DOMContentLoaded', () => {
    window.performanceMonitor = new PerformanceMonitor();
    
    // 在開發模式下顯示效能資訊
    if (process.env.NODE_ENV === 'development') {
        console.log('效能監控器已啟動');
        
        // 每30秒輸出效能報告
        setInterval(() => {
            console.log('效能報告:', window.performanceMonitor.getPerformanceReport());
        }, 30000);
    }
```
