```php
<script>
function multiStepForm(steps, persistKey) {
    return {
        steps: steps,
        currentStep: 1,
        formData: {},
        errors: {},
        loading: false,
        submitting: false,
        draftSaving: false,
        showExitConfirm: false,
        hasUnsavedChanges: false,
        persistKey: persistKey,
        
        init() {
            // 載入保存的資料
            this.loadSavedData();
            
            // 監聽頁面離開
            window.addEventListener('beforeunload', (e) => {
                if (this.hasUnsavedChanges) {
                    e.preventDefault();
                    e.returnValue = '';
                }
            });
            
            // 表單變更監聽
            this.$watch('formData', () => {
                this.hasUnsavedChanges = true;
            }, { deep: true });
        },
        
        async nextStep() {
            if (this.loading) return;
            
            // 驗證當前步驟
            const isValid = await this.validateCurrentStep();
            if (!isValid) return;
            
            this.loading = true;
            
            try {
                // 保存當前步驟資料
                await this.saveStepData();
                
                if (this.currentStep < this.steps.length) {
                    this.currentStep++;
                    this.scrollToTop();
                }
            } catch (error) {
                this.showError('步驟切換失敗：' + error.message);
            } finally {
                this.loading = false;
            }
        },
        
        previousStep() {
            if (this.currentStep > 1) {
                this.currentStep--;
                this.scrollToTop();
            }
        },
        
        async validateCurrentStep() {
            this.errors = {};
            
            const stepValidation = this.getStepValidation(this.currentStep);
            if (!stepValidation) return true;
            
            const stepData = this.getStepData(this.currentStep);
            
            for (const [field, rules] of Object.entries(stepValidation)) {
                const value = stepData[field];
                
                for (const rule of rules) {
                    const error = this.validateField(field, value, rule);
                    if (error) {
                        this.errors[field] = error;
                        break;
                    }
                }
            }
            
            return Object.keys(this.errors).length === 0;
        },
        
        validateField(field, value, rule) {
            switch (rule.type) {
                case 'required':
                    if (!value || (Array.isArray(value) && value.length === 0)) {
                        return rule.message || `${field} 為必填欄位`;
                    }
                    break;
                    
                case 'min':
                    if (value && value.length < rule.value) {
                        return rule.message || `${field} 至少需要 ${rule.value} 個字元`;
                    }
                    break;
                    
                case 'email':
                    if (value && !/^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$/.test(value)) {
                        return rule.message || `${field} 格式不正確`;
                    }
                    break;
                    
                case 'numeric':
                    if (value && !/^\\d+(\\.\\d+)?$/.test(value)) {
                        return rule.message || `${field} 必須為數字`;
                    }
                    break;
            }
            
            return null;
        },
        
        getStepValidation(step) {
            const validations = {
                1: {
                    'customer_id': [{ type: 'required' }],
                    'title': [{ type: 'required' }, { type: 'min', value: 3 }],
                    'quote_date': [{ type: 'required' }],
                    'valid_until': [{ type: 'required' }]
                },
                2: {
                    'items': [{ type: 'required', message: '至少需要選擇一個商品' }]
                },
                3: {
                    'tax_rate': [{ type: 'numeric' }],
                    'shipping_cost': [{ type: 'numeric' }]
                },
                4: {
                    'payment_terms': [{ type: 'required' }]
                }
            };
            
            return validations[step];
        },
        
        getStepData(step) {
            // 從表單元素中提取當前步驟的資料
            const formElement = document.getElementById(this.formId || 'multiStepForm');
            const stepData = {};
            
            const inputs = formElement.querySelectorAll(`[data-step="${step}"] input, [data-step="${step}"] select, [data-step="${step}"] textarea`);
            
            inputs.forEach(input => {
                if (input.name) {
                    if (input.type === 'checkbox' || input.type === 'radio') {
                        if (input.checked) {
                            stepData[input.name] = input.value;
                        }
                    } else {
                        stepData[input.name] = input.value;
                    }
                }
            });
            
            return stepData;
        },
        
        async saveStepData() {
            const stepData = this.getStepData(this.currentStep);
            this.formData[`step_${this.currentStep}`] = stepData;
            
            // 保存到 localStorage
            if (this.persistKey) {
                localStorage.setItem(this.persistKey, JSON.stringify(this.formData));
            }
        },
        
        loadSavedData() {
            if (this.persistKey) {
                const savedData = localStorage.getItem(this.persistKey);
                if (savedData) {
                    try {
                        this.formData = JSON.parse(savedData);
                        this.restoreFormData();
                    } catch (error) {
                        console.error('載入保存資料失敗:', error);
                    }
                }
            }
        },
        
        restoreFormData() {
            // 將保存的資料填回表單
            Object.entries(this.formData).forEach(([stepKey, stepData]) => {
                Object.entries(stepData).forEach(([fieldName, value]) => {
                    const input = document.querySelector(`[name="${fieldName}"]`);
                    if (input) {
                        if (input.type === 'checkbox' || input.type === 'radio') {
                            input.checked = input.value === value;
                        } else {
                            input.value = value;
                        }
                    }
                });
            });
        },
        
        async saveDraft() {
            if (this.draftSaving) return;
            
            this.draftSaving = true;
            
            try {
                await this.saveStepData();
                
                // 發送到後端保存草稿
                const response = await fetch('/api/quotes/draft', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content
                    },
                    body: JSON.stringify(this.formData)
                });
                
                if (response.ok) {
                    this.showSuccess('草稿已保存');
                    this.hasUnsavedChanges = false;
                } else {
                    throw new Error('保存草稿失敗');
                }
            } catch (error) {
                this.showError('保存草稿失敗：' + error.message);
            } finally {
                this.draftSaving = false;
            }
        },
        
        async handleSubmit() {
            if (this.submitting) return;
            
            // 最終驗證所有步驟
            for (let step = 1; step <= this.steps.length; step++) {
                this.currentStep = step;
                const isValid = await this.validateCurrentStep();
                if (!isValid) {
                    this.showError(`第 ${step} 步驟有錯誤，請檢查後重試`);
                    return;
                }
            }
            
            this.submitting = true;
            
            try {
                await this.saveStepData();
                
                // 提交完整表單
                const response = await fetch('/api/quotes', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content
                    },
                    body: JSON.stringify(this.formData)
                });
                
                if (response.ok) {
                    const result = await response.json();
                    this.showSuccess('報價單建立成功');
                    this.hasUnsavedChanges = false;
                    
                    // 清除保存的資料
                    if (this.persistKey) {
                        localStorage.removeItem(this.persistKey);
                    }
                    
                    // 跳轉到報價單詳情頁
                    window.location.href = `/quotes/${result.id}`;
                } else {
                    const error = await response.json();
                    throw new Error(error.message || '建立報價單失敗');
                }
            } catch (error) {
                this.showError('提交失敗：' + error.message);
            } finally {
                this.submitting = false;
            }
        },
        
        scrollToTop() {
            window.scrollTo({ top: 0, behavior: 'smooth' });
        },
        
        showSuccess(message) {
            // 顯示成功訊息 (可以整合通知系統)
            console.log('Success:', message);
        },
        
        showError(message) {
            // 顯示錯誤訊息 (可以整合通知系統)
            console.error('Error:', message);
        },
        
        confirmExit() {
            this.hasUnsavedChanges = false;
            this.showExitConfirm = false;
            // 執行離開邏輯
        }
    }
}
</script>
@endpush
```

```php
<script>
function productSelector() {
    return {
        // 搜尋狀態
        searchTerm: '',
        selectedCategory: '',
        stockFilter: '',
        selectedBrand: '',
        sortBy: 'name_asc',
        showAdvancedFilters: false,
        priceRange: { min: '', max: '' },
        
        // 顯示狀態
        viewMode: 'grid',
        searchResults: [],
        selectedProducts: [],
        loading: false,
        
        init() {
            // 初始化時載入推薦商品
            this.loadRecommendedProducts();
        },
        
        async searchProducts() {
            if (this.loading) return;
            
            this.loading = true;
            
            try {
                const params = new URLSearchParams({
                    search: this.searchTerm,
                    category: this.selectedCategory,
                    stock_filter: this.stockFilter,
                    brand: this.selectedBrand,
                    sort: this.sortBy,
                    price_min: this.priceRange.min,
                    price_max: this.priceRange.max,
                    limit: 50
                });
                
                const response = await fetch(`/api/products/search?${params}`);
                if (!response.ok) throw new Error('搜尋失敗');
                
                const data = await response.json();
                this.searchResults = data.products || [];
            } catch (error) {
                console.error('搜尋商品失敗:', error);
                this.searchResults = [];
            } finally {
                this.loading = false;
            }
        },
        
        async loadRecommendedProducts() {
            try {
                const response = await fetch('/api/products/recommended');
                if (response.ok) {
                    const data = await response.json();
                    this.searchResults = data.products || [];
                }
            } catch (error) {
                console.error('載入推薦商品失敗:', error);
            }
        },
        
        addProduct(product) {
            if (this.isProductSelected(product.id)) return;
            
            this.selectedProducts.push({
                product_id: product.id,
                name: product.name,
                sku: product.sku,
                image_url: product.image_url,
                quantity: 1,
                unit_price: product.unit_price,
                unit: product.unit,
                stock_quantity: product.stock_quantity,
                cost_price: product.cost_price
            });
            
            // 觸發事件通知父組件
            this.$dispatch('products-updated', this.selectedProducts);
        },
        
        removeProduct(index) {
            this.selectedProducts.splice(index, 1);
            this.$dispatch('products-updated', this.selectedProducts);
        },
        
        updateQuantity(index, quantity) {
            const qty = parseInt(quantity) || 1;
            const maxQty = this.selectedProducts[index].stock_quantity;
            
            this.selectedProducts[index].quantity = Math.min(qty, maxQty);
            this.$dispatch('products-updated', this.selectedProducts);
        },
        
        updatePrice(index, price) {
            this.selectedProducts[index].unit_price = parseFloat(price) || 0;
            this.$dispatch('products-updated', this.selectedProducts);
        },
        
        isProductSelected(productId) {
            return this.selectedProducts.some(item => item.product_id === productId);
        },
        
        calculateSubtotal() {
            return this.selectedProducts.reduce((sum, item) => {
                return sum + (item.quantity * item.unit_price);
            }, 0);
        },
        
        formatPrice(price) {
            return parseFloat(price || 0).toLocaleString('zh-TW', {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            });
        },
        
        getStockClass(current, minimum) {
            if (current === 0) return 'text-red-600';
            if (current <= minimum) return 'text-orange-600';
            return 'text-green-600';
        }
    }
}
```
