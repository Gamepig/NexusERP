// 檔案: backend/internal/middleware/cors.go

package middleware

import (
    "net/http"
    "strings"
    "time"
    
    "github.com/gin-gonic/gin"
    "github.com/NexusERP/backend/internal/config"
    "github.com/NexusERP/backend/internal/models"
    "github.com/NexusERP/backend/internal/services"
)

type CORSConfig struct {
    AllowOrigins     []string
    AllowMethods     []string
    AllowHeaders     []string
    AllowCredentials bool
    MaxAge          int
    ExposeHeaders   []string
}

type DynamicCORS struct {
    configService *services.ConfigService
    defaultConfig *CORSConfig
    cache        map[string]*CORSConfig
    cacheExpiry  time.Time
}

func NewDynamicCORS(configService *services.ConfigService) *DynamicCORS {
    return &DynamicCORS{
        configService: configService,
        defaultConfig: &CORSConfig{
            AllowOrigins:     []string{"http://localhost:3000", "http://localhost:8000"},
            AllowMethods:     []string{"GET", "POST", "PUT", "DELETE", "OPTIONS", "PATCH"},
            AllowHeaders:     []string{"Origin", "Content-Type", "Accept", "Authorization", "X-Requested-With", "X-API-Key"},
            AllowCredentials: true,
            MaxAge:          86400, // 24 hours
            ExposeHeaders:   []string{"X-Total-Count", "X-Page-Count"},
        },
        cache: make(map[string]*CORSConfig),
    }
}

func (dc *DynamicCORS) Middleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        origin := c.Request.Header.Get("Origin")
        
        // 取得適用的CORS配置
        corsConfig := dc.getCORSConfig(origin, c)
        
        // 設定CORS標頭
        dc.setCORSHeaders(c, corsConfig, origin)
        
        // 記錄CORS請求
        dc.logCORSRequest(c, origin, corsConfig)
        
        // 處理預檢請求
        if c.Request.Method == "OPTIONS" {
            c.AbortWithStatus(http.StatusNoContent)
            return
        }
        
        c.Next()
    }
}

func (dc *DynamicCORS) getCORSConfig(origin string, c *gin.Context) *CORSConfig {
    // 檢查快取
    if time.Now().Before(dc.cacheExpiry) {
        if config, exists := dc.cache[origin]; exists {
            return config
        }
    }
    
    // 從資料庫載入配置
    corsSettings, err := dc.configService.GetCORSSettings(origin)
    if err != nil {
        return dc.defaultConfig
    }
    
    config := &CORSConfig{
        AllowOrigins:     corsSettings.AllowOrigins,
        AllowMethods:     corsSettings.AllowMethods,
        AllowHeaders:     corsSettings.AllowHeaders,
        AllowCredentials: corsSettings.AllowCredentials,
        MaxAge:          corsSettings.MaxAge,
        ExposeHeaders:   corsSettings.ExposeHeaders,
    }
    
    // 更新快取
    dc.cache[origin] = config
    dc.cacheExpiry = time.Now().Add(5 * time.Minute)
    
    return config
}

func (dc *DynamicCORS) setCORSHeaders(c *gin.Context, config *CORSConfig, origin string) {
    // 檢查來源是否被允許
    if dc.isOriginAllowed(origin, config.AllowOrigins) {
        c.Header("Access-Control-Allow-Origin", origin)
    }
    
    c.Header("Access-Control-Allow-Methods", strings.Join(config.AllowMethods, ", "))
    c.Header("Access-Control-Allow-Headers", strings.Join(config.AllowHeaders, ", "))
    
    if config.AllowCredentials {
        c.Header("Access-Control-Allow-Credentials", "true")
    }
    
    if config.MaxAge > 0 {
        c.Header("Access-Control-Max-Age", string(rune(config.MaxAge)))
    }
    
    if len(config.ExposeHeaders) > 0 {
        c.Header("Access-Control-Expose-Headers", strings.Join(config.ExposeHeaders, ", "))
    }
}

func (dc *DynamicCORS) isOriginAllowed(origin string, allowedOrigins []string) bool {
    for _, allowed := range allowedOrigins {
        if allowed == "*" || allowed == origin {
            return true
        }
        
        // 支援萬用字元匹配
        if strings.Contains(allowed, "*") {
            pattern := strings.ReplaceAll(allowed, "*", ".*")
            if matched, _ := regexp.MatchString(pattern, origin); matched {
                return true
            }
        }
    }
    return false
}

func (dc *DynamicCORS) logCORSRequest(c *gin.Context, origin string, config *CORSConfig) {
    // 記錄CORS請求到日誌系統
    log := &models.CORSLog{
        Origin:    origin,
        Method:    c.Request.Method,
        Path:      c.Request.URL.Path,
        UserAgent: c.Request.Header.Get("User-Agent"),
        IP:        c.ClientIP(),
        Allowed:   dc.isOriginAllowed(origin, config.AllowOrigins),
        Timestamp: time.Now(),
    }
    
    go func() {
        if err := dc.configService.LogCORSRequest(log); err != nil {
            // 記錄錯誤但不影響主要流程
            fmt.Printf("Failed to log CORS request: %v\n", err)
        }
    }()
}

// CORS配置管理
type CORSSettingsManager struct {
    db *gorm.DB
}

func NewCORSSettingsManager(db *gorm.DB) *CORSSettingsManager {
    return &CORSSettingsManager{db: db}
}

func (csm *CORSSettingsManager) UpdateCORSSettings(settings *models.CORSSettings) error {
    // 驗證設定的安全性
    if err := csm.validateCORSSettings(settings); err != nil {
        return err
    }
    
    return csm.db.Save(settings).Error
}

func (csm *CORSSettingsManager) validateCORSSettings(settings *models.CORSSettings) error {
    // 檢查危險的配置
    for _, origin := range settings.AllowOrigins {
        if origin == "*" && settings.AllowCredentials {
            return errors.New("不能在允許憑證的情況下使用萬用字元來源")
        }
    }
    
    // 檢查允許的方法
    dangerousMethods := []string{"TRACE", "CONNECT"}
    for _, method := range settings.AllowMethods {
        for _, dangerous := range dangerousMethods {
            if method == dangerous {
                return fmt.Errorf("不允許的HTTP方法: %s", method)
            }
        }
    }
    
    return nil
}

<?php
// 檔案: frontend/app/Http/Controllers/Admin/CORSController.php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\CORSSettings;
use App\Models\CORSLog;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Validator;

class CORSController extends Controller
{
    public function index()
    {
        $settings = CORSSettings::orderBy('created_at', 'desc')->paginate(20);
        $recentLogs = CORSLog::with('user')
            ->orderBy('created_at', 'desc')

    public function create()
    {
        return view('admin.cors.create');
    }
    
    public function store(Request $request)
    {
        $validator = $this->validateCORSSettings($request);
        
        if ($validator->fails()) {
            return redirect()->back()
                ->withErrors($validator)
                ->withInput();
        }
        
        $settings = CORSSettings::create([
            'name' => $request->name,
            'origin_pattern' => $request->origin_pattern,
            'allow_origins' => $request->allow_origins,
            'allow_methods' => $request->allow_methods,
            'allow_headers' => $request->allow_headers,
            'allow_credentials' => $request->boolean('allow_credentials'),
            'max_age' => $request->max_age,
            'expose_headers' => $request->expose_headers,
            'is_active' => $request->boolean('is_active'),
        ]);
        
        return redirect()->route('admin.cors.index')
            ->with('success', 'CORS設定建立成功');
    }
    
    public function edit(CORSSettings $corsSettings)
    {
        return view('admin.cors.edit', compact('corsSettings'));
    }
    
    public function update(Request $request, CORSSettings $corsSettings)
    {
        $validator = $this->validateCORSSettings($request);
        
        if ($validator->fails()) {
            return redirect()->back()
                ->withErrors($validator)
                ->withInput();
        }
        
        $corsSettings->update([
            'name' => $request->name,
            'origin_pattern' => $request->origin_pattern,
            'allow_origins' => $request->allow_origins,
            'allow_methods' => $request->allow_methods,
            'allow_headers' => $request->allow_headers,
            'allow_credentials' => $request->boolean('allow_credentials'),
            'max_age' => $request->max_age,
            'expose_headers' => $request->expose_headers,
            'is_active' => $request->boolean('is_active'),
        ]);
        
        return redirect()->route('admin.cors.index')
            ->with('success', 'CORS設定更新成功');
    }
    
    public function destroy(CORSSettings $corsSettings)
    {
        $corsSettings->delete();
        
        return redirect()->route('admin.cors.index')
            ->with('success', 'CORS設定刪除成功');
    }
    
    public function logs()
    {
        $logs = CORSLog::with('user')
            ->orderBy('created_at', 'desc')
            ->paginate(50);
            
        return view('admin.cors.logs', compact('logs'));
    }
    
    public function testCORS(Request $request)
    {
        $origin = $request->origin;
        $method = $request->method;
        
        // 測試CORS配置
        $result = $this->simulateCORSRequest($origin, $method);
        
        return response()->json($result);
    }
    
    private function validateCORSSettings(Request $request)
    {
        return Validator::make($request->all(), [
            'name' => 'required|string|max:255',
            'origin_pattern' => 'required|string|max:255',
            'allow_origins' => 'required|array',
            'allow_origins.*' => 'required|string',
            'allow_methods' => 'required|array',
            'allow_methods.*' => 'required|string|in:GET,POST,PUT,DELETE,OPTIONS,PATCH,HEAD',
            'allow_headers' => 'required|array',
            'allow_headers.*' => 'required|string',
            'max_age' => 'required|integer|min:0|max:86400',
            'expose_headers' => 'array',
            'expose_headers.*' => 'string',
        ], [
            'name.required' => '設定名稱為必填',
            'origin_pattern.required' => '來源模式為必填',
            'allow_origins.required' => '允許來源為必填',
            'allow_methods.required' => '允許方法為必填',
            'allow_headers.required' => '允許標頭為必填',
        ]);
    }
    
    private function simulateCORSRequest($origin, $method)
    {
        // 模擬CORS請求驗證邏輯
        $settings = CORSSettings::where('is_active', true)
            ->where(function($query) use ($origin) {
                $query->where('origin_pattern', $origin)
                      ->orWhere('origin_pattern', '*')
                      ->orWhere('origin_pattern', 'like', '%*%');
            })
            ->first();
            
        if (!$settings) {
            return [
                'allowed' => false,
                'reason' => '沒有找到匹配的CORS設定'
            ];
        }
        
        if (!in_array($method, $settings->allow_methods)) {
            return [
                'allowed' => false,
                'reason' => '方法不被允許'
            ];
        }
        
        return [
            'allowed' => true,
            'settings' => $settings,
            'headers' => [
                'Access-Control-Allow-Origin' => $origin,
                'Access-Control-Allow-Methods' => implode(', ', $settings->allow_methods),
                'Access-Control-Allow-Headers' => implode(', ', $settings->allow_headers),
                'Access-Control-Allow-Credentials' => $settings->allow_credentials ? 'true' : 'false',
                'Access-Control-Max-Age' => $settings->max_age,
            ]
        ];
    }
}

// 檔案: backend/internal/services/api_key_service.go

package services

import (
    "crypto/rand"
    "crypto/sha256"
    "encoding/base64"
    "encoding/hex"
    "fmt"
    "time"
    
    "github.com/NexusERP/backend/internal/models"
    "gorm.io/gorm"
)

type APIKeyService struct {
    db *gorm.DB
}

func NewAPIKeyService(db *gorm.DB) *APIKeyService {
    return &APIKeyService{db: db}
}

type APIKeyRequest struct {
    Name        string    `json:"name" binding:"required"`
    Description string    `json:"description"`
    ExpiresAt   time.Time `json:"expires_at"`
    Permissions []string  `json:"permissions"`
    RateLimit   int       `json:"rate_limit"`
    IPWhitelist []string  `json:"ip_whitelist"`
}

type APIKeyResponse struct {
    ID          uint      `json:"id"`
    Name        string    `json:"name"`
    Key         string    `json:"key"` // 只在建立時返回
    KeyPreview  string    `json:"key_preview"`
    Description string    `json:"description"`
    ExpiresAt   time.Time `json:"expires_at"`
    Permissions []string  `json:"permissions"`
    RateLimit   int       `json:"rate_limit"`
    IPWhitelist []string  `json:"ip_whitelist"`
    IsActive    bool      `json:"is_active"`
    LastUsedAt  time.Time `json:"last_used_at"`
    CreatedAt   time.Time `json:"created_at"`
}

func (aks *APIKeyService) CreateAPIKey(userID uint, req *APIKeyRequest) (*APIKeyResponse, error) {
    // 生成API Key
    key, keyHash, err := aks.generateAPIKey()
    if err != nil {
        return nil, fmt.Errorf("生成API Key失敗: %w", err)
    }
    
    // 建立資料庫記錄
    apiKey := &models.APIKey{
        UserID:      userID,
        Name:        req.Name,
        Description: req.Description,
        KeyHash:     keyHash,
        KeyPreview:  key[:8] + "..." + key[len(key)-4:],
        ExpiresAt:   req.ExpiresAt,
        Permissions: req.Permissions,
        RateLimit:   req.RateLimit,
        IPWhitelist: req.IPWhitelist,
        IsActive:    true,
        CreatedAt:   time.Now(),
    }
    
    if err := aks.db.Create(apiKey).Error; err != nil {
        return nil, fmt.Errorf("儲存API Key失敗: %w", err)
    }
    
    // 記錄API Key建立事件
    aks.logAPIKeyEvent(apiKey.ID, "created", userID, "")
    
    return &APIKeyResponse{
        ID:          apiKey.ID,
        Name:        apiKey.Name,
        Key:         key, // 完整的key只在建立時返回
        KeyPreview:  apiKey.KeyPreview,
        Description: apiKey.Description,
        ExpiresAt:   apiKey.ExpiresAt,
        Permissions: apiKey.Permissions,
        RateLimit:   apiKey.RateLimit,
        IPWhitelist: apiKey.IPWhitelist,
        IsActive:    apiKey.IsActive,
        CreatedAt:   apiKey.CreatedAt,
    }, nil
}

func (aks *APIKeyService) ValidateAPIKey(key string, clientIP string) (*models.APIKey, error) {
    keyHash := aks.hashAPIKey(key)
    
    var apiKey models.APIKey
    err := aks.db.Where("key_hash = ? AND is_active = ? AND (expires_at IS NULL OR expires_at > ?)", 
        keyHash, true, time.Now()).First(&apiKey).Error
    
    if err != nil {
        if err == gorm.ErrRecordNotFound {
            return nil, fmt.Errorf("無效的API Key")
        }
        return nil, fmt.Errorf("驗證API Key失敗: %w", err)
    }
    
    // 檢查IP白名單
    if len(apiKey.IPWhitelist) > 0 && !aks.isIPWhitelisted(clientIP, apiKey.IPWhitelist) {
        aks.logAPIKeyEvent(apiKey.ID, "ip_blocked", 0, clientIP)
        return nil, fmt.Errorf("IP地址不在白名單中")
    }
    
    // 更新最後使用時間
    aks.updateLastUsed(apiKey.ID)
    
    // 記錄使用事件
    aks.logAPIKeyEvent(apiKey.ID, "used", 0, clientIP)
    
    return &apiKey, nil
}

func (aks *APIKeyService) ListAPIKeys(userID uint) ([]*APIKeyResponse, error) {
    var apiKeys []models.APIKey
    err := aks.db.Where("user_id = ?", userID).Find(&apiKeys).Error
    if err != nil {
        return nil, fmt.Errorf("查詢API Keys失敗: %w", err)
    }
    
    result := make([]*APIKeyResponse, len(apiKeys))
    for i, key := range apiKeys {
        result[i] = &APIKeyResponse{
            ID:          key.ID,
            Name:        key.Name,
            KeyPreview:  key.KeyPreview,
            Description: key.Description,
            ExpiresAt:   key.ExpiresAt,
            Permissions: key.Permissions,
            RateLimit:   key.RateLimit,
            IPWhitelist: key.IPWhitelist,
            IsActive:    key.IsActive,
            LastUsedAt:  key.LastUsedAt,
            CreatedAt:   key.CreatedAt,
        }
    }
    
    return result, nil
}

func (aks *APIKeyService) RevokeAPIKey(keyID uint, userID uint) error {
    result := aks.db.Model(&models.APIKey{}).
        Where("id = ? AND user_id = ?", keyID, userID).
        Update("is_active", false)
    
    if result.Error != nil {
        return fmt.Errorf("撤銷API Key失敗: %w", result.Error)
    }
    
    if result.RowsAffected == 0 {
        return fmt.Errorf("API Key不存在或無權限")
    }
    
    // 記錄撤銷事件
    aks.logAPIKeyEvent(keyID, "revoked", userID, "")
    
    return nil
}

func (aks *APIKeyService) generateAPIKey() (string, string, error) {
    // 生成32字節的隨機數據
    randomBytes := make([]byte, 32)
    if _, err := rand.Read(randomBytes); err != nil {
        return "", "", err
    }
    
    // 轉換為base64
    key := base64.URLEncoding.EncodeToString(randomBytes)
    
    // 生成hash用於儲存
    keyHash := aks.hashAPIKey(key)
    
    return "ner_" + key, keyHash, nil
}

func (aks *APIKeyService) hashAPIKey(key string) string {
    hash := sha256.Sum256([]byte(key))
    return hex.EncodeToString(hash[:])
}

func (aks *APIKeyService) isIPWhitelisted(clientIP string, whitelist []string) bool {
    for _, allowedIP := range whitelist {
        if allowedIP == clientIP {
            return true
        }
        // 可以添加CIDR範圍檢查
    }
    return false
}

func (aks *APIKeyService) updateLastUsed(keyID uint) {
    aks.db.Model(&models.APIKey{}).Where("id = ?", keyID).Update("last_used_at", time.Now())
}

func (aks *APIKeyService) logAPIKeyEvent(keyID uint, event string, userID uint, metadata string) {
    log := &models.APIKeyLog{
        APIKeyID:  keyID,
        Event:     event,
        UserID:    userID,
        Metadata:  metadata,
        Timestamp: time.Now(),
    }
    
    go func() {
        if err := aks.db.Create(log).Error; err != nil {
            // 記錄錯誤但不影響主要流程
            fmt.Printf("Failed to log API key event: %v\n", err)
        }
    }()
}

// API Key檢查中間件
func (aks *APIKeyService) APIKeyMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        apiKey := c.GetHeader("X-API-Key")
        if apiKey == "" {
            c.JSON(http.StatusUnauthorized, gin.H{"error": "API Key required"})
            c.Abort()
            return
        }
        
        keyData, err := aks.ValidateAPIKey(apiKey, c.ClientIP())
        if err != nil {
            c.JSON(http.StatusUnauthorized, gin.H{"error": err.Error()})
            c.Abort()
            return
        }
        
        // 檢查權限
        if !aks.hasPermission(keyData, c.Request.Method, c.Request.URL.Path) {
            c.JSON(http.StatusForbidden, gin.H{"error": "Insufficient permissions"})
            c.Abort()
            return
        }
        
        // 將API Key資訊存儲到context
        c.Set("api_key", keyData)
        c.Next()
    }
}

func (aks *APIKeyService) hasPermission(apiKey *models.APIKey, method, path string) bool {
    // 實作權限檢查邏輯
    for _, permission := range apiKey.Permissions {
        if aks.matchesPermission(permission, method, path) {
            return true
        }
    }
    return false
}

func (aks *APIKeyService) matchesPermission(permission, method, path string) bool {
    // 實作權限匹配邏輯
    // 例如: "GET:/api/v1/products" 或 "POST:/api/v1/*"
    return true // 簡化實作
}

type APILogEntry struct {

    Path            string    `json:"path"`
    StatusCode      int       `json:"status_code"`
    ResponseTime    int64     `json:"response_time_ms"`

    UserID          uint      `json:"user_id,omitempty"`

func NewAPILogger(logService *services.LogService, config *APILoggerConfig) *APILogger {
    if config == nil {
        config = &APILoggerConfig{
            SkipPaths:        []string{"/health", "/metrics"},
            LogRequestBody:   true,
            LogResponseBody:  false,
            MaxBodySize:      1024 * 1024, // 1MB
            SensitiveHeaders: []string{"Authorization", "X-API-Key", "Cookie"},
        }
    }
    
    return &APILogger{
        logService: logService,
        config:     config,
    }
}

func (al *APILogger) Middleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        // 檢查是否要跳過記錄
        if al.shouldSkip(c.Request.URL.Path) {
            c.Next()
            return
        }
        
        startTime := time.Now()
        
        // 讀取請求體
        var requestBody []byte
        if al.config.LogRequestBody && c.Request.Body != nil {
            requestBody, _ = io.ReadAll(io.LimitReader(c.Request.Body, al.config.MaxBodySize))
            c.Request.Body = io.NopCloser(bytes.NewBuffer(requestBody))
        }
        
        // 包裝ResponseWriter以捕獲回應
        responseWriter := &responseWriterWrapper{
            ResponseWriter: c.Writer,
            body:          bytes.NewBuffer([]byte{}),
        }
        c.Writer = responseWriter
        
        // 處理請求
        c.Next()
        
        // 計算回應時間
        responseTime := time.Since(startTime)
        
        // 建立日誌記錄
        logEntry := &APILogEntry{
            Method:       c.Request.Method,
            Path:         c.Request.URL.Path,
            StatusCode:   c.Writer.Status(),
            ResponseTime: responseTime.Milliseconds(),
            RequestSize:  c.Request.ContentLength,
            ResponseSize: int64(responseWriter.body.Len()),
            ClientIP:     c.ClientIP(),
            UserAgent:    c.Request.UserAgent(),
            Timestamp:    startTime,
        }
        
        // 添加API Key資訊
        if apiKey, exists := c.Get("api_key"); exists {
            if key, ok := apiKey.(*models.APIKey); ok {
                logEntry.APIKeyID = key.ID
                logEntry.UserID = key.UserID
            }
        }
        
        // 添加用戶資訊
        if userID, exists := c.Get("user_id"); exists {
            if uid, ok := userID.(uint); ok {
                logEntry.UserID = uid
            }
        }
        
        // 添加標頭資訊
        logEntry.RequestHeaders = al.sanitizeHeaders(c.Request.Header)
        logEntry.ResponseHeaders = al.sanitizeHeaders(c.Writer.Header())
        
        // 添加請求體
        if al.config.LogRequestBody && len(requestBody) > 0 {
            logEntry.RequestBody = string(requestBody)
        }
        
        // 添加回應體
        if al.config.LogResponseBody {
            logEntry.ResponseBody = responseWriter.body.String()
        }
        
        // 添加錯誤資訊
        if len(c.Errors) > 0 {
            logEntry.ErrorMessage = c.Errors.String()
        }
        
        // 非同步記錄
        go al.logAPICall(logEntry)
    }
}

func (al *APILogger) shouldSkip(path string) bool {
    for _, skipPath := range al.config.SkipPaths {
        if path == skipPath {
            return true
        }
    }
    return false
}

func (al *APILogger) sanitizeHeaders(headers map[string][]string) string {
    sanitized := make(map[string][]string)
    
    for key, values := range headers {
        isSensitive := false
        for _, sensitive := range al.config.SensitiveHeaders {
            if key == sensitive {
                isSensitive = true
                break
            }
        }
        
        if isSensitive {
            sanitized[key] = []string{"[REDACTED]"}
        } else {
            sanitized[key] = values
        }
    }
    
    data, _ := json.Marshal(sanitized)
    return string(data)
}

func (al *APILogger) logAPICall(entry *APILogEntry) {

    }

func (ams *APIMonitoringService) GetAPIMetrics(timeRange string) (*APIMetrics, error) {
    var startTime time.Time
    
    switch timeRange {
    case "1h":
        startTime = time.Now().Add(-time.Hour)
    case "24h":
        startTime = time.Now().Add(-24 * time.Hour)
    case "7d":
        startTime = time.Now().Add(-7 * 24 * time.Hour)
    case "30d":
        startTime = time.Now().Add(-30 * 24 * time.Hour)
    default:
        startTime = time.Now().Add(-24 * time.Hour)
    }
    
    metrics := &APIMetrics{
        TimeRange: timeRange,
        StartTime: startTime,
        EndTime:   time.Now(),
    }
    
    // 總請求數
    ams.db.Model(&models.APILog{}).
        Where("timestamp >= ?", startTime).
        Count(&metrics.TotalRequests)
    
    // 成功請求數
    ams.db.Model(&models.APILog{}).
        Where("timestamp >= ? AND status_code < 400", startTime).
        Count(&metrics.SuccessfulRequests)
    
    // 平均回應時間
    var avgResponseTime float64
    ams.db.Model(&models.APILog{}).
        Select("AVG(response_time_ms)").
        Where("timestamp >= ?", startTime).
        Scan(&avgResponseTime)
    metrics.AverageResponseTime = avgResponseTime
    
    // 錯誤率
    if metrics.TotalRequests > 0 {
        metrics.ErrorRate = float64(metrics.TotalRequests-metrics.SuccessfulRequests) / float64(metrics.TotalRequests) * 100
    }
    
    // 熱門端點
    ams.getTopEndpoints(startTime, &metrics.TopEndpoints)
    
    // 狀態碼分布
    ams.getStatusCodeDistribution(startTime, &metrics.StatusCodeDistribution)
    
    // 回應時間分布
    ams.getResponseTimeDistribution(startTime, &metrics.ResponseTimeDistribution)
    
    return metrics, nil
}

type APIMetrics struct {
    TimeRange                string                       `json:"time_range"`
    StartTime                time.Time                    `json:"start_time"`
    EndTime                  time.Time                    `json:"end_time"`
    TotalRequests            int64                        `json:"total_requests"`
    SuccessfulRequests       int64                        `json:"successful_requests"`
    AverageResponseTime      float64                      `json:"average_response_time_ms"`
    ErrorRate                float64                      `json:"error_rate_percent"`
    TopEndpoints             []EndpointMetric             `json:"top_endpoints"`
    StatusCodeDistribution   []StatusCodeMetric           `json:"status_code_distribution"`
    ResponseTimeDistribution []ResponseTimeDistribution   `json:"response_time_distribution"`
}

type EndpointMetric struct {
    Method          string  `json:"method"`
    Path            string  `json:"path"`
    RequestCount    int64   `json:"request_count"`
    AverageTime     float64 `json:"average_response_time_ms"`
    ErrorRate       float64 `json:"error_rate_percent"`
}

type StatusCodeMetric struct {
    StatusCode int   `json:"status_code"`
    Count      int64 `json:"count"`
}

type ResponseTimeDistribution struct {
    Range string `json:"range"`
    Count int64  `json:"count"`
}

func (ams *APIMonitoringService) getTopEndpoints(startTime time.Time, topEndpoints *[]EndpointMetric) {
    rows, err := ams.db.Raw(`
        SELECT 
            method,
            path,
            COUNT(*) as request_count,
            AVG(response_time_ms) as average_time,
            SUM(CASE WHEN status_code >= 400 THEN 1 ELSE 0 END) * 100.0 / COUNT(*) as error_rate
        FROM api_logs 
        WHERE timestamp >= ? 
        GROUP BY method, path 
        ORDER BY request_count DESC 
        LIMIT 10
    `, startTime).Rows()
    
    if err != nil {
        return
    }
    defer rows.Close()
    
    for rows.Next() {
        var metric EndpointMetric
        rows.Scan(&metric.Method, &metric.Path, &metric.RequestCount, &metric.AverageTime, &metric.ErrorRate)
        *topEndpoints = append(*topEndpoints, metric)
    }
}

func (ams *APIMonitoringService) getStatusCodeDistribution(startTime time.Time, distribution *[]StatusCodeMetric) {
    rows, err := ams.db.Raw(`
        SELECT status_code, COUNT(*) as count
        FROM api_logs 
        WHERE timestamp >= ? 
        GROUP BY status_code 
        ORDER BY count DESC
    `, startTime).Rows()
    
    if err != nil {
        return
    }
    defer rows.Close()
    
    for rows.Next() {
        var metric StatusCodeMetric
        rows.Scan(&metric.StatusCode, &metric.Count)
        *distribution = append(*distribution, metric)
    }
}

func (ams *APIMonitoringService) getResponseTimeDistribution(startTime time.Time, distribution *[]ResponseTimeDistribution) {
    ranges := []struct {
        Label string
        Min   int
        Max   int
    }{
        {"0-100ms", 0, 100},
        {"100-500ms", 100, 500},
        {"500ms-1s", 500, 1000},
        {"1s-5s", 1000, 5000},
        {"5s+", 5000, 999999},
    }
    
    for _, r := range ranges {
        var count int64
        ams.db.Model(&models.APILog{}).
            Where("timestamp >= ? AND response_time_ms >= ? AND response_time_ms < ?", startTime, r.Min, r.Max).
            Count(&count)
        
        *distribution = append(*distribution, ResponseTimeDistribution{
            Range: r.Label,
            Count: count,
        })
    }
}

    Version     string    `json:"version"`
    Status      string    `json:"status"` // active, deprecated, sunset

func NewAPIVersioning(versionService *services.APIVersionService) *APIVersioning {
    return &APIVersioning{

func (av *APIVersioning) Middleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        version := av.extractVersion(c)
        
        // 驗證版本
        apiVersion, valid := av.validateVersion(version)
        if !valid {
            c.JSON(http.StatusBadRequest, gin.H{
                "error": "Invalid API version",
                "supported_versions": av.getSupportedVersions(),
            })
            c.Abort()
            return
        }
        
        // 檢查版本狀態
        if err := av.checkVersionStatus(apiVersion); err != nil {
            c.JSON(http.StatusGone, gin.H{
                "error": err.Error(),
                "supported_versions": av.getSupportedVersions(),
            })
            c.Abort()
            return
        }
        
        // 設定版本資訊
        c.Set("api_version", version)
        c.Set("api_version_info", apiVersion)
        
        // 添加版本標頭
        c.Header("API-Version", version)
        c.Header("API-Supported-Versions", strings.Join(av.getSupportedVersions(), ", "))
        
        // 添加棄用警告
        if apiVersion.Status == "deprecated" {
            c.Header("Warning", fmt.Sprintf("299 - \"API version %s is deprecated. Please migrate to a newer version.\"", version))
        }
        
        // 記錄版本使用
        go av.logVersionUsage(c, version)
        
        c.Next()
    }
}

func (av *APIVersioning) extractVersion(c *gin.Context) string {
    // 1. 檢查URL路徑 (/api/v2/products)
    if version := av.extractVersionFromPath(c.Request.URL.Path); version != "" {
        return version
    }
    
    // 2. 檢查Accept標頭 (application/vnd.nexuserp.v2+json)
    if version := av.extractVersionFromAcceptHeader(c.GetHeader("Accept")); version != "" {
        return version
    }
    
    // 3. 檢查自定義標頭 (API-Version: v2)
    if version := c.GetHeader("API-Version"); version != "" {
        return version
    }
    
    // 4. 檢查查詢參數 (?version=v2)
    if version := c.Query("version"); version != "" {
        return version
    }
    
    // 使用預設版本
    return av.defaultVersion
}

func (av *APIVersioning) extractVersionFromPath(path string) string {
    re := regexp.MustCompile(`^/api/(v\d+)/`)
    matches := re.FindStringSubmatch(path)
    if len(matches) > 1 {
        return matches[1]
    }
    return ""
}

func (av *APIVersioning) extractVersionFromAcceptHeader(accept string) string {
    re := regexp.MustCompile(`application/vnd\.nexuserp\.(v\d+)\+json`)
    matches := re.FindStringSubmatch(accept)
    if len(matches) > 1 {
        return matches[1]
    }
    return ""
}

func (av *APIVersioning) validateVersion(version string) (*APIVersion, bool) {
    apiVersion, exists := av.supportedVersions[version]
    return apiVersion, exists
}

func (av *APIVersioning) checkVersionStatus(apiVersion *APIVersion) error {
    now := time.Now()
    
    switch apiVersion.Status {
    case "sunset":
        return fmt.Errorf("API version %s has been sunset and is no longer available", apiVersion.Version)
    case "deprecated":
        if !apiVersion.SunsetDate.IsZero() && now.After(apiVersion.SunsetDate) {
            return fmt.Errorf("API version %s has reached its sunset date and is no longer available", apiVersion.Version)
        }
    }
    
    return nil
}

func (av *APIVersioning) getSupportedVersions() []string {
    var versions []string
    for version, info := range av.supportedVersions {
        if info.Status != "sunset" {
            versions = append(versions, version)
        }
    }
    return versions
}

func (av *APIVersioning) logVersionUsage(c *gin.Context, version string) {
    usage := &models.APIVersionUsage{
        Version:   version,
        Method:    c.Request.Method,
        Path:      c.Request.URL.Path,
        ClientIP:  c.ClientIP(),
        UserAgent: c.Request.UserAgent(),
        Timestamp: time.Now(),
    }
    
    if apiKey, exists := c.Get("api_key"); exists {
        if key, ok := apiKey.(*models.APIKey); ok {
            usage.APIKeyID = &key.ID
        }
    }
    
    if err := av.versionService.LogVersionUsage(usage); err != nil {
        fmt.Printf("Failed to log version usage: %v\n", err)
    }
}

type APIVersionMigrator struct {

func NewAPIVersionMigrator() *APIVersionMigrator {

func (avm *APIVersionMigrator) RegisterMigration(from, to string, migrationFunc func(interface{}) interface{}) {
    key := fmt.Sprintf("%s->%s", from, to)
    avm.migrations[key] = migrationFunc
}

func (avm *APIVersionMigrator) Migrate(data interface{}, fromVersion, toVersion string) interface{} {
    key := fmt.Sprintf("%s->%s", fromVersion, toVersion)
    if migrationFunc, exists := avm.migrations[key]; exists {
        return migrationFunc(data)
    }
    return data
}

// 自動版本遷移中間件
func (avm *APIVersionMigrator) AutoMigrateMiddleware(targetVersion string) gin.HandlerFunc {
    return func(c *gin.Context) {
        requestVersion := c.GetString("api_version")

                    c.Set("migrated_request", migratedData)
                }
            }
        }
        
        c.Next()
        
        // 處理回應數據遷移
        if requestVersion != targetVersion {
            // 這裡需要攔截回應並進行遷移
            // 實作會比較複雜，需要包裝ResponseWriter
        }
    }
