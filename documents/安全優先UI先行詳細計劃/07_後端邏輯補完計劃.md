<?php
// 檔案: app/Services/QuoteService.php

namespace App\Services;

use App\Models\{Quote, QuoteItem, Product, Customer};
use App\Events\{QuoteCreated, QuoteStatusChanged};
use App\Exceptions\{InsufficientStockException, InvalidQuoteException};
use Illuminate\Support\Facades\{DB, Log};
use Carbon\Carbon;

class QuoteService
{
    private PricingService $pricingService;
    private InventoryService $inventoryService;
    private NotificationService $notificationService;
    
    public function __construct(
        PricingService $pricingService,
        InventoryService $inventoryService,
        NotificationService $notificationService
    ) {
        $this->pricingService = $pricingService;
        $this->inventoryService = $inventoryService;
        $this->notificationService = $notificationService;
    }
    
    /**
     * 建立新報價
     */
    public function createQuote(array $quoteData, array $items): Quote
    {
        DB::beginTransaction();
        
        try {
            // 1. 驗證客戶信用額度
            $this->validateCustomerCreditLimit($quoteData['customer_id'], $quoteData['estimated_total'] ?? 0);
            
            // 2. 驗證商品庫存可用性
            $this->validateProductAvailability($items);
            
            // 3. 建立報價主檔
            $quote = Quote::create([
                'quote_number' => $this->generateQuoteNumber(),
                'customer_id' => $quoteData['customer_id'],
                'title' => $quoteData['title'],
                'quote_date' => $quoteData['quote_date'] ?? now(),
                'valid_until' => $quoteData['valid_until'],
                'currency' => $quoteData['currency'] ?? 'TWD',
                'status' => 'draft',
                'created_by' => auth()->id(),
                'company_id' => auth()->user()->company_id,
                
                // 條款資訊
                'payment_terms' => $quoteData['payment_terms'] ?? 'NET_30',
                'delivery_terms' => $quoteData['delivery_terms'],
                'warranty_terms' => $quoteData['warranty_terms'],
                'terms_conditions' => $quoteData['terms_conditions'],
                'notes' => $quoteData['notes'],
            ]);
            
            // 4. 建立報價明細
            $this->createQuoteItems($quote, $items);
            
            // 5. 計算報價金額
            $this->calculateQuoteTotals($quote, $quoteData);
            
            // 6. 預留庫存 (軟預留)
            $this->reserveInventoryForQuote($quote);
            
            // 7. 記錄操作日誌
            $this->logQuoteActivity($quote, 'created', '建立報價');
            
            DB::commit();
            
            // 8. 觸發事件
            event(new QuoteCreated($quote));
            
            // 9. 發送通知
            $this->notificationService->notifyQuoteCreated($quote);
            
            return $quote->load(['customer', 'items.product', 'createdBy']);
            
        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('報價建立失敗', [
                'error' => $e->getMessage(),
                'quote_data' => $quoteData,
                'items' => $items,
                'user_id' => auth()->id(),
            ]);
            throw $e;
        }
    }
    
    /**
     * 更新報價狀態
     */
    public function updateQuoteStatus(Quote $quote, string $newStatus, string $reason = null): Quote
    {
        $oldStatus = $quote->status;
        
        // 驗證狀態轉換合法性
        $this->validateStatusTransition($oldStatus, $newStatus);
        
        DB::beginTransaction();
        
        try {
            $quote->update([
                'status' => $newStatus,
                'status_updated_at' => now(),
                'status_updated_by' => auth()->id(),
                'status_reason' => $reason,
            ]);
            
            // 狀態特殊處理
            switch ($newStatus) {
                case 'sent':
                    $this->handleQuoteSent($quote);
                    break;
                    
                case 'accepted':
                    $this->handleQuoteAccepted($quote);
                    break;
                    
                case 'rejected':
                    $this->handleQuoteRejected($quote);
                    break;
                    
                case 'expired':
                    $this->handleQuoteExpired($quote);
                    break;
                    
                case 'cancelled':
                    $this->handleQuoteCancelled($quote);
                    break;
            }
            
            // 記錄狀態變更日誌
            $this->logQuoteActivity($quote, 'status_changed', "狀態從 {$oldStatus} 變更為 {$newStatus}", [
                'old_status' => $oldStatus,
                'new_status' => $newStatus,
                'reason' => $reason,
            ]);
            
            DB::commit();
            
            // 觸發狀態變更事件
            event(new QuoteStatusChanged($quote, $oldStatus, $newStatus));
            
            return $quote;
            
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }
    }
    
    /**
     * 報價轉訂單
     */
    public function convertToOrder(Quote $quote, array $orderData = []): Order
    {
        if ($quote->status !== 'accepted') {
            throw new InvalidQuoteException('只有已接受的報價才能轉換為訂單');
        }
        
        DB::beginTransaction();
        
        try {
            // 1. 建立訂單
            $order = Order::create([
                'order_number' => $this->generateOrderNumber(),
                'quote_id' => $quote->id,
                'customer_id' => $quote->customer_id,
                'title' => $quote->title,
                'order_date' => now(),
                'expected_delivery_date' => $orderData['expected_delivery_date'] ?? now()->addDays(7),
                'currency' => $quote->currency,
                'status' => 'pending',
                'created_by' => auth()->id(),
                'company_id' => $quote->company_id,
                
                // 承繼報價條款
                'payment_terms' => $quote->payment_terms,
                'delivery_terms' => $quote->delivery_terms,
                'warranty_terms' => $quote->warranty_terms,
                'terms_conditions' => $quote->terms_conditions,
                
                // 金額資訊
                'subtotal' => $quote->subtotal,
                'discount_amount' => $quote->discount_amount,
                'tax_amount' => $quote->tax_amount,
                'shipping_cost' => $quote->shipping_cost,
                'total_amount' => $quote->total_amount,
            ]);
            
            // 2. 建立訂單明細
            foreach ($quote->items as $quoteItem) {
                OrderItem::create([
                    'order_id' => $order->id,
                    'product_id' => $quoteItem->product_id,
                    'sku' => $quoteItem->sku,
                    'name' => $quoteItem->name,
                    'description' => $quoteItem->description,
                    'quantity' => $quoteItem->quantity,
                    'unit_price' => $quoteItem->unit_price,
                    'total_price' => $quoteItem->total_price,
                    'unit' => $quoteItem->unit,
                ]);
            }
            
            // 3. 確認庫存預留
            $this->confirmInventoryReservation($quote, $order);
            
            // 4. 更新報價狀態
            $quote->update([
                'status' => 'converted',
                'converted_to_order_at' => now(),
                'converted_by' => auth()->id(),
            ]);
            
            // 5. 記錄轉換日誌
            $this->logQuoteActivity($quote, 'converted_to_order', '報價已轉換為訂單', [
                'order_id' => $order->id,
                'order_number' => $order->order_number,
            ]);
            
            DB::commit();
            
            // 發送通知
            $this->notificationService->notifyOrderCreated($order);
            
            return $order;
            
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }
    }
    
    /**
     * 智慧定價建議
     */
    public function calculateSmartPricing(Quote $quote): array
    {
        $recommendations = [];
        
        foreach ($quote->items as $item) {
            $product = $item->product;
            
            // 1. 成本加成定價
            $costPlusPrice = $product->cost_price * (1 + config('pricing.default_margin', 0.3));
            
            // 2. 競爭對手定價 (模擬)
            $competitorPrice = $this->getCompetitorPrice($product);
            
            // 3. 歷史成交價格
            $historicalPrice = $this->getHistoricalAveragePrice($product, $quote->customer_id);
            
            // 4. 數量折扣
            $volumeDiscount = $this->calculateVolumeDiscount($item->quantity, $product);
            
            // 5. 客戶等級定價
            $customerTierPricing = $this->getCustomerTierPricing($quote->customer, $product);
            
            $recommendations[] = [
                'product_id' => $product->id,
                'current_price' => $item->unit_price,
                'cost_plus_price' => $costPlusPrice,
                'competitor_price' => $competitorPrice,
                'historical_price' => $historicalPrice,
                'volume_discount' => $volumeDiscount,
                'customer_tier_price' => $customerTierPricing,
                'recommended_price' => $this->calculateOptimalPrice([
                    'cost_plus' => $costPlusPrice,
                    'competitor' => $competitorPrice,
                    'historical' => $historicalPrice,
                    'customer_tier' => $customerTierPricing,
                ]),
                'margin_analysis' => [
                    'current_margin' => ($item->unit_price - $product->cost_price) / $item->unit_price,
                    'recommended_margin' => 0.25, // 建議毛利率
                ],
            ];
        }
        
        return $recommendations;
    }
    
    /**
     * 批量報價處理
     */
    public function processBulkQuotes(array $quoteIds, string $action, array $params = []): array
    {
        $results = [];
        $errors = [];
        
        foreach ($quoteIds as $quoteId) {
            try {
                $quote = Quote::findOrFail($quoteId);
                
                switch ($action) {
                    case 'send':
                        $result = $this->sendQuote($quote);
                        break;
                        
                    case 'expire':
                        $result = $this->updateQuoteStatus($quote, 'expired', $params['reason'] ?? '批量過期');
                        break;
                        
                    case 'cancel':
                        $result = $this->updateQuoteStatus($quote, 'cancelled', $params['reason'] ?? '批量取消');
                        break;
                        
                    case 'update_validity':
                        $result = $this->updateQuoteValidity($quote, $params['valid_until']);
                        break;
                        
                    default:
                        throw new \InvalidArgumentException("不支援的操作: {$action}");
                }
                
                $results[] = [
                    'quote_id' => $quoteId,
                    'success' => true,
                    'result' => $result,
                ];
                
            } catch (\Exception $e) {
                $errors[] = [
                    'quote_id' => $quoteId,
                    'error' => $e->getMessage(),
                ];
            }
        }
        
        return [
            'success_count' => count($results),
            'error_count' => count($errors),
            'results' => $results,
            'errors' => $errors,
        ];
    }
    
    // 私有輔助方法
    
    private function generateQuoteNumber(): string
    {
        $prefix = 'Q' . date('Ym');
        $sequence = Quote::whereYear('created_at', date('Y'))
                        ->whereMonth('created_at', date('m'))
                        ->count() + 1;
        
        return $prefix . str_pad($sequence, 4, '0', STR_PAD_LEFT);
    }
    
    private function validateCustomerCreditLimit(int $customerId, float $amount): void
    {
        $customer = Customer::findOrFail($customerId);
        
        if ($customer->credit_limit > 0) {
            $currentDebt = $customer->getCurrentDebt();
            
            if (($currentDebt + $amount) > $customer->credit_limit) {
                throw new \RuntimeException('超過客戶信用額度限制');
            }
        }
    }
    
    private function validateProductAvailability(array $items): void
    {
        foreach ($items as $item) {
            $product = Product::findOrFail($item['product_id']);
            
            if ($product->stock_quantity < $item['quantity']) {
                throw new InsufficientStockException(
                    "商品 {$product->name} 庫存不足，庫存: {$product->stock_quantity}，需求: {$item['quantity']}"
                );
            }
        }
    }
    
    private function createQuoteItems(Quote $quote, array $items): void
    {
        foreach ($items as $itemData) {
            $product = Product::findOrFail($itemData['product_id']);
            
            QuoteItem::create([
                'quote_id' => $quote->id,
                'product_id' => $product->id,
                'sku' => $product->sku,
                'name' => $product->name,
                'description' => $itemData['description'] ?? $product->description,
                'quantity' => $itemData['quantity'],
                'unit_price' => $itemData['unit_price'] ?? $product->unit_price,
                'total_price' => $itemData['quantity'] * ($itemData['unit_price'] ?? $product->unit_price),
                'unit' => $product->unit->code ?? 'PCS',
                'notes' => $itemData['notes'] ?? null,
            ]);
        }
    }
    
    private function calculateQuoteTotals(Quote $quote, array $quoteData): void
    {
        $subtotal = $quote->items->sum('total_price');
        
        // 計算折扣
        $discountAmount = 0;
        if (isset($quoteData['discount_type']) && $quoteData['discount_value'] > 0) {
            if ($quoteData['discount_type'] === 'percentage') {
                $discountAmount = $subtotal * ($quoteData['discount_value'] / 100);
            } else {
                $discountAmount = $quoteData['discount_value'];
            }
        }
        
        // 計算稅金
        $taxableAmount = $subtotal - $discountAmount + ($quoteData['shipping_cost'] ?? 0);
        $taxAmount = $taxableAmount * (($quoteData['tax_rate'] ?? 5) / 100);
        
        // 總計
        $totalAmount = $taxableAmount + $taxAmount;
        
        $quote->update([
            'subtotal' => $subtotal,
            'discount_type' => $quoteData['discount_type'] ?? null,
            'discount_value' => $quoteData['discount_value'] ?? 0,
            'discount_amount' => $discountAmount,
            'tax_rate' => $quoteData['tax_rate'] ?? 5,
            'tax_amount' => $taxAmount,
            'shipping_cost' => $quoteData['shipping_cost'] ?? 0,
            'total_amount' => $totalAmount,
        ]);
    }
    
    private function validateStatusTransition(string $currentStatus, string $newStatus): void
    {
        $allowedTransitions = [
            'draft' => ['sent', 'cancelled'],
            'sent' => ['accepted', 'rejected', 'expired', 'revised'],
            'revised' => ['sent', 'cancelled'],
            'accepted' => ['converted'],
            'rejected' => [],
            'expired' => ['revised'],
            'cancelled' => [],
            'converted' => [],
        ];
        
        if (!in_array($newStatus, $allowedTransitions[$currentStatus] ?? [])) {
            throw new InvalidQuoteException("無法從狀態 {$currentStatus} 轉換為 {$newStatus}");
        }
    }
    
    private function logQuoteActivity(Quote $quote, string $action, string $description, array $metadata = []): void
    {
        QuoteActivity::create([
            'quote_id' => $quote->id,
            'user_id' => auth()->id(),
            'action' => $action,
            'description' => $description,
            'metadata' => $metadata,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
        ]);
    }
}

<?php
// 檔案: app/Services/InventoryService.php

namespace App\Services;

use App\Models\{Product, InventoryTransaction, StockAlert, Supplier};

class InventoryService
{

    /**
     * 庫存調整
     */
    public function adjustStock(
        Product $product, 
        int $quantity, 
        string $type, 
        string $reason, 
        array $metadata = []
    ): InventoryTransaction {
        DB::beginTransaction();
        
        try {
            $oldQuantity = $product->stock_quantity;
            $newQuantity = $this->calculateNewStock($oldQuantity, $quantity, $type);
            
            // 驗證庫存不能為負數 (除非允許負庫存)
            if ($newQuantity < 0 && !config('inventory.allow_negative_stock', false)) {
                throw new \RuntimeException('庫存不足，無法執行此操作');
            }
            
            // 更新產品庫存
            $product->update([
                'stock_quantity' => $newQuantity,
                'last_stock_update' => now(),
                'updated_by' => auth()->id(),
            ]);
            
            // 記錄庫存異動
            $transaction = InventoryTransaction::create([
                'product_id' => $product->id,
                'transaction_type' => $type,
                'quantity' => $quantity,
                'old_quantity' => $oldQuantity,
                'new_quantity' => $newQuantity,
                'reason' => $reason,
                'reference_id' => $metadata['reference_id'] ?? null,
                'reference_type' => $metadata['reference_type'] ?? null,
                'cost_per_unit' => $metadata['cost_per_unit'] ?? $product->cost_price,
                'total_cost' => $quantity * ($metadata['cost_per_unit'] ?? $product->cost_price),
                'location' => $metadata['location'] ?? $product->location,
                'batch_number' => $metadata['batch_number'] ?? null,
                'expiry_date' => $metadata['expiry_date'] ?? null,
                'created_by' => auth()->id(),
                'company_id' => auth()->user()->company_id,
                'metadata' => $metadata,
            ]);
            
            // 更新平均成本 (移動平均法)
            if ($type === 'in') {
                $this->updateAverageCost($product, $quantity, $metadata['cost_per_unit'] ?? $product->cost_price);
            }
            
            // 檢查庫存警示
            $this->checkStockAlerts($product);
            
            // 更新庫存價值
            $this->updateInventoryValue($product);
            
            // 清除相關快取
            $this->clearProductCache($product);
            
            DB::commit();
            
            // 觸發事件
            event(new StockLevelChanged($product, $oldQuantity, $newQuantity));
            
            // 記錄審計日誌
            $this->auditService->logInventoryChange($product, $transaction);
            
            return $transaction;
            
        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('庫存調整失敗', [
                'product_id' => $product->id,
                'quantity' => $quantity,
                'type' => $type,
                'error' => $e->getMessage(),
            ]);
            throw $e;
        }
    }

    /**
     * 智慧補貨建議
     */
    public function getReplenishmentSuggestions(): array
    {
        // 獲取需要補貨的商品
        $lowStockProducts = Product::where('stock_quantity', '<=', DB::raw('min_stock_level'))
                                 ->where('status', 'active')
                                 ->with(['category', 'unit', 'suppliers'])
                                 ->get();
        
        $suggestions = [];
        
        foreach ($lowStockProducts as $product) {
            // 計算平均日用量
            $avgDailyUsage = $this->calculateAverageDailyUsage($product->id, 90); // 90天
            
            // 計算前置時間
            $leadTime = $product->suppliers->min('lead_time_days') ?? 7;
            
            // 計算安全庫存
            $safetyStock = $avgDailyUsage * $leadTime * 1.5;
            
            // 建議補貨數量
            $suggestedQuantity = max(
                $product->max_stock_level - $product->stock_quantity, // 補足到最大庫存
                $safetyStock + ($avgDailyUsage * $leadTime) - $product->stock_quantity // 安全庫存計算
            );
            
            // 選擇最佳供應商
            $bestSupplier = $this->selectBestSupplier($product);
            
            // 計算預估成本
            $estimatedCost = $suggestedQuantity * ($bestSupplier->pivot->unit_price ?? $product->cost_price);
            
            $suggestions[] = [
                'product' => $product,
                'current_stock' => $product->stock_quantity,
                'min_level' => $product->min_stock_level,
                'max_level' => $product->max_stock_level,
                'avg_daily_usage' => $avgDailyUsage,
                'lead_time_days' => $leadTime,
                'safety_stock' => $safetyStock,
                'suggested_quantity' => ceil($suggestedQuantity),
                'best_supplier' => $bestSupplier,
                'estimated_cost' => $estimatedCost,
                'urgency_score' => $this->calculateUrgencyScore($product, $avgDailyUsage),
                'last_purchase_date' => $this->getLastPurchaseDate($product->id),
                'seasonal_factor' => $this->getSeasonalFactor($product->id),
            ];
        }
        
        // 按緊急程度排序
        usort($suggestions, function ($a, $b) {
            return $b['urgency_score'] <=> $a['urgency_score'];
        });
        
        return $suggestions;
    }
    
    /**
     * 庫存盤點
     */
    public function performStockCount(array $countData): array
    {
        $results = [
            'total_items' => 0,
            'discrepancies' => 0,
            'adjustments_made' => 0,
            'total_value_difference' => 0,
            'items' => [],
        ];
        
        DB::beginTransaction();
        
        try {
            foreach ($countData as $item) {
                $product = Product::findOrFail($item['product_id']);
                $systemQuantity = $product->stock_quantity;
                $countedQuantity = $item['counted_quantity'];
                $difference = $countedQuantity - $systemQuantity;
                
                $itemResult = [
                    'product_id' => $product->id,
                    'product_name' => $product->name,
                    'sku' => $product->sku,
                    'system_quantity' => $systemQuantity,
                    'counted_quantity' => $countedQuantity,
                    'difference' => $difference,
                    'value_difference' => $difference * $product->cost_price,
                    'adjustment_made' => false,
                ];
                
                // 如果有差異且超過容忍範圍，進行調整
                $tolerance = config('inventory.count_tolerance', 0);
                if (abs($difference) > $tolerance) {
                    $adjustmentType = $difference > 0 ? 'count_increase' : 'count_decrease';
                    
                    $this->adjustStock(
                        $product,
                        abs($difference),
                        $adjustmentType,
                        '盤點調整',
                        [
                            'counted_by' => $item['counted_by'] ?? auth()->id(),
                            'count_date' => $item['count_date'] ?? now(),
                            'notes' => $item['notes'] ?? null,
                        ]
                    );
                    
                    $itemResult['adjustment_made'] = true;
                    $results['adjustments_made']++;
                    $results['total_value_difference'] += $itemResult['value_difference'];
                }
                
                if ($difference != 0) {
                    $results['discrepancies']++;
                }
                
                $results['items'][] = $itemResult;
                $results['total_items']++;
            }
            
            // 建立盤點記錄
            $stockCount = StockCount::create([
                'count_date' => now(),
                'created_by' => auth()->id(),
                'company_id' => auth()->user()->company_id,
                'total_items' => $results['total_items'],
                'discrepancies' => $results['discrepancies'],
                'adjustments_made' => $results['adjustments_made'],
                'total_value_difference' => $results['total_value_difference'],
                'status' => 'completed',
                'notes' => request('notes'),
            ]);
            
            // 記錄盤點明細
            foreach ($results['items'] as $item) {
                StockCountItem::create([
                    'stock_count_id' => $stockCount->id,
                    'product_id' => $item['product_id'],
                    'system_quantity' => $item['system_quantity'],
                    'counted_quantity' => $item['counted_quantity'],
                    'difference' => $item['difference'],
                    'value_difference' => $item['value_difference'],
                    'adjustment_made' => $item['adjustment_made'],
                ]);
            }
            
            DB::commit();
            
            // 發送盤點報告
            $this->notificationService->sendStockCountReport($stockCount, $results);
            
            return $results;
            
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }
    }
    
    /**
     * ABC庫存分析
     */
    public function performABCAnalysis(int $analysisMonths = 12): array
    {
        // 獲取指定期間的銷售數據
        $salesData = DB::table('inventory_transactions')
                      ->select('product_id', DB::raw('SUM(quantity * cost_per_unit) as total_value'))
                      ->where('transaction_type', 'out')
                      ->where('created_at', '>=', now()->subMonths($analysisMonths))
                      ->groupBy('product_id')
                      ->orderBy('total_value', 'desc')
                      ->get();
        
        $totalValue = $salesData->sum('total_value');
        $runningTotal = 0;
        $results = [];
        
        foreach ($salesData as $item) {
            $runningTotal += $item->total_value;
            $percentageValue = ($item->total_value / $totalValue) * 100;
            $cumulativePercentage = ($runningTotal / $totalValue) * 100;
            
            // ABC分類
            if ($cumulativePercentage <= 80) {
                $category = 'A';
            } elseif ($cumulativePercentage <= 95) {
                $category = 'B';
            } else {
                $category = 'C';
            }
            
            $product = Product::find($item->product_id);
            
            $results[] = [
                'product' => $product,
                'total_value' => $item->total_value,
                'percentage_value' => $percentageValue,
                'cumulative_percentage' => $cumulativePercentage,
                'abc_category' => $category,
                'current_stock_value' => $product->stock_quantity * $product->cost_price,
                'recommended_policy' => $this->getRecommendedPolicy($category),
            ];
        }
        
        // 更新產品ABC分類
        foreach ($results as $result) {
            $result['product']->update(['abc_category' => $result['abc_category']]);
        }
        
        return [
            'analysis_period' => $analysisMonths,
            'total_products' => count($results),
            'total_value' => $totalValue,
            'category_summary' => $this->summarizeABCCategories($results),
            'products' => $results,
        ];
    }
    
    // 私有輔助方法
    
    private function calculateNewStock(int $oldQuantity, int $adjustmentQuantity, string $type): int
    {
        switch ($type) {
            case 'in':
            case 'purchase':
            case 'count_increase':
                return $oldQuantity + $adjustmentQuantity;
                
            case 'out':
            case 'sale':
            case 'count_decrease':
                return $oldQuantity - $adjustmentQuantity;
                
            case 'set':
                return $adjustmentQuantity;
                
            default:
                throw new \InvalidArgumentException("不支援的庫存異動類型: {$type}");
        }
    }
    
    private function updateAverageCost(Product $product, int $incomingQuantity, float $incomingCost): void
    {
        $currentQuantity = $product->stock_quantity - $incomingQuantity; // 調整前的數量
        $currentCost = $product->cost_price;
        
        if ($currentQuantity > 0) {
            $totalCost = ($currentQuantity * $currentCost) + ($incomingQuantity * $incomingCost);
            $totalQuantity = $currentQuantity + $incomingQuantity;
            $newAverageCost = $totalCost / $totalQuantity;
        } else {
            $newAverageCost = $incomingCost;
        }
        
        $product->update(['cost_price' => $newAverageCost]);
    }
    
    private function checkStockAlerts(Product $product): void
    {
        $currentStock = $product->stock_quantity;
        
        // 檢查低庫存警示
        if ($currentStock <= $product->min_stock_level) {
            $alertType = $currentStock == 0 ? 'out_of_stock' : 'low_stock';
            
            $alert = StockAlert::firstOrCreate([
                'product_id' => $product->id,
                'alert_type' => $alertType,
                'is_resolved' => false,
            ], [
                'current_quantity' => $currentStock,
                'threshold_quantity' => $product->min_stock_level,
                'created_by' => auth()->id(),
                'company_id' => auth()->user()->company_id,
            ]);
            
            if ($alert->wasRecentlyCreated) {
                event(new LowStockAlert($product, $currentStock));
                $this->notificationService->sendLowStockAlert($product, $currentStock);
            }
        } else {
            // 解決已存在的警示
            StockAlert::where('product_id', $product->id)
                     ->where('is_resolved', false)
                     ->update([
                         'is_resolved' => true,
                         'resolved_at' => now(),
                         'resolved_by' => auth()->id(),
                     ]);
        }
    }
    
    private function calculateAverageDailyUsage(int $productId, int $days): float
    {
        $totalUsage = InventoryTransaction::where('product_id', $productId)
                                        ->where('transaction_type', 'out')
                                        ->where('created_at', '>=', now()->subDays($days))
                                        ->sum('quantity');
        
        return $totalUsage / $days;
    }
    
    private function calculateUrgencyScore(Product $product, float $avgDailyUsage): float
    {
        $daysUntilStockout = $avgDailyUsage > 0 ? $product->stock_quantity / $avgDailyUsage : 999;
        
        // 緊急程度評分 (0-100)
        if ($daysUntilStockout <= 0) return 100;      // 已缺貨
        if ($daysUntilStockout <= 3) return 90;       // 3天內缺貨
        if ($daysUntilStockout <= 7) return 70;       // 一週內缺貨
        if ($daysUntilStockout <= 14) return 50;      // 兩週內缺貨
        if ($daysUntilStockout <= 30) return 30;      // 一個月內缺貨
        
        return 10; // 超過一個月
    }
    
    private function clearProductCache(Product $product): void
    {
        Cache::tags(['product', "product.{$product->id}"])->flush();
    }
}

<?php
// 檔案: app/Services/WorkflowService.php

namespace App\Services;

use App\Models\{Workflow, WorkflowInstance, WorkflowStep, WorkflowAction};
use App\Events\WorkflowStepCompleted;

class WorkflowService
{
    /**
     * 啟動工作流程
     */
    public function startWorkflow(string $workflowCode, $entity, array $context = []): WorkflowInstance
    {
        $workflow = Workflow::where('code', $workflowCode)->where('is_active', true)->firstOrFail();
        
        $instance = WorkflowInstance::create([
            'workflow_id' => $workflow->id,
            'entity_type' => get_class($entity),
            'entity_id' => $entity->id,
            'status' => 'running',
            'context' => $context,
            'started_by' => auth()->id(),
            'company_id' => auth()->user()->company_id,
        ]);
        
        // 執行第一個步驟
        $this->executeNextStep($instance);
        
        return $instance;
    }
    
    /**
     * 執行下一個步驟
     */
    public function executeNextStep(WorkflowInstance $instance): void
    {
        $currentStep = $instance->currentStep;
        
        if (!$currentStep) {
            // 查找第一個步驟
            $nextStep = WorkflowStep::where('workflow_id', $instance->workflow_id)
                                   ->where('is_start_step', true)
                                   ->first();
        } else {
            // 查找下一個步驟
            $nextStep = $this->getNextStep($currentStep, $instance->context);
        }
        
        if ($nextStep) {
            $this->executeStep($instance, $nextStep);
        } else {
            $this->completeWorkflow($instance);
        }
    }
    
    /**
     * 自動審批規則引擎
     */
    public function processAutoApproval($entity, string $approvalType): bool
    {
        $rules = $this->getApprovalRules($approvalType);
        
        foreach ($rules as $rule) {
            if ($this->evaluateRule($rule, $entity)) {
                $this->executeAutoApproval($entity, $rule);
                return true;
            }
        }
        
        return false; // 需要人工審批
    }
}
