```php
<?php
// 檔案: tests/Unit/Services/QuoteServiceTest.php

namespace Tests\Unit\Services;

use Tests\TestCase;
use App\Services\QuoteService;
use App\Models\{Quote, Customer, Product, User, Company};
use App\Exceptions\{InsufficientStockException, InvalidQuoteException};
use Illuminate\Foundation\Testing\RefreshDatabase;
use Mockery;

class QuoteServiceTest extends TestCase
{
    use RefreshDatabase;
    
    private QuoteService $quoteService;
    private User $user;
    private Company $company;
    private Customer $customer;
    private Product $product;
    
    protected function setUp(): void
    {
        parent::setUp();
        
        // 建立測試數據
        $this->company = Company::factory()->create();
        $this->user = User::factory()->create(['company_id' => $this->company->id]);
        $this->customer = Customer::factory()->create(['company_id' => $this->company->id]);
        $this->product = Product::factory()->create([
            'company_id' => $this->company->id,
            'stock_quantity' => 100,
            'unit_price' => 1000,
        ]);
        
        // 模擬認證用戶
        $this->actingAs($this->user);
        
        // 初始化服務（使用依賴注入）
        $this->quoteService = app(QuoteService::class);
    }
    
    /** @test */
    public function it_can_create_quote_with_valid_data(): void
    {
        // 準備測試數據
        $quoteData = [
            'customer_id' => $this->customer->id,
            'title' => '測試報價單',
            'quote_date' => now()->toDateString(),
            'valid_until' => now()->addDays(30)->toDateString(),
            'currency' => 'TWD',
        ];
        
        $items = [
            [
                'product_id' => $this->product->id,
                'quantity' => 10,
                'unit_price' => 950,
            ],
        ];
        
        // 執行測試
        $quote = $this->quoteService->createQuote($quoteData, $items);
        
        // 驗證結果
        $this->assertInstanceOf(Quote::class, $quote);
        $this->assertEquals($quoteData['customer_id'], $quote->customer_id);
        $this->assertEquals($quoteData['title'], $quote->title);
        $this->assertEquals('draft', $quote->status);
        $this->assertEquals(9500, $quote->subtotal); // 10 * 950
        
        // 驗證報價項目
        $this->assertCount(1, $quote->items);
        $this->assertEquals($this->product->id, $quote->items->first()->product_id);
        $this->assertEquals(10, $quote->items->first()->quantity);
        $this->assertEquals(950, $quote->items->first()->unit_price);
    }
    
    /** @test */
    public function it_throws_exception_when_insufficient_stock(): void
    {
        // 準備測試數據 - 庫存不足
        $quoteData = [
            'customer_id' => $this->customer->id,
            'title' => '庫存不足測試',
            'quote_date' => now()->toDateString(),
            'valid_until' => now()->addDays(30)->toDateString(),
        ];
        
        $items = [
            [
                'product_id' => $this->product->id,
                'quantity' => 150, // 超過庫存數量(100)
            ],
        ];
        
        // 驗證異常拋出
        $this->expectException(InsufficientStockException::class);
        $this->expectExceptionMessage('庫存不足');
        
        $this->quoteService->createQuote($quoteData, $items);
    }
    
    /** @test */
    public function it_can_update_quote_status_with_valid_transition(): void
    {
        // 建立測試報價
        $quote = Quote::factory()->create([
            'status' => 'draft',
            'company_id' => $this->company->id,
        ]);
        
        // 執行狀態更新
        $updatedQuote = $this->quoteService->updateQuoteStatus($quote, 'sent', '發送給客戶');
        
        // 驗證結果

        $this->assertEquals('發送給客戶', $updatedQuote->status_reason);
        $this->assertNotNull($updatedQuote->status_updated_at);
        $this->assertEquals($this->user->id, $updatedQuote->status_updated_by);
    }
    
    /** @test */
    public function it_throws_exception_for_invalid_status_transition(): void
    {
        // 建立已接受的報價
        $quote = Quote::factory()->create([
            'status' => 'accepted',
            'company_id' => $this->company->id,
        ]);
        
        // 驗證無效轉換異常
        $this->expectException(InvalidQuoteException::class);
        $this->expectExceptionMessage('無法從狀態 accepted 轉換為 draft');
        
        $this->quoteService->updateQuoteStatus($quote, 'draft');
    }
    
    /** @test */
    public function it_can_calculate_quote_totals_correctly(): void
    {
        $quoteData = [
            'customer_id' => $this->customer->id,
            'title' => '計算測試',
            'quote_date' => now()->toDateString(),
            'valid_until' => now()->addDays(30)->toDateString(),
            'discount_type' => 'percentage',
            'discount_value' => 10, // 10% 折扣
            'tax_rate' => 5, // 5% 稅率
            'shipping_cost' => 200,
        ];
        
        $items = [
            [
                'product_id' => $this->product->id,
                'quantity' => 10,
                'unit_price' => 1000,
            ],
        ];
        
        $quote = $this->quoteService->createQuote($quoteData, $items);
        
        // 驗證計算結果
        $this->assertEquals(10000, $quote->subtotal); // 10 * 1000
        $this->assertEquals(1000, $quote->discount_amount); // 10000 * 0.1
        $this->assertEquals(460, $quote->tax_amount); // (10000 - 1000 + 200) * 0.05
        $this->assertEquals(200, $quote->shipping_cost);
        $this->assertEquals(9660, $quote->total_amount); // 10000 - 1000 + 200 + 460
    }
    
    /** @test */
    public function it_generates_unique_quote_numbers(): void
    {
        // 建立多個報價
        $quotes = [];
        for ($i = 0; $i < 5; $i++) {
            $quotes[] = Quote::factory()->create([
                'company_id' => $this->company->id,
            ]);
        }
        
        // 驗證報價單號唯一性
        $quoteNumbers = collect($quotes)->pluck('quote_number')->toArray();
        $this->assertEquals(count($quoteNumbers), count(array_unique($quoteNumbers)));
        
        // 驗證報價單號格式
        foreach ($quoteNumbers as $number) {
            $this->assertMatchesRegularExpression('/^Q\d{6}\d{4}$/', $number);
        }
    }
    
    /** @test */
    public function it_can_convert_quote_to_order(): void
    {
        // 建立已接受的報價
        $quote = Quote::factory()->create([
            'status' => 'accepted',
            'customer_id' => $this->customer->id,
            'company_id' => $this->company->id,
            'subtotal' => 10000,
            'total_amount' => 10500,
        ]);
        
        // 添加報價項目
        $quote->items()->create([
            'product_id' => $this->product->id,
            'quantity' => 10,
            'unit_price' => 1000,
            'total_price' => 10000,
        ]);
        
        // 執行轉換
        $order = $this->quoteService->convertToOrder($quote);
        
        // 驗證訂單建立
        $this->assertDatabaseHas('orders', [
            'quote_id' => $quote->id,
            'customer_id' => $quote->customer_id,
            'total_amount' => $quote->total_amount,
        ]);
        
        // 驗證報價狀態更新
        $quote->refresh();
        $this->assertEquals('converted', $quote->status);
        $this->assertNotNull($quote->converted_to_order_at);
    }
    
    /** @test */
    public function it_handles_concurrent_quote_creation(): void
    {
        // 模擬併發情況
        $promises = [];
        
        for ($i = 0; $i < 10; $i++) {
            $promises[] = async(function() {
                $quoteData = [
                    'customer_id' => $this->customer->id,
                    'title' => '併發測試' . rand(1000, 9999),
                    'quote_date' => now()->toDateString(),
                    'valid_until' => now()->addDays(30)->toDateString(),
                ];
                
                $items = [
                    [
                        'product_id' => $this->product->id,
                        'quantity' => 1,
                    ],
                ];
                
                return $this->quoteService->createQuote($quoteData, $items);
            });
        }
        
        $results = await($promises);
        
        // 驗證所有報價都成功建立
        $this->assertCount(10, $results);
        
        // 驗證報價單號唯一性
        $quoteNumbers = collect($results)->pluck('quote_number')->unique();
        $this->assertCount(10, $quoteNumbers);
    }
    
    protected function tearDown(): void
    {
        Mockery::close();
        parent::tearDown();
    }
}
```

```php
// 檔案: tests/Integration/Api/QuoteApiTest.php

namespace Tests\Integration\Api;

use Tests\TestCase;
use App\Models\{User, Customer, Product, Quote};
use Illuminate\Foundation\Testing\RefreshDatabase;
use Laravel\Sanctum\Sanctum;

class QuoteApiTest extends TestCase
{
    use RefreshDatabase;
    
    private User $user;
    
    protected function setUp(): void
    {
        parent::setUp();
        
        $this->user = User::factory()->create();
        $this->customer = Customer::factory()->create(['company_id' => $this->user->company_id]);

        ]);
        
        Sanctum::actingAs($this->user);

    /** @test */
    public function it_can_create_quote_through_api(): void
    {
        $payload = [
            'customer_id' => $this->customer->id,
            'title' => 'API測試報價',
            'quote_date' => now()->toDateString(),
            'valid_until' => now()->addDays(30)->toDateString(),
            'items' => [
                [
                    'product_id' => $this->product->id,
                    'quantity' => 5,
                    'unit_price' => 1200,
                ],
            ],
        ];
        
        $response = $this->postJson('/api/quotes', $payload);
        
        $response->assertStatus(201)
                ->assertJsonStructure([
                    'data' => [
                        'id',
                        'quote_number',
                        'customer_id',
                        'title',
                        'status',
                        'subtotal',
                        'total_amount',
                        'items',
                    ],
                ]);
        
        $this->assertDatabaseHas('quotes', [
            'title' => 'API測試報價',
            'customer_id' => $this->customer->id,
        ]);
    }
    
    /** @test */
    public function it_validates_quote_creation_data(): void
    {
        $payload = [
            'customer_id' => 99999, // 不存在的客戶
            'title' => '', // 空標題
            'items' => [], // 空項目
        ];
        
        $response = $this->postJson('/api/quotes', $payload);
        
        $response->assertStatus(422)
                ->assertJsonValidationErrors([
                    'customer_id',
                    'title',
                    'items',
                ]);
    }
    
    /** @test */
    public function it_can_update_quote_status(): void
    {
        $quote = Quote::factory()->create([
            'status' => 'draft',
            'company_id' => $this->user->company_id,
        ]);
        
        $response = $this->patchJson("/api/quotes/{$quote->id}/status", [
            'status' => 'sent',
            'reason' => '發送給客戶確認',
        ]);
        
        $response->assertStatus(200)
                ->assertJson([
                    'data' => [
                        'status' => 'sent',
                    ],
                ]);
        
        $this->assertDatabaseHas('quotes', [
            'id' => $quote->id,
            'status' => 'sent',
            'status_reason' => '發送給客戶確認',
        ]);
    }
    
    /** @test */
    public function it_can_paginate_quotes_list(): void
    {
        Quote::factory()->count(25)->create([
            'company_id' => $this->user->company_id,
        ]);
        
        $response = $this->getJson('/api/quotes?per_page=10&page=2');
        
        $response->assertStatus(200)
                ->assertJsonStructure([
                    'data',
                    'links',
                    'meta' => [
                        'current_page',
                        'per_page',
                        'total',
                    ],
                ])
                ->assertJson([
                    'meta' => [
                        'current_page' => 2,
                        'per_page' => 10,
                    ],
                ]);
        
        $this->assertCount(10, $response->json('data'));
    }
    
    /** @test */
    public function it_filters_quotes_by_status(): void
    {
        Quote::factory()->count(5)->create([
            'status' => 'draft',
            'company_id' => $this->user->company_id,
        ]);
        
        Quote::factory()->count(3)->create([
            'status' => 'sent',
            'company_id' => $this->user->company_id,
        ]);
        
        $response = $this->getJson('/api/quotes?status=draft');
        
        $response->assertStatus(200);
        $this->assertCount(5, $response->json('data'));
        
        foreach ($response->json('data') as $quote) {
            $this->assertEquals('draft', $quote['status']);
        }
    }
    
    /** @test */
    public function it_enforces_company_isolation(): void
    {
        // 建立其他公司的報價
        $otherCompanyQuote = Quote::factory()->create();
        
        // 嘗試存取其他公司的報價
        $response = $this->getJson("/api/quotes/{$otherCompanyQuote->id}");
        
        $response->assertStatus(404);
    }
    
    /** @test */
    public function it_handles_concurrent_status_updates(): void
    {
        $quote = Quote::factory()->create([
            'status' => 'draft',
            'company_id' => $this->user->company_id,
        ]);
        
        // 模擬併發更新
        $promises = [];
        $statuses = ['sent', 'revised', 'cancelled'];
        
        foreach ($statuses as $status) {
            $promises[] = async(function() use ($quote, $status) {
                return $this->patchJson("/api/quotes/{$quote->id}/status", [
                    'status' => $status,
                    'reason' => "更新為 {$status}",
                ]);
            });
        }
        
        $responses = await($promises);
        
        // 驗證只有一個更新成功，其他返回衝突錯誤
        $successCount = 0;
        $conflictCount = 0;
        
        foreach ($responses as $response) {
            if ($response->status() === 200) {
                $successCount++;
            } elseif ($response->status() === 409) {
                $conflictCount++;
            }
        }
        
        $this->assertEquals(1, $successCount);
        $this->assertEquals(2, $conflictCount);
    }
}
```

```php
<?php
// 檔案: app/Console/Commands/GenerateTestReport.php

namespace App\Console\Commands;

use Illuminate\Console\Command;

class GenerateTestReport extends Command
{
    protected $signature = 'test:report {--format=html : 報告格式 (html|json|xml)}';
    protected $description = '生成測試報告';
    
    public function handle(): int
    {
        $format = $this->option('format');
        
        $this->info('正在生成測試報告...');
        
        // 執行測試並收集結果
        $testResults = $this->runTestSuites();
        
        // 生成報告
        $report = $this->generateReport($testResults, $format);
        
        // 保存報告
        $reportPath = $this->saveReport($report, $format);
        
        $this->info("測試報告已生成: {$reportPath}");
        
        return 0;
    }
    
    private function runTestSuites(): array
    {
        $results = [];
        
        // 單元測試
        $this->line('執行單元測試...');
        $results['unit'] = $this->runPhpUnitTests('Unit');
        
        // 整合測試
        $this->line('執行整合測試...');
        $results['integration'] = $this->runPhpUnitTests('Integration');
        
        // E2E測試
        $this->line('執行 E2E 測試...');
        $results['e2e'] = $this->runDuskTests();
        
        // 效能測試
        $this->line('執行效能測試...');
        $results['performance'] = $this->runPerformanceTests();
        
        return $results;
    }
    
    private function generateReport(array $testResults, string $format): string
    {
        $summary = [
            'timestamp' => now()->toISOString(),
            'total_tests' => array_sum(array_column($testResults, 'total')),
            'passed_tests' => array_sum(array_column($testResults, 'passed')),
            'failed_tests' => array_sum(array_column($testResults, 'failed')),
            'coverage' => $this->calculateCoverage($testResults),
            'suites' => $testResults,
        ];
        
        $summary['success_rate'] = $summary['total_tests'] > 0 
            ? ($summary['passed_tests'] / $summary['total_tests']) * 100 
            : 0;
        
        switch ($format) {
            case 'json':
                return json_encode($summary, JSON_PRETTY_PRINT);
                
            case 'xml':
                return $this->generateXmlReport($summary);
                
            default:
                return $this->generateHtmlReport($summary);
        }
    }
    
    private function generateHtmlReport(array $summary): string
    {
        return view('admin.test-report', compact('summary'))->render();
    }
}
```
