```php
function initializeSalesChart() {
    const ctx = document.getElementById('salesChart').getContext('2d');
    
    new Chart(ctx, {
        type: 'line',
        data: {
            labels: @json($chartData['labels']),
            datasets: [{

// 自動刷新設定
function setupDashboardAutoRefresh() {
    // 每5分鐘刷新統計數據
    setInterval(function() {
        fetch('/api/dashboard/stats')
            .then(response => response.json())
            .then(data => updateDashboardStats(data))
            .catch(error => console.error('Stats refresh failed:', error));
    }, 300000); // 5分鐘
}

// 更新統計數據
function updateDashboardStats(stats) {
    Object.keys(stats).forEach(key => {
        const element = document.querySelector(`[data-stat="${key}"]`);
        if (element) {
            element.textContent = stats[key];
            element.classList.add('animate-pulse');
            setTimeout(() => element.classList.remove('animate-pulse'), 1000);
        }
    });
}
</script>
@endpush
```

#### **Controller實作**
```php
<?php
// 檔案: app/Http/Controllers/DashboardController.php

namespace App\Http\Controllers;

use App\Services\DashboardService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;

class DashboardController extends Controller
{
    private DashboardService $dashboardService;
    
    public function __construct(DashboardService $dashboardService)
    {
        $this->dashboardService = $dashboardService;
    }
    
    public function index()
    {
        // 使用快取提升效能 (5分鐘)
        $cacheKey = 'dashboard_data_' . auth()->user()->company_id;
        
        $dashboardData = Cache::remember($cacheKey, 300, function () {
            return [
                'stats' => $this->dashboardService->getStatistics(),
                'chartData' => $this->dashboardService->getSalesChartData(),
                'recentActivities' => $this->dashboardService->getRecentActivities(10),
            ];
        });
        
        return view('dashboard.index', $dashboardData);
    }
    
    // API端點：即時統計數據
    public function statsApi()
    {
        $stats = $this->dashboardService->getStatistics();
        
        return response()->json($stats);
    }
}
```

```php
<script>
let deleteProductId = null;

// 確認刪除
function confirmDelete(productId, productName) {
    deleteProductId = productId;
    document.getElementById('deleteProductName').textContent = productName;
    openModal('delete-product-modal');
}

// 執行刪除
function executeDelete() {
    if (!deleteProductId) return;
    
    fetch(`/inventory/${deleteProductId}`, {
        method: 'DELETE',
        headers: {
            'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content,
            'Accept': 'application/json',
        }
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            location.reload();
        } else {
            alert('刪除失敗：' + data.message);
        }
    })
    .catch(error => {
        console.error('Delete error:', error);
        alert('刪除時發生錯誤');
    })
    .finally(() => {
        closeModal('delete-product-modal');
        deleteProductId = null;
    });
}

// 排序功能
document.querySelectorAll('.sortable').forEach(header => {
    header.addEventListener('click', function() {
        const sortField = this.dataset.sort;
        const url = new URL(window.location);
        const currentSort = url.searchParams.get('sort');
        const currentDirection = url.searchParams.get('direction') || 'asc';
        
        // 切換排序方向
        const newDirection = (currentSort === sortField && currentDirection === 'asc') ? 'desc' : 'asc';
        
        url.searchParams.set('sort', sortField);
        url.searchParams.set('direction', newDirection);
        
        window.location.href = url.toString();
    });
});

// 庫存狀態即時更新
function updateStockStatus() {
    document.querySelectorAll('.stock-quantity').forEach(element => {
        const stock = parseInt(element.dataset.stock);
        const row = element.closest('tr');
        
        // 根據庫存數量調整樣式
        if (stock === 0) {
            row.classList.add('out-of-stock');
        } else if (stock < 10) {
            row.classList.add('low-stock');
        }
    });
}
```

<script>
function quoteForm() {
    return {
        currentStep: 1,
        isSubmitting: false,
        form: {
            customer_id: '',
            title: '',
            quote_date: '{{ date("Y-m-d") }}',
            valid_until: '{{ date("Y-m-d", strtotime("+30 days")) }}',
            currency: 'TWD',
            items: [],
            discount_type: 'none',
            discount_value: 0,
            tax_rate: 5,
            shipping_cost: 0,
            payment_terms: 'net_30',
            delivery_terms: '',
            warranty_terms: '',
            notes: '',
            terms_conditions: ''
        },
        selectedCustomer: null,
        productSearch: '',
        productCategoryFilter: '',
        allProducts: @json($products),
        productCategories: @json($categories),
        filteredProducts: [],
        calculations: {
            subtotal: 0,
            discount: 0,
            taxable_amount: 0,
            tax_amount: 0,
            total: 0
        },
        
        init() {
            this.filteredProducts = this.allProducts.slice(0, 12); // 初始顯示12個商品
            this.calculateTotals();
        },
        
        nextStep() {
            if (this.validateCurrentStep()) {
                this.currentStep++;
            }
        },
        
        previousStep() {
            this.currentStep--;
        },
        
        validateCurrentStep() {
            switch (this.currentStep) {
                case 1:
                    return this.form.customer_id && this.form.title;
                case 2:
                    return this.form.items.length > 0;
                case 3:
                    return true; // 定價設定都是可選的
                case 4:
                    return true;
                default:
                    return true;
            }
        },
        
        loadCustomerData() {
            if (this.form.customer_id) {
                // 從 API 載入客戶詳細資料
                fetch(`/api/customers/${this.form.customer_id}`)
                    .then(response => response.json())
                    .then(data => {
                        this.selectedCustomer = data;
                    });
            }
        },
        
        searchProducts() {
            let filtered = this.allProducts;
            
            // 文字搜尋
            if (this.productSearch) {
                const search = this.productSearch.toLowerCase();
                filtered = filtered.filter(product => 
                    product.name.toLowerCase().includes(search) ||
                    product.sku.toLowerCase().includes(search) ||
                    (product.description && product.description.toLowerCase().includes(search))
                );
            }
            
            // 分類篩選
            if (this.productCategoryFilter) {
                filtered = filtered.filter(product => 
                    product.category_id == this.productCategoryFilter
                );
            }
            
            this.filteredProducts = filtered.slice(0, 20); // 限制顯示數量
        },
        
        addProduct(product) {
            if (!this.isProductAdded(product.id)) {
                this.form.items.push({
                    product_id: product.id,
                    name: product.name,
                    sku: product.sku,
                    image_url: product.image_url,
                    quantity: 1,
                    unit_price: product.unit_price,
                    unit: product.unit
                });
                this.calculateTotals();
            }
        },
        
        removeItem(index) {
            this.form.items.splice(index, 1);
            this.calculateTotals();
        },
        
        isProductAdded(productId) {
            return this.form.items.some(item => item.product_id === productId);
        },
        
        updateItemQuantity(index, quantity) {
            this.form.items[index].quantity = parseFloat(quantity) || 1;
            this.calculateTotals();
        },
        
        updateItemPrice(index, price) {
            this.form.items[index].unit_price = parseFloat(price) || 0;
            this.calculateTotals();
        },
        
        calculateTotals() {
            // 計算小計
            this.calculations.subtotal = this.form.items.reduce((sum, item) => {
                return sum + (item.quantity * item.unit_price);
            }, 0);
            
            // 計算折扣
            if (this.form.discount_type === 'percentage') {
                this.calculations.discount = this.calculations.subtotal * (this.form.discount_value / 100);
            } else if (this.form.discount_type === 'fixed') {
                this.calculations.discount = parseFloat(this.form.discount_value) || 0;
            } else {
                this.calculations.discount = 0;
            }
            
            // 計算應稅金額
            this.calculations.taxable_amount = this.calculations.subtotal - this.calculations.discount + parseFloat(this.form.shipping_cost || 0);
            
            // 計算稅金
            this.calculations.tax_amount = this.calculations.taxable_amount * (this.form.tax_rate / 100);
            
            // 計算總計
            this.calculations.total = this.calculations.taxable_amount + this.calculations.tax_amount;
        },
        
        formatNumber(number) {
            return parseFloat(number || 0).toLocaleString('en-US', {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            });
        }
    }
}
