### **開發環境快速啟動腳本**
```bash
#!/bin/bash
# 檔案: scripts/dev-setup.sh

set -e

echo "🚀 NexusERP 開發環境快速設定"

# 檢查必要工具
check_requirements() {
    echo "📋 檢查系統需求..."
    
    if ! command -v docker &> /dev/null; then
        echo "❌ Docker 未安裝，請先安裝 Docker"
        exit 1
    fi
    
    if ! command -v docker-compose &> /dev/null; then
        echo "❌ Docker Compose 未安裝，請先安裝 Docker Compose"
        exit 1
    fi
    
    if ! command -v node &> /dev/null; then
        echo "❌ Node.js 未安裝，請先安裝 Node.js"
        exit 1
    fi
    
    echo "✅ 系統需求檢查完成"
}

# 環境變數設定
setup_environment() {
    echo "🔧 設定環境變數..."
    
    if [ ! -f .env ]; then
        cp .env.example .env
        echo "📝 已建立 .env 檔案"
    fi
    
    # 生成應用程式金鑰
    if ! grep -q "APP_KEY=base64:" .env; then
        echo "🔑 生成應用程式金鑰..."
        APP_KEY=$(openssl rand -base64 32)
        sed -i.bak "s/APP_KEY=/APP_KEY=base64:$APP_KEY/" .env
    fi
    
    # 設定 JWT 密鑰
    if ! grep -q "JWT_SECRET=" .env; then
        echo "🔐 生成 JWT 密鑰..."
        JWT_SECRET=$(openssl rand -base64 32)
        echo "JWT_SECRET=$JWT_SECRET" >> .env
    fi
    
    echo "✅ 環境變數設定完成"
}

# Docker 服務啟動
start_services() {
    echo "🐳 啟動 Docker 服務..."
    
    # 停止現有服務
    docker-compose -f docker-compose.dev.yml down --remove-orphans
    
    # 建置並啟動服務
    docker-compose -f docker-compose.dev.yml up -d --build
    
    echo "⏳ 等待服務啟動..."
    sleep 30
    
    echo "✅ Docker 服務啟動完成"
}

# 應用程式初始化
initialize_app() {
    echo "🎯 初始化應用程式..."
    
    # 安裝 PHP 依賴
    docker-compose -f docker-compose.dev.yml exec app composer install
    
    # 安裝 Node.js 依賴
    docker-compose -f docker-compose.dev.yml exec app npm install
    
    # 資料庫遷移
    docker-compose -f docker-compose.dev.yml exec app php artisan migrate:fresh
    
    # 建立符號連結
    docker-compose -f docker-compose.dev.yml exec app php artisan storage:link
    
    # 清除快取
    docker-compose -f docker-compose.dev.yml exec app php artisan config:clear
    docker-compose -f docker-compose.dev.yml exec app php artisan cache:clear
    docker-compose -f docker-compose.dev.yml exec app php artisan view:clear
    
    echo "✅ 應用程式初始化完成"
}

# 開發工具設定
setup_dev_tools() {
    echo "🛠️ 設定開發工具..."
    
    # 安裝 IDE Helper
    docker-compose -f docker-compose.dev.yml exec app php artisan ide-helper:generate
    docker-compose -f docker-compose.dev.yml exec app php artisan ide-helper:models --nowrite
    docker-compose -f docker-compose.dev.yml exec app php artisan ide-helper:meta
    
    # 設定 Git hooks
    if [ -d .git ]; then
        cp scripts/git-hooks/* .git/hooks/
        chmod +x .git/hooks/*
        echo "🪝 Git hooks 已設定"
    fi
    
    echo "✅ 開發工具設定完成"
}

# 載入測試資料
load_demo_data() {
    echo "📊 載入測試資料..."
    
    docker-compose -f docker-compose.dev.yml exec app php artisan db:seed --class=DemoSeeder
    
    echo "✅ 測試資料載入完成"
}

# 驗證安裝
verify_installation() {
    echo "🔍 驗證安裝..."
    
    # 檢查應用程式狀態
    if curl -f http://localhost:8000/health > /dev/null 2>&1; then
        echo "✅ 應用程式運行正常"
    else
        echo "❌ 應用程式無法存取"
        exit 1
    fi
    
    # 檢查資料庫連線
    if docker-compose -f docker-compose.dev.yml exec app php artisan migrate:status > /dev/null 2>&1; then
        echo "✅ 資料庫連線正常"
    else
        echo "❌ 資料庫連線失敗"
        exit 1
    fi
    
    echo "✅ 安裝驗證完成"
}

# 顯示完成資訊
show_completion_info() {
    echo ""
    echo "🎉 NexusERP 開發環境設定完成！"
    echo ""
    echo "📍 服務存取資訊："
    echo "   • 應用程式: http://localhost:8000"
    echo "   • 資料庫管理: http://localhost:8080"
    echo "   • 郵件檢視: http://localhost:8025"
    echo "   • MinIO 控制台: http://localhost:9001"
    echo ""
    echo "🔧 常用指令："
    echo "   • 查看日誌: docker-compose -f docker-compose.dev.yml logs -f app"
    echo "   • 進入容器: docker-compose -f docker-compose.dev.yml exec app bash"
    echo "   • 執行測試: docker-compose -f docker-compose.dev.yml exec app php artisan test"
    echo "   • 停止服務: docker-compose -f docker-compose.dev.yml down"
    echo ""
    echo "📚 說明文件: https://docs.nexuserp.com/development"
    echo ""
}

# 主要執行流程
main() {
    check_requirements
    setup_environment
    start_services
    initialize_app
    setup_dev_tools
    
    # 詢問是否載入測試資料
    read -p "是否載入測試資料？ (y/N): " -n 1 -r
    echo
    if [[ $REPLY =~ ^[Yy]$ ]]; then
        load_demo_data
    fi
    
    verify_installation
    show_completion_info
}

# 執行主程序
main "$@"
```

#!/bin/bash
# 檔案: scripts/git-hooks/commit-msg

# Conventional Commits 規範驗證
commit_regex='^(feat|fix|docs|style|refactor|perf|test|chore|ci|build|revert)(\(.+\))?: .{1,50}'

if ! grep -qE "$commit_regex" "$1"; then
    echo "❌ 提交訊息格式錯誤！"
    echo ""
    echo "正確格式："
    echo "  <type>[optional scope]: <description>"
    echo ""
    echo "類型 (type)："
    echo "  feat:     新功能"
    echo "  fix:      錯誤修復"
    echo "  docs:     文件變更"

    echo "  refactor: 程式碼重構"

## 🚀 **CI/CD 最佳實踐**

### **GitHub Actions 工作流程**
```yaml
# 檔案: .github/workflows/ci.yml
name: CI/CD Pipeline

on:
  push:
    branches: [ main, develop, staging ]
  pull_request:
    branches: [ main, develop ]

env:
  PHP_VERSION: '8.2'
  NODE_VERSION: '18'

jobs:
  code-quality:
    name: Code Quality Checks
    runs-on: ubuntu-latest
    
    steps:
    - name: Checkout code
      uses: actions/checkout@v3
    
    - name: Setup PHP
      uses: shivammathur/setup-php@v2
      with:
        php-version: ${{ env.PHP_VERSION }}
        extensions: mbstring, dom, fileinfo, pgsql, redis
        tools: composer:v2
    
    - name: Cache Composer dependencies
      uses: actions/cache@v3
      with:
        path: vendor
        key: composer-${{ hashFiles('**/composer.lock') }}
        restore-keys: composer-
    
    - name: Install PHP dependencies
      run: composer install --no-progress --prefer-dist --optimize-autoloader
    
    - name: Run PHP CS Fixer
      run: vendor/bin/php-cs-fixer fix --dry-run --diff --verbose
    
    - name: Run PHPStan
      run: vendor/bin/phpstan analyse --memory-limit=2G
    
    - name: Run Psalm
      run: vendor/bin/psalm --output-format=github
    
    - name: Setup Node.js
      uses: actions/setup-node@v3
      with:
        node-version: ${{ env.NODE_VERSION }}
        cache: 'npm'
    
    - name: Install Node.js dependencies
      run: npm ci
    
    - name: Run ESLint
      run: npm run lint
    
    - name: Run Prettier
      run: npm run format:check

  security-scan:
    name: Security Scan
    runs-on: ubuntu-latest
    
    steps:
    - name: Checkout code
      uses: actions/checkout@v3
    
    - name: Setup PHP
      uses: shivammathur/setup-php@v2
      with:
        php-version: ${{ env.PHP_VERSION }}
    
    - name: Install dependencies
      run: composer install --no-progress --prefer-dist --optimize-autoloader
    
    - name: Run Security Checker
      run: |
        composer require --dev enlightn/security-checker
        vendor/bin/security-checker security:check
    
    - name: Run Semgrep
      uses: returntocorp/semgrep-action@v1
      with:
        config: >-
          p/security-audit
          p/secrets
          p/owasp-top-ten
    
    - name: Upload Semgrep results to GitHub
      if: always()
      uses: github/codeql-action/upload-sarif@v2
      with:
        sarif_file: semgrep.sarif

  test:
    name: Tests
    runs-on: ubuntu-latest
    needs: code-quality
    
    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: nexus_erp_test
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
      
      redis:
        image: redis:7
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    
    steps:
    - name: Checkout code
      uses: actions/checkout@v3
    
    - name: Setup PHP
      uses: shivammathur/setup-php@v2
      with:
        php-version: ${{ env.PHP_VERSION }}
        extensions: mbstring, dom, fileinfo, pgsql, redis
        coverage: xdebug
    
    - name: Install dependencies
      run: |
        composer install --no-progress --prefer-dist --optimize-autoloader
        npm ci && npm run build
    
    - name: Setup environment
      run: |
        cp .env.testing .env
        php artisan key:generate
        php artisan config:cache
    
    - name: Run migrations
      run: php artisan migrate --force
    
    - name: Run tests
      run: |
        php artisan test --coverage --min=80 --coverage-clover=coverage.xml
    
    - name: Upload coverage to Codecov

<?php
// 檔案: app/Console/Commands/GenerateApiDocs.php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use ReflectionClass;
use ReflectionMethod;

class GenerateApiDocs extends Command
{
    protected $signature = 'docs:generate {--format=openapi : 文件格式}';
    protected $description = '自動生成 API 文件';
    
    public function handle(): int
    {
        $this->info('正在生成 API 文件...');
        
        $routes = $this->getApiRoutes();
        $documentation = $this->generateDocumentation($routes);
        
        $format = $this->option('format');
        $output = $this->formatOutput($documentation, $format);
        
        $this->saveDocumentation($output, $format);
        
        $this->info('API 文件生成完成！');
        
        return 0;
    }
    
    private function getApiRoutes(): array
    {
        $routes = [];
        
        foreach (\Route::getRoutes() as $route) {
            if (str_starts_with($route->uri(), 'api/')) {
                $routes[] = $this->analyzeRoute($route);
            }
        }
        
        return $routes;
    }
    
    private function analyzeRoute($route): array
    {
        $controller = $route->getController();
        $method = $route->getActionMethod();
        
        $reflection = new ReflectionMethod($controller, $method);
        $docComment = $reflection->getDocComment();
        
        return [
            'uri' => $route->uri(),
            'methods' => $route->methods(),
            'controller' => get_class($controller),
            'method' => $method,
            'middleware' => $route->middleware(),
            'parameters' => $this->extractParameters($route),
            'documentation' => $this->parseDocComment($docComment),
            'responses' => $this->generateResponseExamples($controller, $method),
        ];
    }
}
```
